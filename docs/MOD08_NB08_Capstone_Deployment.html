Module 08 - Reproducible Research & Deployment

Capstone · Full Deployment Pipeline

This capstone integrates NB-01 through NB-07 into a complete production-ready clinical ML deployment.

Pipeline stages: 1. Reproducible data provenance (checksums, MANIFEST.json) 2. Multi-model training with run logging 3. Automated model validation (unit tests + performance tests) 4. Fairness analysis across demographic subgroups 5. Model Card generation 6. plumber endpoint simulation 7. Publication-quality 8-panel summary figure (300 dpi) 8. Deployment readiness checklist

Research context: 30-day hospital readmission prediction — from raw data to production API

Estimated time: 4 hours | Level: Advanced

avg_precision <- function(y, p) {
  ord <- order(p, decreasing = TRUE)
  y <- y[ord]; tp <- cumsum(y); prec <- tp / seq_along(y); rec <- tp / sum(y)
  sum(diff(c(0, rec)) * prec)
}
brier_score <- function(y, p) mean((p - y)^2)

1. Training Run Log & Model Selection

predict_prob <- function(fit, X, type) {
  if (type == "glm") {
    as.numeric(predict(fit, newdata = X, type = "response"))
  } else if (type == "ranger") {
    pred <- predict(fit, data = X)$predictions
    if (is.matrix(pred)) {
      cn <- colnames(pred)
      pred[, cn == "1" | cn == "TRUE"]
    } else pred
  } else {
    predict(fit, as.matrix(X))
  }
}

models_config <- list(

simulate_api <- function(age, los_days, diabetes, hf, copd, patient_id = NULL) {
  if (!all(map_lgl(list(age, los_days, diabetes, hf, copd), is.numeric))) {
    return(list(error = "422 Unprocessable Entity", detail = "Invalid field types"))
  }
  if (!(18 <= age && age <= 90)) return(list(error = "422", detail = sprintf("age must be 18-90, got %s", age)))
  if (!(1 <= los_days && los_days <= 60)) return(list(error = "422", detail = "los_days must be 1-60"))
  X <- tibble(age = age, los_days = los_days, diabetes = diabetes,
              hf = hf, copd = copd, n_comorb = diabetes + hf + copd)
  prob <- as.numeric(predict_prob(best_model, X, best_type))[1]
  tier <- if (prob < 0.10) "Low" else if (prob < 0.20) "Moderate" else if (prob < 0.35) "High" else "Very High"
  list(
    patient_id = patient_id,
    readmission_probability = round(prob, 4),
    risk_tier = tier,
    model_version = "1.0.0",
    timestamp = format(Sys.time(), "%Y-%m-%dT%H:%M:%S")
  )
}

Health Analytics with R · Module 05: NLP for Clinical Text

Learning objectives - Understand NER entity types in clinical NLP - Build a rule-based gazetteer pipeline with 180+ clinical patterns - Apply negation-aware NER with context windows - Evaluate NER quality with precision, recall, and F1 - Visualise entity type distributions

Estimated time: 2.5 hours | Level: Intermediate → Advanced | Libraries: quanteda, stringr, dplyr, ggplot2

spaCyR note. Python tutorials often use spaCy’s EntityRuler. The same teaching goal — dictionary/gazetteer NER over medications, diagnoses, findings, and labs — is implemented here with stringr + a clinical lexicon. The spaCyR package can wrap a spaCy Python process from R if you already have a trained model; it is optional for this notebook.

3. Gazetteer EntityRuler Pipeline

This gazetteer is the R analogue of spaCy’s EntityRuler: the same 180+ patterns are compiled into a dictionary lookup, plus case variants, then applied left-to-right keeping the longest non-overlapping span.

# Build a dictionary of (surface form → label), longest first
build_gazetteer <- function(entity_lists) {
  rows <- purrr::imap_dfr(entity_lists, function(patterns, label) {
    tibble::tibble(
      pattern = unique(c(patterns, tolower(patterns), stringr::str_to_title(patterns))),
      label   = label
    )
  })
  dplyr::arrange(rows, dplyr::desc(nchar(pattern)))
}

gazetteer <- build_gazetteer(entity_lists)

gazetteer_ner <- function(text, gazetteer) {
  entities <- list()
  for (i in seq_len(nrow(gazetteer))) {
    pattern <- gazetteer$pattern[i]
    label   <- gazetteer$label[i]
    regex   <- paste0("\\b", stringr::str_replace_all(pattern, "([\\W])", "\\\\\\1"), "\\b")
    m <- gregexpr(regex, text, perl = TRUE, ignore.case = TRUE)[[1]]
    if (m[1] == -1) next
    matched <- regmatches(text, list(m))[[1]]
    ends <- as.integer(m + attr(m, "match.length"))
    for (j in seq_along(m)) {
      entities[[length(entities) + 1]] <- list(
        text = matched[j], label = label,
        start = as.integer(m[j]), end = ends[j]
      )
    }
  }
  if (length(entities) == 0) {
    return(tibble::tibble(text = character(), label = character(),
                          start = integer(), end = integer()))
  }
  ents <- dplyr::bind_rows(entities) |>
    dplyr::arrange(start, dplyr::desc(end - start))
  filtered <- list(); prev_end <- -1L
  for (k in seq_len(nrow(ents))) {
    if (ents$start[k] >= prev_end) {
      filtered[[length(filtered) + 1]] <- ents[k, ]
      prev_end <- ents$end[k]
    }
  }
  dplyr::bind_rows(filtered)
}

regex_ner <- function(text, entity_lists) {
  entities <- purrr::imap_dfr(entity_lists, function(patterns, label) {
    purrr::map_dfr(patterns, function(pattern) {
      regex <- paste0("\\b", stringr::str_replace_all(pattern, "([\\W])", "\\\\\\1"), "\\b")
      m <- gregexpr(regex, text, perl = TRUE, ignore.case = TRUE)[[1]]
      if (m[1] == -1) {
        return(tibble::tibble(text = character(), label = character(),
                              start = integer(), end = integer()))
      }
      tibble::tibble(
        text  = regmatches(text, list(m))[[1]],
        label = label,
        start = as.integer(m),
        end   = as.integer(m + attr(m, "match.length"))
      )
    })
  })
  if (nrow(entities) == 0) return(entities)
  entities <- dplyr::arrange(entities, start, dplyr::desc(end - start))
  filtered <- list(); prev_end <- -1L
  for (k in seq_len(nrow(entities))) {
    if (entities$start[k] >= prev_end) {
      filtered[[length(filtered) + 1]] <- entities[k, ]
      prev_end <- entities$end[k]
    }
  }
  dplyr::bind_rows(filtered)
}

5. Negation-Aware NER

negation_triggers <- c(
  "no ", "not ", "denies ", "denied ", "without ", "w/o ",
  "never ", "absent ", "negative ", "no evidence of ", "rules out "
)

ner_with_negation <- function(text, entity_lists) {
  entities <- purrr::imap_dfr(entity_lists, function(patterns, label) {
    purrr::map_dfr(patterns, function(pattern) {
      regex <- paste0("\\b", stringr::str_replace_all(pattern, "([\\W])", "\\\\\\1"), "\\b")
      m <- gregexpr(regex, text, perl = TRUE, ignore.case = TRUE)[[1]]
      if (m[1] == -1) {
        return(tibble::tibble())
      }
      matched <- regmatches(text, list(m))[[1]]
      purrr::map_dfr(seq_along(m), function(j) {
        start <- as.integer(m[j])
        ctx_lo <- max(1, start - 70)
        context <- tolower(substr(text, ctx_lo, start - 1))
        tibble::tibble(
          text    = matched[j],
          label   = label,
          start   = start,
          end     = as.integer(m[j] + attr(m, "match.length")[j]),
          negated = any(purrr::map_lgl(negation_triggers, ~ grepl(.x, context, fixed = TRUE))),
          context = stringr::str_squish(substr(context, max(1, nchar(context) - 39), nchar(context)))
        )
      })
    })
  })
  if (nrow(entities) == 0) return(entities)
  dplyr::arrange(entities, start)
}

6. NER Evaluation

evaluate_ner <- function(gold_list, pred_df) {
  gold_text <- unique(purrr::map_chr(gold_list, ~ paste(tolower(.x[1]), .x[2], sep = "||")))
  pred_text <- unique(paste(tolower(pred_df$text), pred_df$label, sep = "||"))
  parse_pair <- function(x) {
    parts <- strsplit(x, "\\|\\|", perl = TRUE)
    tibble::tibble(
      span  = vapply(parts, `[`, character(1), 1),
      etype = vapply(parts, `[`, character(1), 2)
    )
  }
  gold_df <- parse_pair(gold_text)
  pred_p  <- parse_pair(pred_text)
  all_types <- union(gold_df$etype, pred_p$etype)
  purrr::map_dfr(all_types, function(etype) {
    g <- gold_df$span[gold_df$etype == etype]
    p <- pred_p$span[pred_p$etype == etype]
    tp <- length(intersect(g, p))
    fp <- length(setdiff(p, g))
    fn <- length(setdiff(g, p))
    prec <- if ((tp + fp) > 0) tp / (tp + fp) else 0
    rec  <- if ((tp + fn) > 0) tp / (tp + fn) else 0
    f1   <- if ((prec + rec) > 0) 2 * prec * rec / (prec + rec) else 0
    tibble::tibble(
      type = etype, TP = tp, FP = fp, FN = fn,
      Precision = round(prec, 3), Recall = round(rec, 3), F1 = round(f1, 3)
    )
  })
}

extract_lab_values <- function(text) {
  m <- stringr::str_match_all(text, stringr::regex(lab_value_pat, ignore_case = TRUE))[[1]]
  if (nrow(m) == 0) {
    return(tibble::tibble(analyte = character(), value = character(),
                          unit = character(), label = character()))
  }
  tibble::tibble(
    analyte = m[, 2],
    value   = m[, 3],
    unit    = ifelse(is.na(m[, 4]), "", stringr::str_squish(m[, 4])),
    label   = "LAB_VALUE"
  )
}

      cat(sprintf("  %-15s = %s %s\n", analyte, value, unit))
    })

}

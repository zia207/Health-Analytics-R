spatial_smooth <- function(values, row_idx, col_idx, nrow, ncol, sigma = 2) {
  grid <- matrix(0, nrow = nrow, ncol = ncol)
  for (i in seq_along(values)) grid[row_idx[i], col_idx[i]] <- values[i]
  k <- ceiling(3 * sigma)
  xs <- (-k):k
  kern <- outer(dnorm(xs, sd = sigma), dnorm(xs, sd = sigma))
  kern <- kern / sum(kern)
  pad <- k
  gp <- matrix(0, nrow + 2 * pad, ncol + 2 * pad)
  gp[(pad + 1):(pad + nrow), (pad + 1):(pad + ncol)] <- grid
  out <- matrix(0, nrow, ncol)
  for (r in seq_len(nrow)) for (c in seq_len(ncol)) {
    out[r, c] <- sum(gp[r:(r + 2 * k), c:(c + 2 * k)] * kern)
  }
  mapply(function(r, c) out[r, c], row_idx, col_idx)
}

2. OLS Baseline and Diagnostic Tests

OLS model: CVD_i = β₀ + β₁·Poverty_i + β₂·Uninsured_i + β₃·PM25_i + ε_i

Spatial diagnostics: - Moran’s I on residuals: global test for residual autocorrelation - Lagrange Multiplier (LM) Lag test: tests if spatial lag of Y belongs in model - LM Error test: tests if spatial error structure is needed - Robust LM tests: account for presence of the other spatial effect

map_beta <- function(cov, true_b) {
  col <- paste0("gwr_beta_", cov)
  ggplot(gdf) +
    geom_sf(aes(fill = .data[[col]]), colour = "white", linewidth = 0.15) +
    scale_fill_distiller(name = "Local coefficient", palette = "RdBu", direction = -1) +
    labs(

      subtitle = sprintf("Global OLS β=%.3f | True β=%.1f", ols_b[cov], true_b),
      x = "Lon", y = "Lat"
    )
}

Health Analytics with R · Module 05: NLP for Clinical Text

This capstone integrates NB-01 through NB-07 into a single production-ready clinical NLP pipeline.

Pipeline stages: 1. Preprocessing: PHI de-identification, abbreviation expansion, section extraction 2. Named entity recognition: conditions, drugs, findings, symptoms 3. Structured value extraction: vitals, labs, medications 4. ICD-10 code suggestions 5. TF-IDF readmission prediction from note text 6. Hybrid features: TF-IDF + NLP-extracted values 7. Extractive summarisation + ROUGE evaluation 8. 8-panel publication figure at 300 dpi + methods narrative

Research question: Can NLP-derived features from discharge summaries improve 30-day readmission prediction over structured EHR features alone?

Estimated time: 4 hours | Level: Advanced

# ── Abbreviation expansion (NB-01) ───────────────────────────────────────────
abbrevs <- c(
  DM2 = "type 2 diabetes", HTN = "hypertension", HF = "heart failure",
  CKD = "chronic kidney disease", COPD = "chronic obstructive pulmonary disease",
  PE = "pulmonary embolism", SOB = "shortness of breath", CP = "chest pain",
  HR = "heart rate", BP = "blood pressure", SpO2 = "oxygen saturation",
  EF = "ejection fraction", BNP = "brain natriuretic peptide",
  PO = "by mouth", IV = "intravenous", BID = "twice daily", QD = "once daily",
  Pt = "patient", pt = "patient"
)
expand_abbrevs <- function(text) {
  ord <- order(nchar(names(abbrevs)), decreasing = TRUE)
  for (i in ord) {
    pattern <- paste0("\\b", names(abbrevs)[i], "\\b")
    text <- stringr::str_replace_all(text, stringr::regex(pattern), unname(abbrevs[i]))
  }
  text
}

# ── PHI de-identification (NB-01) ───────────────────────────────────────────
phi_pats <- c(
  MRN  = "MRN\\s*:\\s*\\d{6,10}",
  DATE = "\\b\\d{1,2}/\\d{1,2}/\\d{4}\\b",
  AGE  = "\\b\\d{2,3}\\s*(?:y/o|year[-\\s]old|yo)\\b"
)
deidentify <- function(text) {
  for (t in names(phi_pats)) {
    text <- stringr::str_replace_all(
      text, stringr::regex(phi_pats[[t]], ignore_case = TRUE), paste0("[", t, "]")
    )
  }
  text
}

# ── Vital/lab extraction (NB-01) ─────────────────────────────────────────────
vital_pats <- c(
  hr   = "HR\\s*:?\\s*(\\d{2,3})",
  sbp  = "BP\\s*:?\\s*(\\d{2,3})/(\\d{2,3})",
  spo2 = "SpO2\\s*:?\\s*(\\d{2,3})\\s*%",
  bnp  = "BNP\\s*(\\d{2,5})",
  cr   = "Cr\\s*(\\d\\.\\d)",
  ef   = "EF\\s*(\\d{2})\\s*%"
)
extract_vitals <- function(text) {
  vals <- list()
  for (name in names(vital_pats)) {
    m <- stringr::str_match(text, stringr::regex(vital_pats[[name]], ignore_case = TRUE))
    if (!is.na(m[1, 1])) {
      vals[[name]] <- as.numeric(if (name == "sbp") m[1, 2] else m[1, 2])
    }
  }
  vals
}

# ── NER (NB-02) ──────────────────────────────────────────────────────────────
entity_pats <- list(
  DISEASE = c("heart failure", "diabetes", "COPD", "sepsis", "pneumonia",
              "pulmonary embolism", "DKA", "ketoacidosis", "hypertension",
              "acute kidney injury", "myocardial infarction", "atrial fibrillation"),
  DRUG    = c("furosemide", "metformin", "insulin", "albuterol", "vancomycin",
              "carvedilol", "lisinopril", "sacubitril", "methylprednisolone",
              "ceftriaxone", "rivaroxaban", "aspirin", "spironolactone"),
  FINDING = c("bilateral pitting edema", "reduced ejection fraction", "hypercapnia",
              "volume overload", "bilateral infiltrates", "anion gap", "ST elevation")
)
count_ents <- function(text) {
  counts <- list()
  for (label in names(entity_pats)) {
    n <- 0L
    for (p in entity_pats[[label]]) {
      regex <- paste0("\\b", stringr::str_replace_all(p, "([\\W])", "\\\\\\1"), "\\b")
      if (stringr::str_detect(text, stringr::regex(regex, ignore_case = TRUE))) n <- n + 1L
    }
    counts[[label]] <- n
  }
  counts
}

icd_texts <- paste(icd10_ref$kw, icd10_ref$desc)
toks_icd <- quanteda::tokens(icd_texts, remove_punct = TRUE) |>
  quanteda::tokens_tolower() |>
  quanteda::tokens_ngrams(n = 1:2)
dfm_icd <- quanteda::dfm_tfidf(quanteda::dfm(toks_icd))
if (nfeat(dfm_icd) > 200) {
  keep <- names(quanteda::topfeatures(dfm_icd, n = 200))
  dfm_icd <- quanteda::dfm_select(dfm_icd, pattern = keep)
}
icd_mat <- as.matrix(dfm_icd)

suggest_top1 <- function(text) {
  toks <- quanteda::tokens(tolower(text), remove_punct = TRUE) |>
    quanteda::tokens_ngrams(n = 1:2)
  d <- quanteda::dfm_match(quanteda::dfm(toks), features = colnames(icd_mat))
  vec <- as.numeric(as.matrix(d))
  nrms <- sqrt(rowSums(icd_mat^2)); nrms[nrms == 0] <- 1
  vn <- sqrt(sum(vec^2)); if (vn == 0) vn <- 1
  scores <- as.numeric((icd_mat / nrms) %*% (vec / vn))
  icd10_ref$code[which.max(scores)]
}

make_tfidf <- function(texts, train_dfm = NULL) {
  texts <- as.character(texts)
  names(texts) <- paste0("doc", seq_along(texts))
  toks <- quanteda::tokens(texts, remove_punct = TRUE) |>
    quanteda::tokens_tolower() |>
    quanteda::tokens_remove(stops) |>
    quanteda::tokens_ngrams(n = 1:2)
  d <- quanteda::dfm(toks)
  if (is.null(train_dfm)) {
    keep_n <- min(500L, nfeat(d))
    keep <- names(quanteda::topfeatures(d, n = keep_n))
    d <- quanteda::dfm_select(d, pattern = keep)
    quanteda::dfm_tfidf(d)
  } else {
    d <- quanteda::dfm_match(d, features = quanteda::featnames(train_dfm))
    quanteda::dfm_tfidf(d)
  }
}

impute_median <- function(x) {
  x[is.na(x)] <- median(x, na.rm = TRUE)
  x
}
scale_mat <- function(tr, te) {
  tr <- apply(tr, 2, impute_median)
  te <- apply(te, 2, impute_median)
  mu <- colMeans(tr); sdv <- apply(tr, 2, sd); sdv[sdv == 0] <- 1
  list(tr = scale(tr, center = mu, scale = sdv),
       te = scale(te, center = mu, scale = sdv))
}

fit_eval <- function(xtr, xte) {
  fit <- glmnet::glmnet(xtr, y_tr, family = "binomial",
                        lambda = 0.01, maxit = 1000)
  p <- as.numeric(predict(fit, newx = xte, type = "response"))
  pred <- as.integer(p >= 0.5)
  tibble::tibble(
    AUC = yardstick::roc_auc_vec(factor(y_te, levels = c(0, 1)), p, event_level = "second"),
    AP  = yardstick::pr_auc_vec(factor(y_te, levels = c(0, 1)), p, event_level = "second"),
    F1  = yardstick::f_meas_vec(factor(y_te, levels = c(0, 1)),

}

5. Extraction Rate Analysis

stops_summ <- c(quanteda::stopwords("en"),
                "patient", "mg", "daily", "was", "were", "the", "a", "an",
                "admission", "discharge", "hospital")

split_sentences <- function(text) {
  sents <- unlist(quanteda::tokens(text, what = "sentence"))
  sents <- stringr::str_squish(sents)
  n_words <- vapply(strsplit(sents, "\\s+"), length, integer(1))
  sents[n_words > 5]
}

textrank_summarise <- function(text, n_sents = 4) {
  sents <- split_sentences(text)
  if (length(sents) <= n_sents) return(sents)
  toks <- quanteda::tokens(sents, remove_punct = TRUE) |>
    quanteda::tokens_tolower() |>
    quanteda::tokens_remove(stops_summ)
  d <- quanteda::dfm_tfidf(quanteda::dfm(toks))
  mat <- as.matrix(d)
  if (nrow(mat) == 0 || ncol(mat) == 0) return(sents[seq_len(n_sents)])
  nrms <- sqrt(rowSums(mat^2)); nrms[nrms == 0] <- 1
  sim <- tcrossprod(mat / nrms); diag(sim) <- 0
  n <- nrow(sim)
  scores <- rep(1 / n, n)
  for (iter in seq_len(50)) {
    rs <- rowSums(sim); rs[rs == 0] <- 1
    scores <- 0.15 / n + 0.85 * as.numeric(t(sim / rs) %*% scores)
  }
  sents[sort(order(scores, decreasing = TRUE)[seq_len(n_sents)])]
}

8. Methods Narrative (Template)

Clinical NLP Methods

Discharge summaries (N=400) were preprocessed through a rule-based pipeline: PHI de-identification (regex for MRN, dates, ages), clinical abbreviation expansion (20-term dictionary), and section header extraction (8 section types).

Named entities were identified using a lexicon-based gazetteer (EntityRuler analogue) with 40+ clinical patterns across 3 entity types (DISEASE, DRUG, FINDING) with window-based negation detection (5-token look-back window). Structured values (BNP, creatinine, EF, SpO2) were extracted using clinical measurement regex patterns.

ICD-10 code suggestions were generated via cosine similarity between TF-IDF representations of note text and ICD-10 keyword descriptions (top-1 accuracy: XX%). Medication extraction used a drug name lexicon (80+ drugs) with dose/route/frequency capture patterns.

Readmission prediction compared three feature sets: (1) structured NLP-extracted values only, (2) TF-IDF unigram+bigram features, and (3) hybrid TF-IDF + structured — all trained with logistic regression (glmnet, balanced via intercept). Extractive summarisation used TextRank with cosine-similarity-based sentence scoring.

All analyses used R 4.3+ (quanteda, tidytext, glmnet, text2vec, stringr).

Module 03 — Linear Regression for Continuous Outcomes

Linear Regression for Continuous Outcomes

Learning objectives - Fit and interpret linear regression models for LOS and charges - Assess regression assumptions (linearity, normality of residuals, homoscedasticity, independence) - Apply transformations for skewed outcomes (log, Box-Cox) - Interpret coefficients for continuous and categorical predictors - Use model diagnostics plots: residual vs fitted, QQ plot, Cook’s distance

Estimated time: 2 hours | Level: Intermediate | Libraries: stats, broom, ggplot2, dplyr, MASS

mod03_dir <- file.path(tempdir(), "mod03")
dir.create(mod03_dir, showWarnings = FALSE, recursive = TRUE)

logistic <- function(x) 1 / (1 + exp(-x))

set.seed(42)
N <- 1200
base <- matrix(rnorm(N * 3), ncol = 3)

set.seed(99)
ages          <- as.integer(pmin(pmax(rnorm(N, 62, 16), 18), 95))
sexes         <- sample(c("M", "F"), N, replace = TRUE, prob = c(0.48, 0.52))
payers        <- sample(
  c("Medicare", "Medicaid", "Private", "Self-pay", "Other"),
  N, replace = TRUE, prob = c(0.40, 0.22, 0.28, 0.07, 0.03)
)
diabetes      <- rbinom(N, 1, logistic(0.6 * base[, 1] - 0.5))
heart_failure <- rbinom(N, 1, logistic(0.4 * base[, 2] - 1.2))
copd          <- rbinom(N, 1, logistic(0.5 * base[, 3] - 1.0))
admit_type    <- sample(c("Emergency", "Elective", "Urgent"), N, replace = TRUE,
                        prob = c(0.52, 0.30, 0.18))
n_comorb      <- diabetes + heart_failure + copd

# LOS with realistic predictors (right-skewed)
los_true <- (2.5 + 1.5 * heart_failure + 0.8 * diabetes + 0.5 * copd
             + 0.05 * ages + 0.6 * as.integer(admit_type == "Emergency")
             + rexp(N, rate = 1 / 1.5))
los_days <- as.integer(pmin(pmax(round(los_true), 1), 30))

# Total charge ~ LOS * daily rate (correlated)
charges <- pmax(
  round(los_days * runif(N, 1800, 4200) + 2000 * heart_failure + 1500 * diabetes + rnorm(N, 0, 3000), 2),
  500
)

df <- tibble(
  patient_id    = sprintf("PT-%05d", seq_len(N)),
  age           = ages,
  sex           = sexes,
  payer         = relevel(factor(payers), ref = "Private"),
  admit_type    = relevel(factor(admit_type), ref = "Elective"),
  diabetes      = diabetes,
  heart_failure = heart_failure,
  copd          = copd,
  n_comorb      = n_comorb,
  los_days      = los_days,
  total_charge  = charges
) |>
  mutate(
    age_std    = as.numeric(scale(age)),
    log_charge = log(total_charge),
    log_los    = log(los_days)
  )

cat(sprintf("Dataset: %d x %d\n", nrow(df), ncol(df)))

4. Regression Assumption Diagnostics

aug_m <- broom::augment(model_multi)
cooks_d <- cooks.distance(model_multi)
std_resid <- rstandard(model_multi)
threshold <- 4 / nrow(df)
n_inf <- sum(cooks_d > threshold, na.rm = TRUE)

# LOWESS trend for residuals vs fitted
lowess_rf <- lowess(aug_m$.fitted, aug_m$.resid, f = 0.3)
lowess_sl <- lowess(aug_m$.fitted, sqrt(abs(std_resid)), f = 0.3)

p_diag_a <- ggplot(aug_m, aes(.fitted, .resid)) +
  geom_point(alpha = 0.2, size = 1.2, colour = "#2563EB") +
  geom_hline(yintercept = 0, colour = "#DC2626", linetype = "dashed") +
  geom_line(data = tibble(x = lowess_rf$x, y = lowess_rf$y),
            aes(x, y), inherit.aes = FALSE, colour = "orange", linewidth = 1) +
  labs(x = "Fitted values", y = "Residuals",
       title = "A  Residuals vs Fitted\n(should be random scatter around 0)")

p_diag_b <- ggplot(aug_m, aes(sample = .resid)) +
  stat_qq(alpha = 0.4, size = 1, colour = "#2563EB") +
  stat_qq_line(colour = "#DC2626", linewidth = 0.8) +
  labs(x = "Theoretical quantiles", y = "Sample quantiles",
       title = "B  QQ Plot of Residuals")

p_diag_c <- ggplot(tibble(fitted = aug_m$.fitted, sl = sqrt(abs(std_resid))),
                   aes(fitted, sl)) +
  geom_point(alpha = 0.2, size = 1.2, colour = "#16A34A") +
  geom_line(data = tibble(x = lowess_sl$x, y = lowess_sl$y),
            aes(x, y), inherit.aes = FALSE, colour = "orange", linewidth = 1) +
  labs(x = "Fitted values", y = "√|Standardised residuals|",
       title = "C  Scale-Location (homoscedasticity)\n(should be a flat line)")

p_diag_d <- ggplot(tibble(idx = seq_along(cooks_d), cooks = as.numeric(cooks_d)),
                   aes(idx, cooks)) +
  geom_linerange(aes(ymin = 0, ymax = cooks), colour = "#93C5FD") +
  geom_hline(yintercept = threshold, colour = "#DC2626", linetype = "dashed") +
  labs(
    x = "Observation index", y = "Cook's distance",
    title = sprintf("D  Cook's Distance — Influential Points\n(%d observations above threshold)", n_inf)
  )

p_diag <- (p_diag_a + p_diag_b) / (p_diag_c + p_diag_d) +
  plot_annotation(title = "Regression diagnostics — LOS multivariable model")
print(p_diag)
ggsave(file.path(mod03_dir, "regression_diagnostics.png"), p_diag,
       width = 13, height = 10, dpi = 120)

6. Model Comparison with AIC/BIC

# Compare models
glance_row <- function(model, name) {
  g <- broom::glance(model)
  tibble(
    Model          = name,
    `R²`           = g$r.squared,
    `Adj R²`       = g$adj.r.squared,
    AIC            = g$AIC,
    BIC            = g$BIC,
    `N predictors` = length(coef(model)) - 1
  )
}

results_comparison <- bind_rows(
  glance_row(model_simple, "Simple (age only)"),
  glance_row(model_multi,  "Multi (LOS outcome)"),
  glance_row(model_log,    "Log-charge model")
) |>
  mutate(across(where(is.numeric), ~ round(.x, 2)))

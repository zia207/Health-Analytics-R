Health Analytics with R · Module 04: Machine Learning for Clinical Prediction

Learning objectives - Build, visualise and interpret a clinical decision tree - Understand bias-variance tradeoff via tree depth experiments - Fit a Random Forest and tune hyperparameters (trees, tree_depth, min_n) - Extract and plot feature importances (MDI and permutation) - Compare Decision Tree vs Random Forest on clinical holdout

Estimated time: 2 hours | Level: Intermediate | Libraries: tidymodels, rpart, rpart.plot, ranger, dplyr, ggplot2

auc_prob   <- function(y, p) yardstick::roc_auc_vec(factor(y, levels = c(0, 1)), p, event_level = "second")
ap_prob    <- function(y, p) yardstick::pr_auc_vec(factor(y, levels = c(0, 1)), p, event_level = "second")
brier_prob <- function(y, p) mean((p - y)^2)

# ── Dataset (same seed as NB-01) ──────────────────────────────────────────────
set.seed(42); N <- 3000
sig <- function(x) 1 / (1 + exp(-x))
clip <- function(x, lo, hi) pmin(pmax(x, lo), hi)
b <- matrix(rnorm(N * 4), ncol = 4); set.seed(99)
age <- as.integer(clip(rnorm(N, 62, 15), 18, 92))
sex <- sample(c("M", "F"), N, replace = TRUE, prob = c(0.48, 0.52))
payer <- sample(c("Medicare", "Medicaid", "Private", "Self-pay", "Other"),
                N, replace = TRUE, prob = c(0.40, 0.22, 0.28, 0.07, 0.03))
admit_type <- sample(c("Emergency", "Elective", "Urgent"),
                     N, replace = TRUE, prob = c(0.52, 0.30, 0.18))
los_days <- as.integer(clip(round(rgamma(N, shape = 2.5, scale = 1.8)), 1, 30))
diabetes <- rbinom(N, 1, sig(0.6 * b[, 1] - 0.5 + 0.02 * (age - 62) / 15))
hypertension <- rbinom(N, 1, sig(0.7 * b[, 1] - 0.2))
hf <- rbinom(N, 1, sig(0.4 * b[, 2] + 0.5 * hypertension - 1.5))
copd <- rbinom(N, 1, sig(0.5 * b[, 3] - 1.0))
ckd <- rbinom(N, 1, sig(0.5 * b[, 1] + 0.6 * diabetes - 1.2))
obesity <- rbinom(N, 1, sig(0.5 * b[, 1] - 0.8))
depression <- rbinom(N, 1, sig(0.3 * b[, 4] - 1.4))
n_comorb <- diabetes + hypertension + hf + copd + ckd
set.seed(21)
glucose <- round(clip(rnorm(N, 105 + 15 * diabetes, 28), 50, 400), 1)
creatinine <- round(clip(rgamma(N, shape = 1.6 + 0.8 * hf, scale = 0.6), 0.4, 10), 2)
hba1c <- round(clip(rnorm(N, 6.5 + 0.8 * diabetes, 1.4), 4, 14), 1)
sbp <- round(clip(rnorm(N, 128 + 12 * hypertension, 18), 80, 200), 0)
bmi <- round(clip(rnorm(N, 28 + 4 * obesity, 6), 15, 55), 1)
logit <- (-3.2 + 0.6 * hf + 0.5 * diabetes + 0.5 * ckd + 0.3 * copd +
            0.02 * (age - 62) / 15 +
            0.3 * as.integer(admit_type == "Emergency") +
            0.25 * as.integer(payer == "Medicaid") +
            0.2 * as.integer(los_days > 7) +
            0.5 * hf * diabetes + rnorm(N, 0, 0.25))
readmitted <- rbinom(N, 1, sig(logit))
df <- tibble(
  age, sex, payer, admit_type, los_days,
  diabetes, hypertension, hf, copd, ckd, obesity, depression, n_comorb,
  glucose, creatinine, hba1c, sbp, bmi, readmitted
)
df$los_gt7 <- as.integer(df$los_days > 7)
NUMERIC <- c("age", "los_days", "n_comorb", "glucose", "creatinine", "hba1c", "sbp", "bmi")
BINARY <- c("diabetes", "hypertension", "hf", "copd", "ckd", "obesity", "depression", "los_gt7")
CATEGORIC <- c("sex", "payer", "admit_type")
FEATURES <- c(NUMERIC, BINARY, CATEGORIC); TARGET <- "readmitted"

# ── Preprocessing + split ─────────────────────────────────────────────────────
set.seed(42)
df_ml <- df %>%
  select(all_of(c(FEATURES, TARGET))) %>%
  mutate(readmitted = factor(readmitted, levels = c(0, 1)))
split <- initial_split(df_ml, prop = 0.80, strata = readmitted)
df_tr <- training(split); df_te <- testing(split)
y_tr <- as.integer(as.character(df_tr$readmitted))
y_te <- as.integer(as.character(df_te$readmitted))

rec <- recipe(readmitted ~ ., data = df_tr) %>%
  step_impute_median(all_of(NUMERIC)) %>%
  step_normalize(all_of(NUMERIC)) %>%
  step_impute_median(all_of(BINARY)) %>%
  step_impute_mode(all_of(CATEGORIC)) %>%
  step_dummy(all_of(CATEGORIC), one_hot = TRUE)
rec_prep <- prep(rec, training = df_tr)
Xtr_tbl <- bake(rec_prep, new_data = df_tr) %>% select(-readmitted)
Xte_tbl <- bake(rec_prep, new_data = df_te) %>% select(-readmitted)
Xtr <- as.matrix(Xtr_tbl); Xte <- as.matrix(Xte_tbl)
feat_names <- colnames(Xtr)
cat(sprintf("Train: %d x %d | Test: %d x %d | Positive rate: %.1f%%\n",
            nrow(Xtr), ncol(Xtr), nrow(Xte), ncol(Xte), mean(y_tr) * 100))

3. Bias-Variance Tradeoff — Tree Depth Experiment

set.seed(42)
depths <- 1:20
skf <- vfold_cv(dt_data, v = 5, strata = readmitted)

cv_at_depth <- function(d) {
  map_dbl(skf$splits, function(sp) {
    tr <- analysis(sp); te <- assessment(sp)
    w  <- ifelse(tr$readmitted == "1", 0.5 / mean(tr$readmitted == "1"),
                 0.5 / mean(tr$readmitted == "0"))
    fit <- rpart(readmitted ~ ., data = tr, method = "class", weights = w,
                 control = rpart.control(maxdepth = d, minsplit = 20, cp = 0, xval = 0))
    p <- predict(fit, newdata = te, type = "prob")[, "1"]
    auc_prob(as.integer(as.character(te$readmitted)), p)
  }) %>% mean()
}

test_aucs <- map_dbl(depths, cv_at_depth)
best_depth <- depths[which.max(test_aucs)]

depth_df <- tibble(depth = depths, cv_auc = test_aucs)
ggplot(depth_df, aes(depth, cv_auc)) +
  geom_ribbon(aes(ymin = cv_auc - 0.01, ymax = cv_auc + 0.01), fill = col_navy, alpha = 0.10) +
  geom_line(colour = col_navy, linewidth = 1) +
  geom_point(colour = col_navy, size = 2) +
  geom_vline(xintercept = best_depth, colour = col_red, linetype = "dashed", linewidth = 0.8) +
  geom_vline(xintercept = 4, colour = "#D97706", linetype = "dotted", linewidth = 0.8) +
  annotate("text", x = best_depth + 0.4, y = 0.52,
           label = sprintf("Optimal depth=%d", best_depth), colour = col_red, hjust = 0, size = 3.2) +
  annotate("text", x = 4.3, y = 0.55, label = "Depth=4 (interpretable)",
           colour = "#D97706", hjust = 0, size = 3.2) +

5. Feature Importance — MDI vs Permutation

# MDI (Mean Decrease Impurity) — built-in, fast but biased toward high-cardinality features
mdi_vi <- vip::vi(rf) %>% slice_max(Importance, n = 15)
mdi_imp <- tibble(feature = mdi_vi$Variable, importance = mdi_vi$Importance)

# Permutation importance — model-agnostic, unbiased (evaluated on the test set)
set.seed(42)
base_auc <- auc_prob(y_te, predict(rf, data = Xte_tbl)$predictions[, "1"])
perm_imp <- map_dfr(feat_names, function(col) {
  drops <- replicate(10, {
    Xp <- Xte_tbl
    Xp[[col]] <- sample(Xp[[col]])
    base_auc - auc_prob(y_te, predict(rf, data = Xp)$predictions[, "1"])
  })
  tibble(feature = col, importance = mean(drops))
}) %>% arrange(desc(importance)) %>% slice_head(n = 15)

p_mdi <- ggplot(mdi_imp, aes(x = importance, y = reorder(feature, importance))) +
  geom_col(fill = col_blue, alpha = 0.85, colour = "white") +
  geom_text(aes(label = sprintf("%.4f", importance)), hjust = -0.05, size = 2.8) +
  labs(x = "Importance score", y = NULL,
       title = "MDI Feature Importance (built-in, biased toward high-cardinality)")

6. Hyperparameter Tuning with tune::tune_grid

set.seed(42)
rf_spec <- rand_forest(
  trees = tune(), min_n = tune(), mtry = tune()
) %>%
  set_engine("ranger", probability = TRUE, seed = 42,
             class.weights = cw) %>%
  set_mode("classification")

rf_wf <- workflow() %>%
  add_model(rf_spec) %>%
  add_formula(readmitted ~ .)

folds <- vfold_cv(rf_df, v = 3, strata = readmitted)
# n_estimators 100/200; min_samples_leaf 5/15; max_features sqrt/log2
p_feat <- ncol(Xtr)
grid_rf <- expand_grid(
  trees = c(100, 200),
  min_n = c(5, 15),
  mtry  = c(max(1, floor(log2(p_feat))), max(1, floor(sqrt(p_feat))))

gs <- tune_grid(
  rf_wf, resamples = folds, grid = grid_rf,

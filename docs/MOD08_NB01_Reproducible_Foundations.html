Module 08 - Reproducible Research & Deployment

Reproducible Research Foundations

Learning objectives - Understand reproducibility threats in health analytics - Build canonical project structure with dependency management - Implement data provenance with SHA-256 checksums and MANIFEST.json - Build reproducible pipelines with structured logging and wrappers - Write Makefiles for pipeline orchestration - Conduct a reproducibility audit

Estimated time: 2 hours | Level: Intermediate | Libraries: digest, fs, here, jsonlite, renv

1. Reproducibility Threats

Threat Example Prevention
Random seeds ML results change set.seed(42) everywhere
Package versions tidymodels API breaks pipelines Pin versions in packages.R / renv.lock
Data drift Source updated silently Checksums + MANIFEST.json
Hidden notebook state Chunks run out of order Restart & Render All + CI
OS differences Float rounding varies Docker
Undocumented transforms ‘I normalised this manually’ Code-only pipelines

3. Data Provenance & Checksums

compute_checksum <- function(filepath, algorithm = "sha256") {
  p <- fs::path(filepath)
  if (fs::is_file(p)) {
    digest::digest(file = as.character(p), algo = algorithm)
  } else {
    "file_not_found"
  }
}

create_manifest <- function(data_info, output_path) {
  manifest <- list(
    created_at  = format(Sys.time(), "%Y-%m-%dT%H:%M:%S"),
    r_version   = paste(R.version$major, R.version$minor, sep = "."),
    files       = list()
  )
  for (name in names(data_info)) {
    info <- data_info[[name]]
    manifest$files[[name]] <- list(
      path          = info$path %||% "",
      source        = info$source %||% "",
      download_date = info$download_date %||% format(Sys.Date()),
      sha256        = compute_checksum(info$path %||% ""),
      description   = info$description %||% "",
      n_rows        = info$n_rows,
      n_cols        = info$n_cols
    )
  }
  fs::dir_create(fs::path_dir(output_path))
  write_json(manifest, output_path, pretty = TRUE, auto_unbox = TRUE, null = "null")
  manifest
}

4. Reproducible Pipeline with Logging

log_file <- fs::path(PROJECT_ROOT, "pipeline.log")
logger_write <- function(level, message) {
  line <- sprintf("%s | %-7s | %s", format(Sys.time(), "%Y-%m-%d %H:%M:%S"), level, message)
  cat(line, "\n")
  cat(line, "\n", file = log_file, append = TRUE)
}

log_step <- function(name, fn) {
  force(name); force(fn)
  function(...) {
    logger_write("INFO", paste("START ", name))
    t0 <- proc.time()[["elapsed"]]
    tryCatch({
      result <- fn(...)
      elapsed <- proc.time()[["elapsed"]] - t0
      logger_write("INFO", sprintf("OK     %s | %.2fs", name, elapsed))
      result
    }, error = function(e) {
      logger_write("ERROR", sprintf("FAIL   %s | %s", name, e$message))
      stop(e)
    })
  }
}

load_raw <- log_step("load_raw_data", function(path) {
  df <- read_csv(path, show_col_types = FALSE)
  logger_write("INFO", sprintf("  Loaded %dr x %dc", nrow(df), ncol(df)))
  df
})

validate <- log_step("validate_schema", function(df) {
  checks <- list(
    patient_id_unique = n_distinct(df$patient_id) == nrow(df),
    age_18_90         = all(df$age >= 18 & df$age <= 90),
    readmitted_binary = all(df$readmitted %in% c(0, 1)),
    los_positive      = all(df$los_days > 0)
  )
  for (check in names(checks)) {
    if (!isTRUE(checks[[check]])) stop(glue("Validation FAILED: {check}"))
    logger_write("INFO", paste("  PASS:", check))
  }
  df
})

engineer_features <- log_step("feature_engineering", function(df) {
  df <- df %>%
    mutate(
      los_gt7  = as.integer(los_days > 7),
      age_std  = (age - mean(age)) / sd(age),
      n_comorb = diabetes + hf
    )
  logger_write("INFO", sprintf("  Added 3 features | Output: %d x %d", nrow(df), ncol(df)))
  df
})

6. Reproducibility Audit

reproducibility_audit <- function(project_dir) {
  root <- fs::path(project_dir)
  issues <- character()
  score <- 100

  req <- fs::path(root, "packages.R")
  lock <- fs::path(root, "renv.lock")
  if (!fs::file_exists(req) && !fs::file_exists(lock)) {
    issues <- c(issues, "MISSING packages.R / renv.lock")
    score <- score - 20
  } else if (fs::file_exists(req)) {
    lines <- readLines(req)
    pkg_lines <- lines[grepl("=", lines) & !grepl("^\\s*#", lines) & grepl("dplyr|ggplot2|xgboost", lines)]
    unpinned <- pkg_lines[!grepl(">=|==|<=", pkg_lines)]
    if (length(unpinned)) {
      issues <- c(issues, paste("Unpinned packages:", paste(unpinned, collapse = ", ")))
      score <- score - 10
    }
  }

  if (!fs::file_exists(fs::path(root, "data", "MANIFEST.json"))) {
    issues <- c(issues, "MISSING data/MANIFEST.json")
    score <- score - 20
  }

  gi <- fs::path(root, ".gitignore")
  if (fs::file_exists(gi)) {
    content <- paste(readLines(gi), collapse = "\n")
    for (p in c(".Renviron", "data/raw/")) {
      if (!grepl(p, content, fixed = TRUE)) {
        issues <- c(issues, paste(".gitignore missing:", p))
        score <- score - 5
      }
    }
  } else {
    issues <- c(issues, "MISSING .gitignore")
    score <- score - 15
  }

  if (!fs::file_exists(fs::path(root, "pipeline.log"))) {
    issues <- c(issues, "No pipeline.log found")
    score <- score - 5
  }

  list(score = max(0, score), issues = issues)
}

library(jsonlite)
library(digest)
library(fs)

verify_data_integrity <- function() {
  manifest_path <- fs::path('data', 'MANIFEST.json')
  if (!fs::file_exists(manifest_path)) {
    skip('No MANIFEST.json')
  }
  manifest <- jsonlite::fromJSON(manifest_path, simplifyVector = FALSE)
  for (name in names(manifest$files)) {
    info <- manifest$files[[name]]
    p <- fs::path(info$path)
    if (!fs::file_exists(p)) fail(paste('Missing:', p))
    h <- digest::digest(file = as.character(p), algo = 'sha256')
    if (!is.null(info$sha256) && h != info$sha256) {
      fail(paste('Checksum FAILED for', name, '-- data may have changed!'))
    }
  }
  cat(sprintf('[OK] %d files passed integrity check\n', length(manifest$files)))
}

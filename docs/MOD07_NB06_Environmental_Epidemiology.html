Health Analytics with R · Module 07: Geospatial & Spatial Epidemiology

Learning objectives - Interpolate point-source pollution data to area-level estimates (kriging, IDW) - Assess exposure-response relationships with ecological regression - Apply air pollution exposure assessment methods (inverse distance, kriging) - Conduct distributed lag non-linear models (DLNM) for time-series exposure - Build environmental justice metrics: environmental burden by demographic group - Address ecological fallacy in spatial health analyses

Estimated time: 3 hours | Level: Advanced
Libraries: sf, terra, gstat, spdep, ggplot2, broom

spatial_smooth <- function(values, row_idx, col_idx, nrow, ncol, sigma = 2) {
  grid <- matrix(0, nrow = nrow, ncol = ncol)
  for (i in seq_along(values)) grid[row_idx[i], col_idx[i]] <- values[i]
  k <- ceiling(3 * sigma)
  xs <- (-k):k
  kern <- outer(dnorm(xs, sd = sigma), dnorm(xs, sd = sigma))
  kern <- kern / sum(kern)
  pad <- k
  gp <- matrix(0, nrow + 2 * pad, ncol + 2 * pad)
  gp[(pad + 1):(pad + nrow), (pad + 1):(pad + ncol)] <- grid
  out <- matrix(0, nrow, ncol)
  for (r in seq_len(nrow)) for (c in seq_len(ncol)) {
    out[r, c] <- sum(gp[r:(r + 2 * k), c:(c + 2 * k)] * kern)
  }
  mapply(function(r, c) out[r, c], row_idx, col_idx)
}

NROW <- 10; NCOL <- 20; N <- NROW * NCOL
row_idx <- rep(seq_len(NROW), each = NCOL)
col_idx <- rep(seq_len(NCOL), times = NROW)
lon <- -120 + (col_idx - 1) * 2.5 + rnorm(N, 0, 0.3)
lat <- 28 + (row_idx - 1) * 2.0 + rnorm(N, 0, 0.2)
pct_poverty <- 10 + 8 * rbeta(N, 2, 5) + 3 * sin(lon / 20)
population <- as.integer(rlnorm(N, 10.5, 1.0))
spatial_risk <- spatial_smooth(rnorm(N), row_idx, col_idx, NROW, NCOL)
# Spatially patterned PM2.5 (higher in industrial areas — western region)
pm25_base <- 8 + 4 * (lon - min(lon)) / (max(lon) - min(lon))
pm25 <- pm25_base + spatial_smooth(rgamma(N, 1, 1), row_idx, col_idx, NROW, NCOL, 1.5) +
  rnorm(N, 0, 1)
pm25 <- pmin(pmax(pm25, 3), 30)
TRUE_PM25 <- 2.5
cvd_rate <- (160 + 1.2 * pct_poverty + TRUE_PM25 * pm25 + 10 * spatial_risk + rnorm(N, 0, 10))
expected <- (cvd_rate / 100000) * population * 5
observed <- rpois(N, expected)
smr <- observed / pmax(expected, 1)

df <- tibble(
  county_id = sprintf("C%03d", seq_len(N) - 1),
  lon, lat, row = row_idx, col = col_idx,

2. Exposure Assessment — Interpolating Point Data to Areas

Air quality monitors are located at specific points but we need county-level estimates.

Methods: | Method | Principle | Smoothness | Use case | |——–|———–|————|———-| | IDW | Weight = 1/distance^p | Moderate | Quick, no model | | Kriging (ordinary) | Geostatistical, uses variogram | Smooth | Optimal linear prediction | | Nearest monitor | Assign to closest monitor | Abrupt | Worst but common | | Land-use regression | Predict from land use covariates | Flexible | Urban studies |

# Simulate air quality monitoring stations (fewer than counties)
set.seed(13)
N_MONITORS <- 35
mon <- st_as_sf(
  tibble(
    lon = runif(N_MONITORS, min(df$lon), max(df$lon)),
    lat = runif(N_MONITORS, min(df$lat), max(df$lat)),
    pm25 = pmin(pmax(
      8 + 4 * (runif(N_MONITORS, min(df$lon), max(df$lon)) - min(df$lon)) /
        (max(df$lon) - min(df$lon)) + rgamma(N_MONITORS, 1, 1) + rnorm(N_MONITORS, 0, 1.5),
      3), 28)
  ),
  coords = c("lon", "lat"),
  crs = 4326
)
mon$pm25 <- pmin(pmax(
  8 + 4 * (st_coordinates(mon)[, 1] - min(df$lon)) / (max(df$lon) - min(df$lon)) +
    rgamma(N_MONITORS, 1, 1) + rnorm(N_MONITORS, 0, 1.5),
  3), 28)

  fit.variogram(vgm_emp, vgm("Exp")),
  error = function(e) vgm(psill = var(mon$pm25), model = "Exp", range = 5, nugget = 0.5)
)

ok_pred <- krige(pm25 ~ 1, mon, centroids, model = vgm_fit, nmax = 12, debug.level = 0)

# Raster overlay with terra: interpolate monitors to a raster, then extract county means
r <- rast(ext(st_bbox(gdf)), resolution = 0.5, crs = "EPSG:4326")
mon_vect <- vect(mon)
r_idw <- tryCatch(
  interpIDW(r, mon_vect, field = "pm25", radius = 8, power = 2),
  error = function(e) NULL
)
if (is.null(r_idw)) {
  gdf$pm25_raster <- gdf$pm25_idw
} else {
  ext_pm <- terra::extract(r_idw, vect(gdf), fun = mean, na.rm = TRUE)
  gdf$pm25_raster <- ext_pm[[2]]
  gdf$pm25_raster[is.na(gdf$pm25_raster)] <- gdf$pm25_idw[is.na(gdf$pm25_raster)]
}

plot_exp <- function(col, title) {
  r_xy <- cor(gdf$pm25, gdf[[col]], use = "complete.obs")
  ggplot(gdf) +
    geom_sf(aes(fill = .data[[col]]), colour = "white", linewidth = 0.15) +
    geom_sf(data = mon, shape = 17, size = 2.5, colour = "#2563EB") +
    scale_fill_distiller(name = "PM2.5", palette = "YlOrBr", direction = 1) +
    labs(title = sprintf("%s (r=%.3f with truth)", title, r_xy), x = "Lon", y = "Lat")
}

5. Ecological Fallacy — Individual vs Aggregate Data

N_IND <- 10000
ind_county <- sample.int(N, N_IND, replace = TRUE)

6. Knowledge Check

  1. IDW gives PM2.5 coefficient β=2.1 while RBF gives β=2.8. Which should you prefer and why?
  2. PM2.5 exposure and poverty are correlated (r=0.41). How does this affect the ecological regression for PM2.5’s health effect?
  3. The EJ analysis shows highest-poverty quintile has 1.4x higher PM2.5 than lowest. Is this statistically significant? How would you test it?
  4. Individual-level PM2.5 log-OR = 0.015 vs ecological log-RR = 0.07. Why do these differ? What is the ecological fallacy specifically?
  5. Implement Moran’s I for PM2.5 exposure to quantify spatial patterning of the exposure surface.

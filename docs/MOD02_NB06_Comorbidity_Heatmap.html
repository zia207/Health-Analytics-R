Learning objectives - Build a binary comorbidity matrix from ICD-10 flags - Compute pairwise co-occurrence, phi coefficients, and odds ratios - Cluster comorbidities with hierarchical linkage and display with a clustered heatmap - Identify clinically significant comorbidity pairs - Build a network graph of comorbidity associations

3. Phi Coefficient Matrix (Strength of Association)

phi_coeff <- function(a, b) {
  # Phi coefficient for two binary variables.  Range: -1 to +1.
  ct <- table(a, b)
  if (!all(dim(ct) == c(2, 2))) return(0)
  n    <- sum(ct)
  chi2 <- suppressWarnings(chisq.test(ct)$statistic)
  sgn  <- sign(ct[2, 2] * ct[1, 1] - ct[1, 2] * ct[2, 1])
  as.numeric(sqrt(unname(chi2) / n) * sgn)
}

phi_matrix  <- matrix(NA_real_, nrow = length(COMORBS), ncol = length(COMORBS),
                      dimnames = list(COMORBS, COMORBS))
pval_matrix <- matrix(NA_real_, nrow = length(COMORBS), ncol = length(COMORBS),
                      dimnames = list(COMORBS, COMORBS))
for (c1 in COMORBS) {
  for (c2 in COMORBS) {
    phi_matrix[c1, c2] <- phi_coeff(df[[c1]], df[[c2]])
    if (c1 != c2) {
      pval_matrix[c1, c2] <- suppressWarnings(
        chisq.test(table(df[[c1]], df[[c2]]))$p.value
      )
    } else {
      pval_matrix[c1, c2] <- 1
    }
  }
}

4. Clustered Heatmap

# Hierarchical clustering on phi matrix
hc  <- hclust(dist(phi_matrix, method = "euclidean"), method = "ward.D2")
ord <- hc$order
phi_ord <- phi_matrix[ord, ord]
phi_heat <- as.data.frame(as.table(phi_ord), stringsAsFactors = FALSE)
names(phi_heat) <- c("cond_1", "cond_2", "phi")
phi_heat$cond_1 <- factor(phi_heat$cond_1, levels = rownames(phi_ord))
phi_heat$cond_2 <- factor(phi_heat$cond_2, levels = colnames(phi_ord))

p_cl <- ggplot(phi_heat, aes(x = cond_1, y = cond_2, fill = phi)) +

5. Odds Ratios for Key Pairs

odds_ratio <- function(a, b) {
  # Odds ratio and 95% CI (Woolf method) for two binary variables.
  ct <- as.matrix(table(a, b))
  if (!all(dim(ct) == c(2, 2))) return(c(NA, NA, NA))
  # Apply 0.5 continuity correction if any cell is 0
  ct <- ct + 0.5
  OR <- (ct[2, 2] * ct[1, 1]) / (ct[1, 2] * ct[2, 1])
  log_or <- log(OR)
  se_log <- sqrt(1 / ct[1, 1] + 1 / ct[1, 2] + 1 / ct[2, 1] + 1 / ct[2, 2])
  lo <- exp(log_or - 1.96 * se_log)
  hi <- exp(log_or + 1.96 * se_log)
  round(c(OR, lo, hi), 2)
}

top_pairs <- phi_long %>%
  filter(significant) %>%
  arrange(desc(phi)) %>%
  slice_head(n = 8)

or_df <- top_pairs %>%
  rowwise() %>%
  mutate(
    or_ci = list(odds_ratio(df[[as.character(cond_1)]], df[[as.character(cond_2)]])),
    Pair  = paste(cond_1, "×", cond_2),

6. Comorbidity Network Graph

threshold <- 0.15  # minimum phi to draw an edge
sig_pairs <- phi_long %>%
  filter(abs(phi) > threshold, significant)

vertices <- tibble(
  name = COMORBS,
  prev = map_dbl(COMORBS, ~ mean(df[[.x]]))
)

g <- graph_from_data_frame(sig_pairs %>% transmute(from = cond_1, to = cond_2, phi),
                           directed = FALSE, vertices = vertices)
E(g)$weight <- abs(E(g)$phi)
set.seed(42)
lay <- layout_with_fr(g)

e_col <- ifelse(E(g)$phi > 0, "#DC2626", "#2563EB")
v_size <- 8 + vertices$prev[match(V(g)$name, vertices$name)] * 40

op <- par(mar = c(0, 0, 2, 0), bg = "white")
plot(
  g,
  layout              = lay,
  vertex.size         = v_size,
  vertex.color        = "#F0E6FF",
  vertex.frame.color  = "#6A4C93",
  vertex.label.cex    = 0.9,
  vertex.label.font   = 2,
  vertex.label.color  = "grey15",
  edge.width          = E(g)$weight * 8,

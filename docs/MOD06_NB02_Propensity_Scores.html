Module 06 — Causal Inference in Health Research

Propensity Score Methods

Learning objectives - Estimate propensity scores using logistic regression and GBM - Apply PS matching (nearest-neighbour, caliper, 1:k) - Implement inverse probability of treatment weighting (IPTW) - Assess covariate balance with Standardised Mean Differences (SMD) - Estimate ATE and ATT with confidence intervals via bootstrap - Compare PS matching vs IPTW trade-offs

Estimated time: 3 hours | Level: Intermediate-Advanced | Libraries: MatchIt, WeightIt, cobalt, gbm, ggplot2, dplyr

calib_curve <- function(y, p, n_bins = 10) {
  breaks <- seq(0, 1, length.out = n_bins + 1)
  bin <- cut(p, breaks, include.lowest = TRUE)
  tibble(y = y, p = p, bin = bin) |>
    group_by(bin) |>
    summarise(
      mean_predicted    = mean(p),
      fraction_treated  = mean(y),
      .groups = "drop"
    )
}

3. Covariate Balance — Standardised Mean Differences

compute_smd <- function(df_sub, treatment_col, covariates) {
  # Compute Standardised Mean Difference for each covariate.
  treat <- df_sub[df_sub[[treatment_col]] == 1, ]
  ctrl  <- df_sub[df_sub[[treatment_col]] == 0, ]
  smds <- sapply(covariates, function(cov) {
    m1 <- mean(treat[[cov]]); m0 <- mean(ctrl[[cov]])
    s1 <- sd(treat[[cov]]);   s0 <- sd(ctrl[[cov]])
    pooled_sd <- sqrt((s1^2 + s0^2) / 2)
    if (pooled_sd > 0) (m1 - m0) / pooled_sd else 0
  })
  tibble(covariate = covariates, smd = as.numeric(smds))
}

smd_unadjusted <- compute_smd(df, "statin", COVARIATES)

5. ATT Estimation via Matching

estimate_att_matching <- function(df_m, treatment, outcome) {
  # Estimate ATT from matched dataset.
  r1 <- mean(df_m[[outcome]][df_m[[treatment]] == 1])
  r0 <- mean(df_m[[outcome]][df_m[[treatment]] == 0])
  log_or <- log(r1 / (1 - r1)) - log(r0 / (1 - r0))
  list(
    RD     = round(r1 - r0, 4),
    RR     = round(r1 / r0, 4),
    log_OR = round(log_or, 4),
    OR     = round(exp(log_or), 4)
  )
}

# Bootstrap CI for ATT
bootstrap_att <- function(df_m, n_boot = 500, seed = 42) {
  set.seed(seed)
  ests <- replicate(n_boot, {
    idx <- sample.int(nrow(df_m), replace = TRUE)
    boot <- df_m[idx, ]
    estimate_att_matching(boot, "statin", "mi")$log_OR
  })
  as.numeric(quantile(ests, c(0.025, 0.975)))
}

# Weighted SMD (balance check for IPTW)
weighted_smd <- function(dat, treatment, covariates, weights) {
  smds <- sapply(covariates, function(cov) {
    idx1 <- dat[[treatment]] == 1
    idx0 <- dat[[treatment]] == 0
    w1 <- weights[idx1]; x1 <- dat[[cov]][idx1]
    w0 <- weights[idx0]; x0 <- dat[[cov]][idx0]
    m1 <- weighted.mean(x1, w1); m0 <- weighted.mean(x0, w0)
    v1 <- weighted.mean((x1 - m1)^2, w1)
    v0 <- weighted.mean((x0 - m0)^2, w0)
    pooled <- sqrt((v1 + v0) / 2)
    if (pooled > 0) (m1 - m0) / pooled else 0
  })
  tibble(covariate = covariates, smd = as.numeric(smds))
}

# Estimate ATE using IPTW (Horvitz-Thompson estimator)
iptw_estimate <- function(dat, treatment, outcome, weight_col) {
  idx1 <- dat[[treatment]] == 1
  idx0 <- dat[[treatment]] == 0
  r1 <- sum(dat[[outcome]][idx1] * dat[[weight_col]][idx1]) /
    sum(dat[[weight_col]][idx1])
  r0 <- sum(dat[[outcome]][idx0] * dat[[weight_col]][idx0]) /
    sum(dat[[weight_col]][idx0])
  log_or <- log(r1 / (1 - r1)) - log(r0 / (1 - r0))
  list(
    RD     = round(r1 - r0, 4),
    RR     = round(r1 / r0, 4),
    log_OR = round(log_or, 4),
    OR     = round(exp(log_or), 4)
  )
}

# Exercise 5 — Doubly Robust (AIPW) estimator
# Outcome model (Q-model)
fit_out <- glm(
  mi ~ age + smoking + chd_history + ldl_baseline + ses_score + statin,
  data = df, family = binomial()
)

# Predict potential outcomes
df_1 <- df; df_1$statin <- 1
df_0 <- df; df_0$statin <- 0
mu1 <- predict(fit_out, newdata = df_1, type = "response")
mu0 <- predict(fit_out, newdata = df_0, type = "response")

# AIPW correction terms
ps <- df$ps_gb
A  <- df$statin
Y  <- df$mi
# ATE AIPW
aipw_1 <- mu1 + (A / ps) * (Y - mu1)
aipw_0 <- mu0 + ((1 - A) / (1 - ps)) * (Y - mu0)
ate_aipw_rd <- mean(aipw_1) - mean(aipw_0)
mu1_mean <- mean(mu1)
mu0_mean <- mean(mu0)
aipw_lor <- log(mu1_mean / (1 - mu1_mean)) - log(mu0_mean / (1 - mu0_mean))

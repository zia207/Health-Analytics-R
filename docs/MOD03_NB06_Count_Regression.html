Health Analytics with R · Module 03: Statistical Inference

Learning objectives - Identify when count regression is needed (ED visits, hospital admissions, procedures) - Fit Poisson regression and interpret incidence rate ratios (IRRs) - Test for overdispersion and switch to negative binomial when needed - Handle offset terms for rate modelling (visits per person-year) - Fit zero-inflated models for excess zeros

Estimated time: 2 hours | Level: Advanced | Libraries: stats, MASS, broom, ggplot2, pscl

logistic <- function(x) 1 / (1 + exp(-x))

set.seed(42)
N <- 1500
base <- matrix(rnorm(N * 3), ncol = 3)

set.seed(99)
ages          <- as.integer(pmin(pmax(rnorm(N, 58, 15), 18), 90))
sexes         <- sample(c("M", "F"), N, replace = TRUE, prob = c(0.48, 0.52))
diabetes      <- rbinom(N, 1, logistic(0.6 * base[, 1] - 0.5))
heart_failure <- rbinom(N, 1, logistic(0.4 * base[, 2] - 1.2))
copd          <- rbinom(N, 1, logistic(0.5 * base[, 3] - 1.0))
mental_health <- rbinom(N, 1, logistic(0.3 * base[, 3] - 1.4))
payers        <- sample(
  c("Medicare", "Medicaid", "Private", "Self-pay"),
  N, replace = TRUE, prob = c(0.38, 0.24, 0.30, 0.08)
)
follow_up_yrs <- round(runif(N, 0.5, 3.0), 2)  # Person-years of follow-up

# ED visits count: Poisson with mean depending on predictors
log_mu <- (-0.3 + 0.5 * diabetes + 0.8 * heart_failure + 0.6 * copd
           + 0.02 * (ages - 58) + 0.4 * mental_health
           + 0.3 * as.integer(payers == "Medicaid")
           + log(follow_up_yrs))             # offset for follow-up time
true_mu <- exp(log_mu)

# Overdispersed count (NB with dispersion)
ed_visits_pois <- rpois(N, true_mu)                          # Poisson
# nbinom: number of failures before `size` successes; p = size / (size + mu)
ed_visits_nb   <- rnbinom(N, size = 2, mu = true_mu)         # NegBin (overdispersed)

# Zero-inflated count (many true zeros)
zero_mask     <- rbinom(N, 1, 0.30)                     # 30% structural zeros
ed_visits_zip <- ifelse(zero_mask == 1, 0, ed_visits_pois)  # Zero-inflated

df <- tibble(
  patient_id     = sprintf("PT-%05d", seq_len(N)),
  age            = ages,
  sex            = sexes,
  payer          = relevel(factor(payers), ref = "Private"),
  diabetes       = diabetes,
  heart_failure  = heart_failure,
  copd           = copd,
  mental_health  = mental_health,
  follow_up_yrs  = follow_up_yrs,
  log_follow_up  = log(follow_up_yrs),
  ed_visits      = ed_visits_nb,      # primary outcome (overdispersed)
  ed_visits_pois = ed_visits_pois,    # comparison
  ed_visits_zip  = ed_visits_zip      # zero-inflated comparison
) |>
  mutate(age_std = as.numeric(scale(age)))

4. Testing for Overdispersion

overdispersion_test <- function(model_pois, df, outcome_col) {
  # Cameron-Trivedi auxiliary regression test for overdispersion.
  # H0: Var(Y) = μ  (equidispersion — Poisson OK)
  # H1: Var(Y) > μ  (overdispersion — use NB)
  mu_hat <- predict(model_pois, type = "response")
  y <- df[[outcome_col]]
  aux_y <- ((y - mu_hat)^2 - y) / mu_hat
  aux_model <- lm(aux_y ~ 0 + mu_hat)
  sm <- summary(aux_model)
  list(
    alpha = unname(coef(aux_model)[1]),
    t_stat = unname(sm$coefficients[1, "t value"]),
    p_val  = unname(sm$coefficients[1, "Pr(>|t|)"])
  )
}

od <- overdispersion_test(model_pois, df, "ed_visits")
cat("Overdispersion test (Cameron-Trivedi):\n")

# Fit Zero-Inflated Poisson via pscl if available
zip_ok <- requireNamespace("pscl", quietly = TRUE)
if (zip_ok) {
  zip_model <- pscl::zeroinfl(
    ed_visits_zip ~ age_std + diabetes + heart_failure + copd + mental_health +
      offset(log_follow_up) |
      age_std + diabetes + heart_failure + copd + mental_health,
    data = df, dist = "poisson"
  )
  cat("\nZero-Inflated Poisson — selected coefficients:\n")
  zip_count <- summary(zip_model)$coefficients$count
  zip_tid <- as.data.frame(zip_count) |>
    rownames_to_column("term") |>
    mutate(
      IRR = exp(Estimate),
      conf.low = exp(Estimate - 1.96 * `Std. Error`),
      conf.high = exp(Estimate + 1.96 * `Std. Error`)
    ) |>
    mutate(across(where(is.numeric), ~ round(.x, 3)))
  print(zip_tid)
  cat(sprintf("\nAIC: %.1f\n", AIC(zip_model)))
} else {
  # Fallback: two-part hurdle-style sketch with glm (always runs)
  cat("\npscl not installed — hurdle-style two-part model instead.\n")
  df$any_ed <- as.integer(df$ed_visits_zip > 0)
  m_zero <- glm(
    any_ed ~ age_std + diabetes + heart_failure + copd + mental_health,
    data = df, family = binomial()
  )
  m_count <- glm(
    ed_visits_zip ~ age_std + diabetes + heart_failure + copd + mental_health,
    data = df |> filter(ed_visits_zip > 0), family = poisson(),
    offset = log_follow_up[df$ed_visits_zip > 0]
  )
  cat("Zero-process (logistic) ORs:\n")
  print(broom::tidy(m_zero, conf.int = TRUE, exponentiate = TRUE) |>
          filter(term != "(Intercept)"))
  cat("\nPositive-count Poisson IRRs:\n")
  print(broom::tidy(m_count, conf.int = TRUE, exponentiate = TRUE) |>
          filter(term != "(Intercept)"))
}

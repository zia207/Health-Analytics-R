Module 08 - Reproducible Research & Deployment

Plumber for Clinical Model Serving

Learning objectives - Build a production plumber REST API for clinical model serving - Implement request/response validation for patient features - Create single and batch prediction endpoints - Write an API test suite with edge cases and error handling - Add filters: CORS, logging, rate limiting, authentication - Containerise with Docker and docker-compose for deployment

Estimated time: 3 hours | Level: Advanced | Libraries: plumber, jsonlite, xgboost

set.seed(42)
N <- 2000
sigmoid <- function(x) 1 / (1 + exp(-x))
age  <- as.integer(pmin(pmax(rnorm(N, 60, 15), 18), 90))
los  <- as.integer(pmin(pmax(rgamma(N, 2.5, scale = 1.8), 1), 30))
dm   <- rbinom(N, 1, 0.35)
hf   <- rbinom(N, 1, 0.22)
copd <- rbinom(N, 1, 0.18)
logit <- -3.2 + 0.6 * hf + 0.5 * dm + 0.3 * copd + 0.02 * (age - 60) +
  0.2 * as.integer(los > 7)
readmitted <- rbinom(N, 1, sigmoid(logit))
df <- tibble(age = age, los_days = los, diabetes = dm, hf = hf, copd = copd,
             readmitted = readmitted)
FEATURES <- c("age", "los_days", "diabetes", "hf", "copd")
split <- initial_split(df, prop = 0.8, strata = readmitted)
train <- training(split)
test  <- testing(split)
model <- xgb.train(
  params = list(objective = "binary:logistic", max_depth = 4, eta = 0.05,
                eval_metric = "auc"),
  data = xgb.DMatrix(as.matrix(train[FEATURES]), label = train$readmitted),
  nrounds = 200, verbose = 0
)
auc <- as.numeric(pROC::auc(test$readmitted, predict(model, as.matrix(test[FEATURES])), quiet = TRUE))
model_path <- file.path(mod08_dir, "readmission_model.rds")
saveRDS(model, model_path)
cat(sprintf("Model trained | AUC=%.4f | Saved: %s\n", auc, model_path))

2. plumber Architecture for Clinical APIs

POST /predict           <- Single patient prediction
POST /predict/batch     <- Batch predictions (list of patients)
GET  /health            <- Health check (for Docker/k8s)
GET  /model/info        <- Model metadata
GET  /__docs__          <- plumber Swagger UI (auto-generated)
GET  /metrics           <- Prometheus metrics (optional)

Request flow:
  HTTP Request
    -> plumber parser (raise 400 if invalid JSON)
    -> Filters (auth, logging, rate limit)
    -> Endpoint function
    -> predict(model, type = "response")
    -> JSON serialisation
    -> HTTP Response (JSON)

validate_patient <- function(age, los_days, diabetes, hf, copd) {
  errors <- character()
  if (is.null(age) || age < 18 || age > 90) errors <- c(errors, paste('age must be 18-90, got', age))
  if (is.null(los_days) || los_days < 1 || los_days > 60) errors <- c(errors, paste('los_days must be 1-60, got', los_days))
  if (!diabetes %in% c(0, 1)) errors <- c(errors, 'diabetes must be 0 or 1')
  if (!hf %in% c(0, 1)) errors <- c(errors, 'hf must be 0 or 1')
  if (!copd %in% c(0, 1)) errors <- c(errors, 'copd must be 0 or 1')
  errors
}

4. Request Validation & Predictions

PatientFeatures <- function(age, los_days, diabetes, hf, copd, patient_id = NULL) {
  errors <- character()
  if (!(18 <= age && age <= 90)) errors <- c(errors, sprintf("age must be 18-90, got %s", age))
  if (!(1 <= los_days && los_days <= 60)) errors <- c(errors, sprintf("los_days must be 1-60, got %s", los_days))
  if (!diabetes %in% c(0, 1)) errors <- c(errors, "diabetes must be 0 or 1")
  if (!hf %in% c(0, 1)) errors <- c(errors, "hf must be 0 or 1")
  if (!copd %in% c(0, 1)) errors <- c(errors, "copd must be 0 or 1")
  if (length(errors)) stop(paste(errors, collapse = "; "))
  list(age = age, los_days = los_days, diabetes = diabetes, hf = hf, copd = copd,
       patient_id = patient_id)
}

predict_patient <- function(patient, model_obj) {
  X <- matrix(
    c(patient$age, patient$los_days, patient$diabetes, patient$hf, patient$copd),
    nrow = 1, dimnames = list(NULL, FEATURES)
  )
  prob <- as.numeric(predict(model_obj, X))
  tier <- if (prob < 0.10) "Low" else if (prob < 0.20) "Moderate" else if (prob < 0.35) "High" else "Very High"
  list(
    patient_id = patient$patient_id,
    readmission_probability = round(prob, 4),
    risk_tier = tier,
    model_version = "1.0.0"
  )
}

5. API Test Suite

simulate_api_call <- function(endpoint, payload, model_obj) {
  t0 <- proc.time()[["elapsed"]]
  on.exit(cat(sprintf("  [%s] %.1fms\n", endpoint, (proc.time()[["elapsed"]] - t0) * 1000)))
  tryCatch({
    if (endpoint == "/health") {
      return(list(status = "healthy", timestamp = "2024-01-15T10:30:00"))
    } else if (endpoint == "/predict") {
      needed <- c("age", "los_days", "diabetes", "hf", "copd")
      if (!all(needed %in% names(payload))) stop("Missing required fields")
      pt <- do.call(PatientFeatures, payload[intersect(names(payload), c(needed, "patient_id"))])
      return(predict_patient(pt, model_obj))
    } else if (endpoint == "/predict/batch") {
      results <- lapply(payload$patients, function(p) {
        pt <- do.call(PatientFeatures, p[intersect(names(p), c("age", "los_days", "diabetes", "hf", "copd", "patient_id"))])
        predict_patient(pt, model_obj)
      })
      return(list(predictions = results, count = length(results)))
    }
  }, error = function(e) list(error = e$message))
}

# Exercise 5 — /explain endpoint with DALEX / feature importance
explain_prediction <- function(patient_dict, model_obj, features = FEATURES) {
  X <- matrix(
    unlist(patient_dict[features]), nrow = 1,
    dimnames = list(NULL, features)
  )
  prob <- as.numeric(predict(model_obj, X))
  imp <- tryCatch({
    xgb.importance(model = model_obj)
  }, error = function(e) {
    tibble(Feature = features, Gain = rep(1 / length(features), length(features)))
  })
  top3 <- head(imp, 3)
  list(
    patient_id = patient_dict$patient_id,
    readmission_probability = round(prob, 4),
    risk_tier = if (prob > 0.35) "High" else if (prob > 0.20) "Moderate" else "Low",
    top_3_drivers = lapply(seq_len(nrow(top3)), function(i) {
      list(
        feature = as.character(top3$Feature[i]),
        importance = round(as.numeric(top3$Gain[i]), 4),
        direction = "increases risk"
      )
    })
  )
}

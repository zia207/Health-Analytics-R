N <- 3000
sigmoid <- function(x) 1 / (1 + exp(-x))
age  <- as.integer(pmin(pmax(rnorm(N, 60, 15), 18), 90))
los  <- as.integer(pmin(pmax(rgamma(N, 2.5, scale = 1.8), 1), 30))

hf   <- rbinom(N, 1, 0.22)
copd <- rbinom(N, 1, 0.18)
logit <- -3.2 + 0.6 * hf + 0.5 * dm + 0.3 * copd + 0.02 * (age - 60) +

tracker <- list(
  start_run = function(run_name) {
    tracker_env$current_run <- list(
      run_name = run_name, params = list(), metrics = list(), tags = list(),
      start_time = proc.time()[["elapsed"]]
    )
  },
  log_param = function(key, value) tracker_env$current_run$params[[key]] <<- value,
  log_params = function(params) {
    for (k in names(params)) tracker_env$current_run$params[[k]] <<- params[[k]]
  },
  log_metric = function(key, value) {
    tracker_env$current_run$metrics[[key]] <<- round(as.numeric(value), 6)
  },
  log_metrics = function(metrics) {
    for (k in names(metrics)) tracker$log_metric(k, metrics[[k]])
  },
  set_tag = function(key, value) {
    tracker_env$current_run$tags[[key]] <<- as.character(value)
  },
  end_run = function() {
    tracker_env$current_run$duration_s <- round(
      proc.time()[["elapsed"]] - tracker_env$current_run$start_time, 2
    )
    tracker_env$runs[[length(tracker_env$runs) + 1]] <- tracker_env$current_run
    write_json(
      tracker_env$current_run,
      file.path(tracker_env$tracking_dir, paste0(tracker_env$current_run$run_name, ".json")),
      pretty = TRUE, auto_unbox = TRUE
    )
  },
  get_runs_df = function() {
    map_dfr(tracker_env$runs, function(r) {
      params <- r$params; metrics <- r$metrics
      names(params) <- paste0("param.", names(params))
      names(metrics) <- paste0("metric.", names(metrics))
      bind_cols(
        tibble(run_name = r$run_name, duration_s = r$duration_s %||% 0),
        as_tibble(params), as_tibble(metrics)
      )
    })
  }
)

4. Model Comparison Experiment

avg_precision <- function(y, p) {
  ord <- order(p, decreasing = TRUE)
  y <- y[ord]
  tp <- cumsum(y)
  prec <- tp / seq_along(y)
  rec  <- tp / sum(y)
  # step-wise PR-AUC
  sum(diff(c(0, rec)) * prec)
}
brier_score <- function(y, p) mean((p - y)^2)

predict_prob <- function(fit, X, type) {
  if (type == "glm") {
    as.numeric(predict(fit, newdata = X, type = "response"))
  } else if (type == "ranger") {
    pred <- predict(fit, data = X)$predictions
    if (is.matrix(pred)) pred[, colnames(pred) == "1" | colnames(pred) == "TRUE"] else pred
  } else if (type == "xgb") {
    predict(fit, as.matrix(X))
  }
}

train_glm <- function(C) {
  w <- ifelse(y_tr == 1, sum(y_tr == 0) / max(sum(y_tr == 1), 1), 1)
  glm(readmitted ~ age + los_days + diabetes + hf + copd,
      data = bind_cols(X_tr, readmitted = y_tr),
      family = binomial(), weights = w)
}

8. Knowledge Check

  1. You log 50 metrics per run. How do you decide which metrics are PRIMARY for model selection?
  2. MLflow Model Registry has three stages: Staging, Production, Archived. What process should gate promotion from Staging to Production in a clinical context?
  3. A new model has AUC=0.79 vs production model AUC=0.77. Should you always promote? What other factors matter?
  4. Your MLflow database grows to 10GB after 6 months. What’s your data management strategy?
  5. Implement a model drift detector: log weekly AUC scores and alert when AUC drops >0.03 below the baseline.
# Exercise 5 — Model drift detection
set.seed(42)
weeks <- 1:12
baseline_auc <- 0.780
weekly_aucs <- sapply(weeks, function(w) {
  if (w <= 8) baseline_auc + rnorm(1, 0, 0.01)
  else baseline_auc - 0.005 * (w - 8) + rnorm(1, 0, 0.01)
})

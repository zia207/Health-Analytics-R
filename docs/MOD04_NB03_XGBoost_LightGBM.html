Health Analytics with R · Module 04: Machine Learning for Clinical Prediction

Learning objectives - Understand gradient boosting conceptually (additive trees, residual fitting) - Train XGBoost and LightGBM with clinical-appropriate hyperparameters - Use tune::tune_bayes for efficient Bayesian hyperparameter optimisation - Handle categorical features natively in LightGBM - Interpret early stopping and learning curves to prevent overfitting

Estimated time: 2.5 hours | Level: Intermediate → Advanced | Libraries: xgboost, lightgbm, tune, dials, tidymodels

# ── Dataset ───────────────────────────────────────────────────────────────────
set.seed(42); N <- 3000
sig <- function(x) 1 / (1 + exp(-x))
clip <- function(x, lo, hi) pmin(pmax(x, lo), hi)
b <- matrix(rnorm(N * 4), ncol = 4); set.seed(99)
age <- as.integer(clip(rnorm(N, 62, 15), 18, 92))
sex <- sample(c("M", "F"), N, replace = TRUE, prob = c(0.48, 0.52))
payer <- sample(c("Medicare", "Medicaid", "Private", "Self-pay", "Other"),
                N, replace = TRUE, prob = c(0.40, 0.22, 0.28, 0.07, 0.03))
admit_type <- sample(c("Emergency", "Elective", "Urgent"),
                     N, replace = TRUE, prob = c(0.52, 0.30, 0.18))
los_days <- as.integer(clip(round(rgamma(N, shape = 2.5, scale = 1.8)), 1, 30))
diabetes <- rbinom(N, 1, sig(0.6 * b[, 1] - 0.5 + 0.02 * (age - 62) / 15))
hypertension <- rbinom(N, 1, sig(0.7 * b[, 1] - 0.2))
hf <- rbinom(N, 1, sig(0.4 * b[, 2] + 0.5 * hypertension - 1.5))
copd <- rbinom(N, 1, sig(0.5 * b[, 3] - 1.0))
ckd <- rbinom(N, 1, sig(0.5 * b[, 1] + 0.6 * diabetes - 1.2))
obesity <- rbinom(N, 1, sig(0.5 * b[, 1] - 0.8))
depression <- rbinom(N, 1, sig(0.3 * b[, 4] - 1.4))
n_comorb <- diabetes + hypertension + hf + copd + ckd
set.seed(21)
glucose <- round(clip(rnorm(N, 105 + 15 * diabetes, 28), 50, 400), 1)
creatinine <- round(clip(rgamma(N, shape = 1.6 + 0.8 * hf, scale = 0.6), 0.4, 10), 2)
hba1c <- round(clip(rnorm(N, 6.5 + 0.8 * diabetes, 1.4), 4, 14), 1)
sbp <- round(clip(rnorm(N, 128 + 12 * hypertension, 18), 80, 200), 0)
bmi <- round(clip(rnorm(N, 28 + 4 * obesity, 6), 15, 55), 1)
logit <- (-3.2 + 0.6 * hf + 0.5 * diabetes + 0.5 * ckd + 0.3 * copd +
            0.02 * (age - 62) / 15 +
            0.3 * as.integer(admit_type == "Emergency") +
            0.25 * as.integer(payer == "Medicaid") +
            0.2 * as.integer(los_days > 7) +
            0.5 * hf * diabetes + rnorm(N, 0, 0.25))
readmitted <- rbinom(N, 1, sig(logit))

)

2. Gradient Boosting — Conceptual Overview

Iteration 0:  predict mean (F₀)
Iteration 1:  fit a tree to residuals r₁ = y - F₀  →  F₁ = F₀ + η·h₁
Iteration 2:  fit a tree to residuals r₂ = y - F₁  →  F₂ = F₁ + η·h₂
...
Iteration T:  F_T = F₀ + η·Σ hₜ
Concept Effect
Learning rate η Shrinks each tree’s contribution — smaller = more trees needed but better generalization
n_estimators Number of trees — use early stopping to find optimal T
max_depth Tree depth — shallower = more regularization
subsample / colsample Stochastic gradient boosting — reduces overfitting
scale_pos_weight XGBoost class-imbalance correction = N_neg / N_pos 
# ── XGBoost with early stopping ───────────────────────────────────────────────
dtrain <- xgb.DMatrix(Xtr,  label = y_tr2)
dval   <- xgb.DMatrix(Xval, label = y_val)
dte    <- xgb.DMatrix(Xte,  label = y_te)

xgb_params <- list(
  objective        = "binary:logistic",
  eval_metric      = "auc",
  eta              = 0.05,          # learning rate
  max_depth        = 6,
  subsample        = 0.8,
  colsample_bytree = 0.8,
  scale_pos_weight = scale_pos,     # handle class imbalance
  seed             = 42,
  verbosity        = 0
)
xgb_model <- xgb.train(
  params = xgb_params, data = dtrain,
  nrounds = 500,
  evals = list(train = dtrain, val = dval),
  early_stopping_rounds = 30,
  verbose = 0
)
prob_xgb <- predict(xgb_model, dte)
auc_xgb  <- auc_prob(y_te, prob_xgb)
best_iter_xgb <- attr(xgb_model, "early_stop")$best_iteration
if (is.null(best_iter_xgb)) best_iter_xgb <- xgb.get.num.boosted.rounds(xgb_model)
cat(sprintf("XGBoost: best_iteration=%d, AUC=%.4f\n", best_iter_xgb, auc_xgb))

# LightGBM wants integer codes for categoricals
to_lgb_mat <- function(d) {
  d2 <- d
  for (col in CATEGORIC) d2[[col]] <- as.integer(d2[[col]]) - 1L
  as.matrix(d2)
}
cat_idx <- match(CATEGORIC, names(Xl_tr2_x)) - 1L  # 0-based for lightgbm
dtrain_l <- lgb.Dataset(to_lgb_mat(Xl_tr2_x), label = yl_tr2, categorical_feature = cat_idx)
dval_l   <- lgb.Dataset(to_lgb_mat(Xl_val_x),  label = yl_val,  categorical_feature = cat_idx,
                        reference = dtrain_l)

lgb_params <- list(
  objective        = "binary",
  metric           = "auc",
  learning_rate    = 0.05,
  num_leaves       = 31,
  max_depth        = -1,
  subsample        = 0.8,
  colsample_bytree = 0.8,
  is_unbalance     = TRUE,
  verbosity        = -1,
  seed             = 42
)
lgb_model <- lgb.train(
  params = lgb_params, data = dtrain_l,
  nrounds = 500,
  valids = list(train = dtrain_l, val = dval_l),
  early_stopping_rounds = 30,
  verbose = -1
)

3. Bayesian Hyperparameter Optimisation (tune::tune_bayes)

set.seed(42)
# parsnip boost_tree (xgboost engine) + Bayesian optimisation (tune_bayes)
# analogue of Optuna TPE search
xgb_spec <- boost_tree(
  trees = tune(), tree_depth = tune(), learn_rate = tune(),
  sample_size = tune(), min_n = tune(), mtry = tune()
) %>%
  set_engine("xgboost", scale_pos_weight = scale_pos) %>%
  set_mode("classification")

bayes_data <- Xtr_tbl %>% mutate(readmitted = factor(y_tr2, levels = c(0, 1)))
xgb_wf <- workflow() %>%
  add_model(xgb_spec) %>%
  add_formula(readmitted ~ .)

folds <- vfold_cv(bayes_data, v = 3, strata = readmitted)
p_feat <- ncol(Xtr)

xgb_param <- extract_parameter_set_dials(xgb_wf) %>%
  update(
    trees      = trees(range = c(100L, 500L)),
    tree_depth = tree_depth(range = c(3L, 8L)),
    learn_rate = learn_rate(range = c(-2, log10(0.3))),
    sample_size = sample_prop(range = c(0.6, 1.0)),
    min_n      = min_n(range = c(5L, 30L)),
    mtry       = mtry(range = c(max(1L, floor(sqrt(p_feat))), max(2L, floor(p_feat / 2))))
  )

bayes_res <- tune_bayes(
  xgb_wf, resamples = folds, param_info = xgb_param,
  initial = 8, iter = 22,
  metrics = metric_set(roc_auc),
  control = control_bayes(no_improve = 8, verbose = FALSE)
)

5. Knowledge Check

  1. XGBoost stopped at iteration 87 out of 500. What does this tell you about the learning rate and regularization?
  2. LightGBM grows leaf-wise while XGBoost grows level-wise. What are the practical implications for clinical datasets with few features?
  3. scale_pos_weight was set to 6.8 (ratio of negatives to positives). What would happen if you forgot this parameter?
  4. Your tune_bayes study ran 30 trials. The best AUC is 0.78. Would 100 more trials likely improve this substantially?
  5. Re-run LightGBM with learning_rate=0.01 and nrounds=1000. Does it converge to the same AUC?

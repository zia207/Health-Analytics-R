Module 08 - Reproducible Research & Deployment

Testing, CI/CD & Code Quality

Learning objectives - Write unit tests for data pipeline and model functions with testthat - Build model performance tests (AUC thresholds, direction tests) - Implement property-based testing with random valid inputs - Configure GitHub Actions CI/CD for model training and deployment - Apply pre-commit hooks for code quality (styler, lintr) - Measure test coverage and enforce minimums in CI

Estimated time: 2.5 hours | Level: Intermediate-Advanced | Libraries: testthat, lintr, styler

set.seed(42)
N <- 2000
sigmoid <- function(x) 1 / (1 + exp(-x))
age  <- as.integer(pmin(pmax(rnorm(N, 60, 15), 18), 90))
los  <- as.integer(pmin(pmax(rgamma(N, 2.5, scale = 1.8), 1), 30))
dm   <- rbinom(N, 1, 0.35)
hf   <- rbinom(N, 1, 0.22)
copd <- rbinom(N, 1, 0.18)
logit <- -3.2 + 0.6 * hf + 0.5 * dm + 0.3 * copd + 0.02 * (age - 60) +
  0.2 * as.integer(los > 7)
readmitted <- rbinom(N, 1, sigmoid(logit))
df <- tibble(age = age, los_days = los, diabetes = dm, hf = hf, copd = copd,
             readmitted = readmitted)
FEATURES <- c("age", "los_days", "diabetes", "hf", "copd")
split <- initial_split(df, prop = 0.8, strata = readmitted)
train <- training(split)
test  <- testing(split)
X_tr <- train[FEATURES]; y_tr <- train$readmitted
X_te <- test[FEATURES];  y_te <- test$readmitted
model <- xgb.train(
  params = list(objective = "binary:logistic", max_depth = 4, eta = 0.05,
                eval_metric = "auc"),
  data = xgb.DMatrix(as.matrix(X_tr), label = y_tr),
  nrounds = 200, verbose = 0
)
auc <- as.numeric(pROC::auc(y_te, predict(model, as.matrix(X_te)), quiet = TRUE))

2. Testing Pyramid for Health Analytics

         /\
        /  \
       / E2E \         <- End-to-end (1-5 tests: full pipeline smoke test)
      /--------\
     / Integration \   <- Integration (10-20: API endpoints, DB connections)
    /--------------\
   /   Unit Tests   \  <- Unit tests (100+: individual functions, classes)
  /------------------\

Clinical Model Testing Priorities:
  1. Data validation (correct types, ranges, no PHI leakage)
  2. Feature engineering (correct transforms, no data leakage)
  3. Model direction (high-risk > low-risk scores)
  4. Model performance (AUC above clinical minimum)
  5. API contract (correct request/response schemas)
  6. Reproducibility (same seed → same output)

3. Unit Tests: Data Pipeline

# Unit tests for data pipeline and model functions
validate_features <- function(df, required_cols, age_col = "age", target_col = "readmitted") {
  errors <- character()
  missing <- setdiff(required_cols, names(df))
  if (length(missing)) errors <- c(errors, paste("Missing columns:", paste(missing, collapse = ", ")))
  if (age_col %in% names(df) && !all(df[[age_col]] >= 18 & df[[age_col]] <= 90)) {
    errors <- c(errors, paste(age_col, "values outside 18-90 range"))
  }
  if (target_col %in% names(df) && !all(df[[target_col]] %in% c(0, 1))) {
    errors <- c(errors, paste(target_col, "must be binary 0/1"))
  }
  errors
}

compute_positive_rate <- function(y) mean(y)

compute_smr <- function(observed, expected) {
  if (expected <= 0) stop("Expected must be positive")
  observed / expected
}

clean_age <- function(age_series, min_age = 18, max_age = 90) {
  pmin(pmax(age_series, min_age), max_age)
}

df_valid <- tibble(
  age = c(65, 45, 72), los_days = c(5, 3, 12),
  diabetes = c(1, 0, 1), hf = c(0, 0, 1), copd = c(0, 0, 1), readmitted = c(0, 0, 1)
)
df_invalid <- df_valid
df_invalid$age[1] <- 150

pipeline_results <- test_that("data pipeline validation", {
  expect_equal(validate_features(df_valid, c(FEATURES, "readmitted")), character(0))
  expect_true(any(grepl("age", validate_features(df_invalid, FEATURES), ignore.case = TRUE)))
  df_missing <- df_valid %>% select(-hf)
  expect_true(any(grepl("hf", validate_features(df_missing, FEATURES))))
  y <- c(0, 0, 1, 1, 0)
  rate <- compute_positive_rate(y)
  expect_gte(rate, 0)
  expect_lte(rate, 1)
  expect_equal(rate, 0.4, tolerance = 1e-5)
  expect_equal(compute_smr(observed = 8, expected = 5.0), 1.6, tolerance = 1e-5)
  expect_error(compute_smr(observed = 3, expected = 0))
  cleaned <- clean_age(c(15, 50, 100))
  expect_equal(min(cleaned), 18)
  expect_equal(max(cleaned), 90)
})

compute_code_metrics <- function(source_code) {
  parsed <- tryCatch(parse(text = source_code, keep.source = TRUE), error = function(e) e)
  if (inherits(parsed, "error")) return(list(error = parsed$message))
  pd <- getParseData(parsed)
  n_fn <- sum(pd$token == "FUNCTION", na.rm = TRUE)
  lines <- strsplit(source_code, "\n", fixed = TRUE)[[1]]
  code_lines <- lines[nzchar(trimws(lines)) & !startsWith(trimws(lines), "#")]
  list(
    lines_of_code = length(code_lines),
    num_functions = n_fn,
    num_classes   = sum(grepl("R6Class|setClass", source_code)),
    comment_lines = sum(startsWith(trimws(lines), "#"))
  )
}

8. Knowledge Check

  1. A test checks auc > 0.65. The AUC is 0.66, so the test passes. Six months later a data bug makes AUC=0.67 look good but the data is corrupted. What additional tests would catch this?
  2. Your CI pipeline takes 45 minutes to run. 12 developers each push 5 times/day. What optimisations reduce this to <10 minutes?
  3. A property-based test found that age=18, los_days=1 all zeros returns prob=0.0001. Is this a bug?
  4. Pre-commit hooks auto-format an R script (styler) and change 200 lines. How does this affect reproducibility?
  5. Write a test that verifies the model’s AUC on a held-out test set is within 0.05 of the AUC logged in MLflow for the production model.
# Exercise 5 — MLflow AUC consistency test
test_model_auc_consistent_with_registry <- function(model_obj, X_test, y_test,
                                                    registry_path = file.path(mod08_dir, "model_registry.json"),
                                                    tolerance = 0.05) {
  if (!file.exists(registry_path)) {
    cat("SKIP: No model registry file found\n")
    return(TRUE)
  }
  registry <- fromJSON(registry_path, simplifyVector = FALSE)
  prod_versions <- Filter(function(v) identical(v$stage, "Production"), registry$version_history)
  if (!length(prod_versions)) {
    cat("SKIP: No production version in registry\n")
    return(TRUE)
  }
  registered_auc <- prod_versions[[length(prod_versions)]]$auc
  current_auc <- as.numeric(pROC::auc(y_test, predict(model_obj, as.matrix(X_test)), quiet = TRUE))
  auc_diff <- abs(current_auc - registered_auc)
  passed <- auc_diff <= tolerance
  cat(sprintf("Registry AUC : %.4f\n", registered_auc))
  cat(sprintf("Current AUC  : %.4f\n", current_auc))
  cat(sprintf("Difference   : %.4f (tolerance=%.2f)\n", auc_diff, tolerance))
  cat(sprintf("Test result  : %s\n", if (passed) "PASS" else "FAIL - model may have drifted!"))
  passed
}

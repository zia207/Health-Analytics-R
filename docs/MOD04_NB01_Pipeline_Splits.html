2. Why Temporal Splits Matter in Healthcare

Random splitting of clinical data leaks future information into training when: - The same patient appears multiple times (train on visit 1, test on visit 2 of the same patient) - Care patterns change over time (COVID, policy changes, new guidelines) - Models evaluated on temporal holdouts reflect real-world deployment performance

Split type Use when Risk
Random stratified Cross-sectional data, no time structure Data leakage if temporal trends exist
Temporal (chronological) Time-ordered admissions, EHR data Better reflects deployment conditions
Patient-level Same patient across visits Prevents same-patient leakage
Geographic Multi-site data Tests transportability

  select(all_of(c(FEATURES, TARGET))) %>%
  mutate(readmitted = factor(readmitted, levels = c(0, 1)))

# Strategy comparison — SMOTE-style interpolation, under-sampling, SMOTE+Tomek
smote_resample <- function(x, y, seed = 42) {
  set.seed(seed)
  pos <- which(y == 1); neg <- which(y == 0)
  n_need <- length(neg) - length(pos)
  if (n_need <= 0) return(list(x = x, y = y))
  xp <- x[pos, , drop = FALSE]
  i1 <- sample(nrow(xp), n_need, replace = TRUE)
  i2 <- sample(nrow(xp), n_need, replace = TRUE)
  lam <- runif(n_need)
  syn <- xp[i1, , drop = FALSE] * lam + xp[i2, , drop = FALSE] * (1 - lam)
  list(x = rbind(x, syn), y = c(y, rep(1L, n_need)))
}
under_resample <- function(x, y, seed = 42) {
  set.seed(seed)
  pos <- which(y == 1); neg <- which(y == 0)
  n <- min(length(pos), length(neg))
  idx <- c(sample(pos, n), sample(neg, n))
  list(x = x[idx, , drop = FALSE], y = y[idx])
}
tomek_clean <- function(x, y) {
  # Drop majority points whose nearest neighbour is minority (Tomek-style)
  d <- as.matrix(dist(x))
  diag(d) <- Inf
  nn <- apply(d, 1, which.min)
  drop <- which(y == 0 & y[nn] == 1)
  keep <- setdiff(seq_along(y), drop)
  list(x = x[keep, , drop = FALSE], y = y[keep])
}

6. Cross-Validation Strategies

# Compare stratified k-fold vs time-series CV on the training set
cv_auc_fold <- function(x, y, tr_idx, te_idx) {
  dtr <- as_tibble(x[tr_idx, , drop = FALSE]) %>% mutate(y = y[tr_idx])
  dte <- as_tibble(x[te_idx, , drop = FALSE])
  w   <- ifelse(y[tr_idx] == 1, 0.5 / mean(y[tr_idx] == 1),
                0.5 / mean(y[tr_idx] == 0))
  fit <- glm(y ~ ., data = dtr, family = binomial, weights = w)
  auc_prob(y[te_idx], predict(fit, newdata = dte, type = "response"))
}

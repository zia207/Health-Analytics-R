Health Analytics with R · Module 07: Geospatial & Spatial Epidemiology

Learning objectives - Understand spatial data types: points, polygons, rasters, and their health applications - Work with sf for spatial data manipulation - Create publication-quality choropleth maps with quintile and Jenks classification - Apply coordinate reference systems (CRS) and spatial joins - Map health outcomes with appropriate colour schemes and legends - Compute and visualise age-standardised mortality rates (SMR)

1. Setup & Synthetic County Health Dataset

library(tidyverse)
library(sf)
library(classInt)
library(patchwork)

set.seed(42)
out_dir <- file.path(tempdir(), "mod07")
dir.create(out_dir, showWarnings = FALSE, recursive = TRUE)

theme_set(
  theme_minimal(base_size = 11) +
    theme(
      plot.background = element_rect(fill = "white", colour = NA),
      panel.background = element_rect(fill = "white", colour = NA),
      panel.grid.minor = element_blank()
    )
)

spatial_smooth <- function(values, row_idx, col_idx, nrow, ncol, sigma = 1.5) {
  grid <- matrix(0, nrow = nrow, ncol = ncol)
  for (i in seq_along(values)) {
    grid[row_idx[i], col_idx[i]] <- values[i]
  }
  k <- ceiling(3 * sigma)
  xs <- (-k):k
  kern <- outer(dnorm(xs, sd = sigma), dnorm(xs, sd = sigma))
  kern <- kern / sum(kern)
  pad <- k
  gp <- matrix(0, nrow + 2 * pad, ncol + 2 * pad)
  gp[(pad + 1):(pad + nrow), (pad + 1):(pad + ncol)] <- grid
  out <- matrix(0, nrow, ncol)
  for (r in seq_len(nrow)) {
    for (c in seq_len(ncol)) {
      out[r, c] <- sum(gp[r:(r + 2 * k), c:(c + 2 * k)] * kern)
    }
  }
  mapply(function(r, c) out[r, c], row_idx, col_idx)
}

NROW <- 10
NCOL <- 20
N_COUNTIES <- NROW * NCOL
row_idx <- rep(seq_len(NROW), each = NCOL)
col_idx <- rep(seq_len(NCOL), times = NROW)

# Geographic coordinates (lon/lat-like)
lon <- -120 + (col_idx - 1) * 2.5 + rnorm(N_COUNTIES, 0, 0.3)
lat <- 28 + (row_idx - 1) * 2.0 + rnorm(N_COUNTIES, 0, 0.2)

# Socioeconomic and demographic covariates
pct_poverty   <- 10 + 8 * rbeta(N_COUNTIES, 2, 5) + 3 * sin(lon / 20)
pct_uninsured <- 8 + 6 * rbeta(N_COUNTIES, 2, 4) + 0.3 * pct_poverty
pct_black     <- 5 + 20 * rbeta(N_COUNTIES, 1, 4)
pct_rural     <- 30 + 40 * rbeta(N_COUNTIES, 3, 3)
pm25          <- 8 + 4 * rgamma(N_COUNTIES, shape = 2, scale = 1) + 0.05 * pct_rural
median_income <- 45000 - 800 * pct_poverty + rnorm(N_COUNTIES, 0, 3000)
population    <- as.integer(rlnorm(N_COUNTIES, 10.5, 1.0))

# Spatially autocorrelated latent risk factor
spatial_risk <- spatial_smooth(rnorm(N_COUNTIES), row_idx, col_idx, NROW, NCOL, sigma = 2)

# Outcome: age-adjusted CVD mortality rate (per 100k)
TRUE_POVERTY_EFFECT <- 1.2  # each 1pp poverty -> 1.2 extra deaths per 100k
cvd_rate <- (180 + TRUE_POVERTY_EFFECT * pct_poverty + 0.8 * pct_uninsured +
               0.5 * pm25 + 15 * spatial_risk + rnorm(N_COUNTIES, 0, 12))

# Observed deaths and expected (for SMR/SIR)
expected_deaths <- (cvd_rate / 100000) * population * 5  # 5-year count
observed_deaths <- rpois(N_COUNTIES, expected_deaths)
national_rate   <- mean(cvd_rate) / 100000
expected_std    <- national_rate * population * 5
smr             <- observed_deaths / pmax(expected_std, 1)

df <- tibble(
  county_id       = sprintf("C%03d", seq_len(N_COUNTIES) - 1),
  lon             = lon,
  lat             = lat,
  row             = row_idx,
  col             = col_idx,
  pct_poverty     = pct_poverty,
  pct_uninsured   = pct_uninsured,
  pct_black       = pct_black,
  pct_rural       = pct_rural,
  pm25            = pm25,
  median_income   = median_income,
  population      = population,
  cvd_rate        = cvd_rate,
  expected_deaths = expected_deaths,
  observed_deaths = observed_deaths,
  smr             = smr,
  spatial_risk    = spatial_risk
)

cat(sprintf(
  "County dataset: N=%d | CVD rate: %.1f/100k (SD=%.1f)\n",
  N_COUNTIES, mean(cvd_rate), sd(cvd_rate)
))
County dataset: N=200 | CVD rate: 218.9/100k (SD=12.4)
cat(sprintf(
  "Poverty: %.1f%% | Uninsured: %.1f%% | PM2.5: %.1f\n",
  mean(pct_poverty), mean(pct_uninsured), mean(pm25)
))
Poverty: 14.5% | Uninsured: 14.3% | PM2.5: 17.9

2. Spatial Data Structures in Health Analytics

Type Examples R tool
Points Hospital locations, patient addresses, monitoring stations sf POINT
Polygons Counties, census tracts, ZIP codes, health service areas sf POLYGON
Rasters Satellite imagery, PM2.5 grids, land use terra, raster
Lines Roads, rivers, transit routes sf LINESTRING
Grids Aggregated kernel density, interpolated surfaces gstat, terra

Key CRS concepts: - WGS 84 (EPSG:4326): latitude/longitude in degrees — for global data - NAD83 (EPSG:4269): North American datum — for US health data - Albers Equal Area (EPSG:5070): preserves area — for choropleth maps

# Create sf object with synthetic county centroids + bounding boxes
# Each county is a 2.5° x 2.0° rectangle
make_box <- function(lo, la) {
  st_polygon(list(matrix(
    c(lo - 1.25, la - 1.0,
      lo + 1.25, la - 1.0,
      lo + 1.25, la + 1.0,
      lo - 1.25, la + 1.0,
      lo - 1.25, la - 1.0),
    ncol = 2, byrow = TRUE
  )))
}

geoms <- mapply(make_box, df$lon, df$lat, SIMPLIFY = FALSE)
gdf <- st_sf(df, geometry = st_sfc(geoms, crs = 4326))
bb <- st_bbox(gdf)

cat(sprintf("sf object: %d counties | CRS: %s\n", nrow(gdf), st_crs(gdf)$input))
sf object: 200 counties | CRS: EPSG:4326
cat(sprintf(
  "Bounding box: xmin=%.2f ymin=%.2f xmax=%.2f ymax=%.2f\n",
  bb["xmin"], bb["ymin"], bb["xmax"], bb["ymax"]
))
Bounding box: xmin=-121.70 ymin=26.60 xmax=-70.85 ymax=47.48

3. Choropleth Map — CVD Mortality Rate

sf_choropleth <- function(gdf_in, column, title, palette = "RdYlGn",
                          direction = -1, legend = "Rate per 100k") {
  ggplot(gdf_in) +
    geom_sf(aes(fill = .data[[column]]), colour = "white", linewidth = 0.15) +
    scale_fill_distiller(name = legend, palette = palette, direction = direction) +
    labs(title = title, x = "Longitude", y = "Latitude") +
    coord_sf()
}

p_cvd <- sf_choropleth(gdf, "cvd_rate", "CVD Mortality Rate (per 100k)",
                       "RdYlGn", -1, "Rate/100k")
p_pov <- sf_choropleth(gdf, "pct_poverty", "Poverty Rate (%)",
                       "YlOrRd", 1, "Poverty (%)")
p_smr <- sf_choropleth(gdf, "smr", "SMR > 1 = excess mortality | < 1 = deficit",
                       "RdBu", -1, "SMR")
p_pm  <- sf_choropleth(gdf, "pm25", "PM2.5 (µg/m³)",
                       "YlOrBr", 1, "PM2.5")

p_atlas <- (p_cvd | p_pov) / (p_smr | p_pm) +
  plot_annotation(title = "County-Level Health Atlas: CVD Mortality & Risk Factors")

print(p_atlas)
ggsave(file.path(out_dir, "choropleth_atlas.png"), p_atlas,
       width = 16, height = 11, dpi = 150)

cat(sprintf(
  "Counties with SMR > 1.2 (excess mortality): %d (%.1f%%)\n",

4. Classification Schemes for Health Maps

n_classes <- 5
vals <- gdf$cvd_rate

brks_equal <- classIntervals(vals, n = n_classes, style = "equal")
brks_quant <- classIntervals(vals, n = n_classes, style = "quantile")
brks_jenks <- classIntervals(vals, n = n_classes, style = "jenks")

classify_map <- function(gdf_in, brks, title) {
  gdf_in$class <- cut(gdf_in$cvd_rate, breaks = brks$brks, include.lowest = TRUE)
  ggplot(gdf_in) +
    geom_sf(aes(fill = class), colour = "white", linewidth = 0.15) +
    scale_fill_brewer(name = "Rate/100k", palette = "RdYlGn", direction = -1) +
    labs(title = paste("CVD Rate:", title), x = "Lon", y = "Lat")
}

p_eq <- classify_map(gdf, brks_equal, "Equal Interval")
p_q  <- classify_map(gdf, brks_quant, "Quantile (quintiles)")
p_j  <- classify_map(gdf, brks_jenks, "Natural Breaks (Jenks)")

p_class <- p_eq | p_q | p_j
p_class <- p_class +
  plot_annotation(title = "Classification Scheme Comparison — CVD Mortality")
print(p_class)
ggsave(file.path(out_dir, "classification_schemes.png"), p_class,
       width = 18, height = 5, dpi = 150)

# True age-specific rates by county
base_rates <- c(50, 200, 600, 1800)  # per 100k per age group
age_specific_rates <- outer(df$cvd_rate / 200, base_rates)
age_specific_rates <- age_specific_rates + matrix(rnorm(N_COUNTIES * 4, 0, 20), ncol = 4)

6. Spatial Join — Linking Point Data to Polygons

# Simulate hospital locations (point data) and join to county
set.seed(13)
N_HOSPITALS <- 50
hosp <- st_as_sf(
  tibble(
    lon        = runif(N_HOSPITALS, min(df$lon), max(df$lon)),
    lat        = runif(N_HOSPITALS, min(df$lat), max(df$lat)),
    hosp_beds  = pmin(pmax(as.integer(rlnorm(N_HOSPITALS, 5, 0.7)), 50), 2000),
    hosp_trauma = rbinom(N_HOSPITALS, 1, 0.3)
  ),
  coords = c("lon", "lat"),
  crs = 4326
)

# Assign each hospital to nearest county centroid
centroids <- st_centroid(gdf)
Warning: st_centroid assumes attributes are constant over geometries
nn_idx <- st_nearest_feature(hosp, centroids)

gdf$n_hospitals    <- 0
gdf$total_beds     <- 0
gdf$has_trauma_ctr <- 0
for (i in seq_len(N_HOSPITALS)) {
  c_idx <- nn_idx[i]
  gdf$n_hospitals[c_idx]    <- gdf$n_hospitals[c_idx] + 1
  gdf$total_beds[c_idx]     <- gdf$total_beds[c_idx] + hosp$hosp_beds[i]
  gdf$has_trauma_ctr[c_idx] <- gdf$has_trauma_ctr[c_idx] + hosp$hosp_trauma[i]
}
gdf$has_trauma_ctr <- as.integer(gdf$has_trauma_ctr > 0)
gdf$beds_per_1000  <- gdf$total_beds / gdf$population * 1000

p_beds <- ggplot(gdf) +
  geom_sf(aes(fill = beds_per_1000), colour = "white", linewidth = 0.15) +
  geom_sf(data = hosp, shape = 3, size = 2.5, colour = "#DC2626") +
  scale_fill_distiller(name = "Beds per 1,000 pop", palette = "Blues", direction = 1) +
  labs(title = "Hospital Beds per 1,000 Population", x = "Lon", y = "Lat")

corr_beds <- cor(gdf$beds_per_1000, gdf$cvd_rate)
p_corr <- ggplot(st_drop_geometry(gdf), aes(beds_per_1000, cvd_rate)) +
  geom_point(alpha = 0.5, size = 2, colour = "#DC2626") +
  geom_smooth(method = "lm", se = FALSE, colour = "black", linewidth = 1) +
  labs(
    x = "Beds per 1,000", y = "CVD rate per 100k",
    title = sprintf(
      "Hospital Capacity vs CVD Mortality (r=%.3f — ecological correlation only!)",
      corr_beds
    )
  )

p_join <- p_beds | p_corr
print(p_join)

       width = 14, height = 5, dpi = 150)

7. Knowledge Check

  1. A choropleth map uses quantile classification and shows your county in the highest quintile. A colleague uses equal-interval and shows it in the 4th quintile. Which is “correct”?
  2. SMR = 1.45 for a rural county. What does this mean, and what information do you need to know if this is statistically significant?
  3. You want to map CVD mortality for census tracts. Your denominator (population) comes from the 2020 Census but your numerator (deaths) covers 2018-2022. What are the key assumptions?
  4. The correlation between beds/1,000 and CVD rate is -0.31. A colleague concludes “more hospitals reduce CVD deaths.” What fallacy is this?
  5. Compute a bivariate choropleth: classify counties into 3x3 cells by both poverty rate and CVD mortality, and colour-code the 9 combinations.
# Exercise 5 — Bivariate choropleth (3x3 = 9 classes)
poverty_cat <- as.integer(cut(gdf$pct_poverty, breaks = quantile(gdf$pct_poverty, probs = 0:3 / 3),
                              include.lowest = TRUE, labels = FALSE)) - 1
cvd_cat     <- as.integer(cut(gdf$cvd_rate, breaks = quantile(gdf$cvd_rate, probs = 0:3 / 3),
                              include.lowest = TRUE, labels = FALSE)) - 1
gdf$bivariate_class <- poverty_cat * 3 + cvd_cat

biv_colors <- c(
  "0" = "#e8e8e8", "1" = "#dfb0d6", "2" = "#be64ac",
  "3" = "#ace4e4", "4" = "#a5add3", "5" = "#8c62aa",
  "6" = "#5ac8c8", "7" = "#5698b9", "8" = "#3b4994"
)
gdf$biv_lab <- factor(gdf$bivariate_class, levels = 0:8)
pov_lbl <- c("Low", "Med", "High")
cvd_lbl <- c("Low", "Med", "High")
lab_map <- sprintf("Pov:%s CVD:%s", pov_lbl[floor(0:8 / 3) + 1], cvd_lbl[(0:8) %% 3 + 1])
names(biv_colors) <- lab_map
gdf$biv_lab <- factor(lab_map[gdf$bivariate_class + 1], levels = lab_map)

p_biv <- ggplot(gdf) +
  geom_sf(aes(fill = biv_lab), colour = "white", linewidth = 0.15) +
  scale_fill_manual(name = NULL, values = biv_colors) +
  labs(
    title = "Bivariate Choropleth: Poverty Rate × CVD Mortality",
    subtitle = "blue=low/low, red=high/high, purple=high poverty/low CVD",
    x = "Longitude", y = "Latitude"
  ) +
  theme(legend.position = "bottom")

print(p_biv)
ggsave(file.path(out_dir, "bivariate_choropleth.png"), p_biv,
       width = 10, height = 6, dpi = 150)

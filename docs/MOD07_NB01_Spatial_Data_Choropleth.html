Module 07: Spatial Epidemiology

Choropleth Mapping

Learning objectives - Understand spatial data types: points, polygons, rasters, and their health applications - Work with sf for spatial data manipulation - Create publication-quality choropleth maps with quintile and Jenks classification - Apply coordinate reference systems (CRS) and spatial joins - Map health outcomes with appropriate colour schemes and legends - Compute and visualise standardised prevalence ratios (SMR analogue)

Data: Data/usa_data.csv — CDC adult diabetes prevalence and County Health Rankings covariates for 3,108 US counties (CONUS + DC)
Geometries: Data/US_COUNTY.shp — county polygons in NAD83 Albers (EPSG:5070)
Estimated time: 2.5 hours | Level: Intermediate
Libraries: sf, ggplot2, dplyr, classInt, patchwork

1. Setup & US County Health Dataset

library(tidyverse)
library(sf)
library(classInt)
library(patchwork)

set.seed(42)
out_dir <- file.path(tempdir(), "mod07")
dir.create(out_dir, showWarnings = FALSE, recursive = TRUE)

theme_set(
  theme_minimal(base_size = 11) +
    theme(
      plot.background = element_rect(fill = "white", colour = NA),
      panel.background = element_rect(fill = "white", colour = NA),
      panel.grid.minor = element_blank()
    )
)

theme_map <- function() {
  theme_void(base_size = 11) +
    theme(
      plot.background = element_rect(fill = "white", colour = NA),
      panel.background = element_rect(fill = "white", colour = NA),
      plot.title = element_text(face = "bold", size = 11, hjust = 0.5),
      plot.subtitle = element_text(size = 9, hjust = 0.5),
      legend.position = "right"
    )
}

# County Health Rankings + CDC diabetes prevalence (CONUS + DC)
raw <- read_csv("Data/usa_data.csv", show_col_types = FALSE)

# Centroids stored as NAD83 Albers metres (EPSG:5070) — convert to WGS84 lon/lat
pts <- st_as_sf(raw, coords = c("centroid_x_m", "centroid_y_m"), crs = 5070, remove = FALSE)
pts_ll <- st_transform(pts, 4326)
xy <- st_coordinates(pts_ll)

df <- tibble(
  county_id       = sprintf("%05d", as.integer(raw$fips)),
  fips            = as.integer(raw$fips),
  county          = raw$county,
  state_abbr      = raw$state_abbr,
  state_name      = raw$state_name,
  lon             = xy[, 1],
  lat             = xy[, 2],
  x_m             = raw$centroid_x_m,
  y_m             = raw$centroid_y_m,
  diabetes_prev   = raw$y_diabetes_prevalence_adults_with_diabetes,
  pct_65plus      = raw$chr_65_and_older_65_and_over,
  pct_exercise    = raw$chr_access_to_exercise_opportunities,
  pct_obesity     = raw$chr_adult_obesity,
  food_env_index  = raw$chr_food_environment_index,
  n_pcp           = raw$chr_primary_care_physicians,
  rural_pop       = raw$chr_rural,
  n_unemployed    = raw$chr_unemployment,
  n_uninsured     = raw$chr_uninsured
)

# Standardised prevalence ratio: county rate / national mean (SMR analogue)
df$smr <- df$diabetes_prev / mean(df$diabetes_prev, na.rm = TRUE)

N_COUNTIES <- nrow(df)
cat(sprintf(
  "County dataset: N=%d | Diabetes: %.1f%% (SD=%.1f)\n",
  N_COUNTIES, mean(df$diabetes_prev), sd(df$diabetes_prev)
))
County dataset: N=3108 | Diabetes: 10.5% (SD=2.2)
cat(sprintf(
  "Obesity: %.1f%% | 65+: %.1f%% | Food env. index: %.1f\n",
  mean(df$pct_obesity), mean(df$pct_65plus), mean(df$food_env_index)
))
Obesity: 36.2% | 65+: 20.1% | Food env. index: 7.4

2. Spatial Data Structures in Health Analytics

Type Examples R tool
Points Hospital locations, patient addresses, monitoring stations sf POINT
Polygons Counties, census tracts, ZIP codes, health service areas sf POLYGON
Rasters Satellite imagery, PM2.5 grids, land use terra, raster
Lines Roads, rivers, transit routes sf LINESTRING
Grids Aggregated kernel density, interpolated surfaces gstat, terra

Key CRS concepts: - WGS 84 (EPSG:4326): latitude/longitude in degrees — for global data - NAD83 (EPSG:4269): North American datum — for US health data - Albers Equal Area (EPSG:5070): preserves area — for choropleth maps

A choropleth map is a thematic map in which geographic regions are shaded in proportion to a statistical variable (disease rates, population density, income). In health analytics, choropleths visualise the geographic distribution of outcomes, risk factors, or access to care across counties, states, or census tracts.

# Join tabular health data with US county polygons
county_shapes <- st_read("Data/US_COUNTY.shp", quiet = TRUE)
county_shapes$FIPS <- as.integer(county_shapes$FIPS)
# Shannon County SD (46113) was renamed Oglala Lakota (46102)
county_shapes$FIPS[county_shapes$FIPS == 46113] <- 46102

gdf <- county_shapes %>%
  inner_join(df, by = c("FIPS" = "fips"))
bb <- st_bbox(gdf)

cat(sprintf("sf object: %d counties | CRS: %s\n", nrow(gdf), st_crs(gdf)$input))
sf object: 3108 counties | CRS: NAD_1983_Albers
cat(sprintf(
  "Bounding box (Albers metres): xmin=%.0f ymin=%.0f xmax=%.0f ymax=%.0f\n",
  bb["xmin"], bb["ymin"], bb["xmax"], bb["ymax"]
))
Bounding box (Albers metres): xmin=-2356114 ymin=268091 xmax=2258200 ymax=3172568

3. Choropleth Map — Diabetes Prevalence & Risk Factors

sf_choropleth <- function(gdf_in, column, title, palette = "RdYlGn",
                          direction = -1, legend = "Rate") {
  ggplot(gdf_in) +
    geom_sf(aes(fill = .data[[column]]), colour = "white", linewidth = 0.05) +
    scale_fill_distiller(name = legend, palette = palette, direction = direction) +
    labs(title = title) +
    theme_map()
}

p_diab <- sf_choropleth(gdf, "diabetes_prev", "Adult Diabetes Prevalence (%)",
                        "RdYlGn", -1, "Prevalence (%)")
p_ob   <- sf_choropleth(gdf, "pct_obesity", "Adult Obesity (%)",
                        "Spectral", -1, "Obesity (%)")
p_smr  <- sf_choropleth(gdf, "smr",
                        "SPR > 1 = above-average diabetes | < 1 = below average",
                        "RdBu", -1, "SPR")
p_food <- sf_choropleth(gdf, "food_env_index",
                        "Food Environment Index (0–10, higher = better)",
                        "YlGn", 1, "Index")

p_atlas <- (p_diab | p_ob) / (p_smr | p_food) +
  plot_annotation(title = "US County Health Atlas: Diabetes Prevalence & Risk Factors")

print(p_atlas)
ggsave(file.path(out_dir, "choropleth_atlas.png"), p_atlas,
       width = 16, height = 11, dpi = 150)

cat(sprintf(
  "Counties with SPR > 1.2 (excess prevalence): %d (%.1f%%)\n",
  sum(gdf$smr > 1.2), mean(gdf$smr > 1.2) * 100
))
Counties with SPR > 1.2 (excess prevalence): 513 (16.5%)
cat(sprintf(
  "Counties with SPR < 0.8 (deficit prevalence): %d (%.1f%%)\n",

4. Classification Schemes for Health Maps

There are multiple ways to classify continuous values into categories for health maps:

  • Equal interval: Divides the range of data into categories with equal-sized intervals.
  • Quantile: Splits the data so that each class contains the same number of observations (e.g. quintiles).
  • Natural Breaks (Jenks): Groups values to minimise within-class variance and maximise between-class differences.
n_classes <- 5
vals <- gdf$diabetes_prev

brks_equal <- classIntervals(vals, n = n_classes, style = "equal")
brks_quant <- classIntervals(vals, n = n_classes, style = "quantile")
brks_jenks <- classIntervals(vals, n = n_classes, style = "jenks")

classify_map <- function(gdf_in, brks, title) {
  gdf_in$class <- cut(gdf_in$diabetes_prev, breaks = brks$brks, include.lowest = TRUE)
  ggplot(gdf_in) +
    geom_sf(aes(fill = class), colour = "white", linewidth = 0.05) +
    scale_fill_brewer(name = "Prevalence (%)", palette = "RdYlGn", direction = -1) +
    labs(title = paste("Diabetes %:", title)) +
    theme_map()
}

p_eq <- classify_map(gdf, brks_equal, "Equal Interval")
p_q  <- classify_map(gdf, brks_quant, "Quantile (quintiles)")
p_j  <- classify_map(gdf, brks_jenks, "Natural Breaks (Jenks)")

p_class <- (p_eq | p_q | p_j) +
  plot_annotation(title = "Classification Scheme Comparison — Adult Diabetes Prevalence")
print(p_class)
ggsave(file.path(out_dir, "classification_schemes.png"), p_class,
       width = 18, height = 5, dpi = 150)

5. Age Structure — Crude vs Adjusted Prevalence

The published county diabetes rate is a crude prevalence. Age structure (% aged 65+) is observed, not simulated: older counties tend to show higher diabetes even if age-specific risk is the same. Residualising on % aged 65+ produces an age-adjusted analogue.

fit_age <- lm(diabetes_prev ~ pct_65plus, data = gdf)
gdf$diabetes_age_adj <- residuals(fit_age) + mean(gdf$diabetes_prev)

p_age <- ggplot(st_drop_geometry(gdf), aes(pct_65plus, diabetes_prev)) +
  geom_point(alpha = 0.25, size = 0.8, colour = "#2563EB") +
  geom_smooth(method = "lm", se = FALSE, colour = "#DC2626", linewidth = 1) +
  labs(
    x = "% aged 65+",
    y = "Adult diabetes prevalence (%)",
    title = "Crude diabetes vs age structure",
    subtitle = sprintf("r = %.3f", cor(gdf$pct_65plus, gdf$diabetes_prev))
  )

p_adj <- ggplot(st_drop_geometry(gdf), aes(diabetes_prev, diabetes_age_adj)) +
  geom_point(alpha = 0.25, size = 0.8, colour = "#DC2626") +
  geom_abline(slope = 1, intercept = 0, linetype = "dashed") +
  labs(
    x = "Crude diabetes (%)",
    y = "Age-residualised diabetes (%)",
    title = sprintf(
      "Crude vs age-adjusted prevalence (r = %.3f)",
      cor(gdf$diabetes_prev, gdf$diabetes_age_adj)
    )
  )

p_age_adj <- p_age | p_adj
print(p_age_adj)
`geom_smooth()` using formula = 'y ~ x'
ggsave(file.path(out_dir, "crude_vs_asmr.png"), p_age_adj,
       width = 14, height = 5, dpi = 150)
`geom_smooth()` using formula = 'y ~ x'
diff <- gdf$diabetes_age_adj - gdf$diabetes_prev
cat(sprintf(

  sum(diff > 0), mean(diff > 0) * 100
))

cat(sprintf("Mean difference (age-adjusted − crude): %.2f pp\n", mean(diff)))
Mean difference (age-adjusted − crude): 0.00 pp

6. Spatial Join — Linking Point Data to Polygons

Simulate hospital locations as points, place them inside county polygons (weighted by uninsured count as a population-size proxy), then join each hospital to the containing county. Overlay hospitals on a PCP-supply map and check the ecological correlation with diabetes.

set.seed(13)
N_HOSPITALS <- 400
weights <- pmax(gdf$n_uninsured, 1)
weights <- weights / sum(weights)
hosp_idx <- sample.int(nrow(gdf), N_HOSPITALS, replace = TRUE, prob = weights)

# Place each hospital randomly inside its assigned county polygon
place_in_poly <- function(geom) {
  bb <- st_bbox(geom)
  for (i in 1:20) {
    pt <- st_sfc(st_point(c(runif(1, bb["xmin"], bb["xmax"]),
                            runif(1, bb["ymin"], bb["ymax"]))),
                 crs = st_crs(geom))
    if (as.logical(st_within(pt, geom, sparse = FALSE))) return(pt[[1]])
  }
  st_centroid(geom)[[1]]
}

hosp_geoms <- lapply(hosp_idx, function(i) place_in_poly(gdf$geometry[i]))
pop_proxy <- gdf$n_uninsured[hosp_idx]
beds_mean <- 100 + (pop_proxy / max(pop_proxy)) * 700
hosp <- st_sf(
  tibble(
    hosp_beds   = pmin(pmax(as.integer(rlnorm(N_HOSPITALS, log(beds_mean), 0.4)), 50), 1500),
    hosp_trauma = rbinom(N_HOSPITALS, 1, pmin(pmax(pop_proxy / max(pop_proxy) * 0.6, 0.2), 0.7))
  ),
  geometry = st_sfc(hosp_geoms, crs = st_crs(gdf))
)

joined <- st_join(hosp, gdf["FIPS"], join = st_within, left = TRUE)
# Fallback: nearest county if a point missed its polygon
miss <- is.na(joined$FIPS)
if (any(miss)) {
  joined$FIPS[miss] <- gdf$FIPS[st_nearest_feature(hosp[miss, ], gdf)]
}

hosp_agg <- joined %>%
  st_drop_geometry() %>%
  group_by(FIPS) %>%
  summarise(
    n_hospitals    = n(),
    total_beds     = sum(hosp_beds),
    has_trauma_ctr = as.integer(any(hosp_trauma == 1)),
    .groups = "drop"
  )

gdf <- gdf %>%
  left_join(hosp_agg, by = "FIPS") %>%
  mutate(
    n_hospitals    = replace_na(n_hospitals, 0),
    total_beds     = replace_na(total_beds, 0),
    has_trauma_ctr = replace_na(has_trauma_ctr, 0L)
  )

p_pcp <- ggplot(gdf) +
  geom_sf(aes(fill = n_pcp), colour = NA, linewidth = 0) +
  geom_sf(data = hosp, shape = 3, size = 1.4, colour = "#DC2626") +
  scale_fill_distiller(name = "PCP count", palette = "Blues", direction = 1) +
  labs(title = "PCPs (CHR) + simulated hospitals") +
  theme_map()

corr_pcp <- cor(gdf$n_pcp, gdf$diabetes_prev, use = "complete.obs")
p_corr <- ggplot(st_drop_geometry(gdf), aes(n_pcp, diabetes_prev)) +
  geom_point(alpha = 0.25, size = 0.8, colour = "#DC2626") +
  geom_smooth(method = "lm", se = FALSE, colour = "black", linewidth = 1) +
  labs(
    x = "Primary care physicians",
    y = "Adult diabetes prevalence (%)",
    title = sprintf("PCP supply vs diabetes (r=%.3f — ecological correlation only!)", corr_pcp)
  )

p_join <- p_pcp | p_corr
print(p_join)
`geom_smooth()` using formula = 'y ~ x'
ggsave(file.path(out_dir, "hospital_join.png"), p_join,
       width = 14, height = 5, dpi = 150)
`geom_smooth()` using formula = 'y ~ x'

7. Knowledge Check

  1. A choropleth map uses quantile classification and shows your county in the highest quintile. A colleague uses equal-interval and shows it in the 4th class. Which is “correct”?
  2. SPR = 1.45 for a rural county (diabetes / national mean). What does this mean, and what information do you need to know if this is statistically significant?
  3. You want to map diabetes prevalence for census tracts. Your denominator (population) comes from the 2020 Census but your numerator covers 2018–2022. What are the key assumptions?
  4. The correlation between PCP count and diabetes prevalence is negative. A colleague concludes “more doctors reduce diabetes.” What fallacy is this?
  5. Compute a bivariate choropleth: classify counties into 3×3 cells by both obesity and diabetes prevalence, and colour-code the 9 combinations.
# Exercise 5 — Bivariate choropleth (3x3 = 9 classes)
obesity_cat <- as.integer(cut(gdf$pct_obesity,
                              breaks = quantile(gdf$pct_obesity, probs = 0:3 / 3),
                              include.lowest = TRUE, labels = FALSE)) - 1
diab_cat <- as.integer(cut(gdf$diabetes_prev,
                           breaks = quantile(gdf$diabetes_prev, probs = 0:3 / 3),
                           include.lowest = TRUE, labels = FALSE)) - 1
gdf$bivariate_class <- obesity_cat * 3 + diab_cat

biv_colors <- c(
  "Ob:Low Diab:Low"  = "#e8e8e8", "Ob:Low Diab:Med"  = "#dfb0d6", "Ob:Low Diab:High"  = "#be64ac",
  "Ob:Med Diab:Low"  = "#ace4e4", "Ob:Med Diab:Med"  = "#a5add3", "Ob:Med Diab:High"  = "#8c62aa",
  "Ob:High Diab:Low" = "#5ac8c8", "Ob:High Diab:Med" = "#5698b9", "Ob:High Diab:High" = "#3b4994"
)
ob_lbl <- c("Low", "Med", "High")
db_lbl <- c("Low", "Med", "High")
lab_map <- sprintf("Ob:%s Diab:%s", ob_lbl[floor(0:8 / 3) + 1], db_lbl[(0:8) %% 3 + 1])
gdf$biv_lab <- factor(lab_map[gdf$bivariate_class + 1], levels = names(biv_colors))

p_biv <- ggplot(gdf) +
  geom_sf(aes(fill = biv_lab), colour = "white", linewidth = 0.05) +
  scale_fill_manual(name = NULL, values = biv_colors) +
  labs(
    title = "Bivariate Choropleth: Adult Obesity × Diabetes Prevalence",
    subtitle = "light = low/low, dark = high/high"
  ) +
  theme_map() +
  theme(legend.position = "bottom")

print(p_biv)
ggsave(file.path(out_dir, "bivariate_choropleth.png"), p_biv,
       width = 10, height = 6, dpi = 150)

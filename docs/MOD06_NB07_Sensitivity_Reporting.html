Sensitivity Analysis & Reporting

Learning objectives - Conduct E-value analysis for unmeasured confounding - Implement Rosenbaum bounds for observational studies - Perform propensity score trimming and overlap assessment - Build a comprehensive causal inference reporting template - Apply placebo tests and refutation checks - Communicate uncertainty from multiple sensitivity analyses

2. E-Value — Unmeasured Confounding Sensitivity

The E-value (VanderWeele & Ding, 2017) answers:

“How strong would an unmeasured confounder have to be (in terms of its association with both treatment and outcome) to fully explain away the observed effect?”

A large E-value means the result is robust to unmeasured confounding.

Formula for OR:

E-value = OR + sqrt(OR * (OR - 1))     if OR >= 1
E-value = 1/OR + sqrt(1/OR * (1/OR - 1))  if OR < 1
# Fit adjusted model
model_adj <- glm(
  mi ~ statin + age + smoking + chd_history + ldl_baseline + ses_score,
  data = df, family = binomial()
)
tid_statin <- broom::tidy(model_adj, conf.int = TRUE, exponentiate = TRUE) |>
  filter(term == "statin")
adj_or   <- tid_statin$estimate
adj_or_lo <- tid_statin$conf.low
adj_or_hi <- tid_statin$conf.high

evalue_or <- function(OR) {
  if (OR >= 1) {
    OR + sqrt(OR * (OR - 1))
  } else {
    inv <- 1 / OR
    inv + sqrt(inv * (inv - 1))
  }
}

3. Overlap & Positivity Assessment

# Estimate PS
ps_model <- glm(
  as.formula(paste("statin ~", paste(COVARIATES, collapse = " + "))),
  data = df, family = binomial()
)
df$ps <- predict(ps_model, type = "response")

ps_min_treated <- min(df$ps[df$statin == 1])
ps_max_control <- max(df$ps[df$statin == 0])
in_support  <- df$ps >= ps_min_treated & df$ps <= ps_max_control
pct_support <- mean(in_support) * 100

p_ps <- ggplot(df, aes(ps, fill = factor(statin))) +
  geom_histogram(aes(y = after_stat(density)), bins = 40, alpha = 0.6, position = "identity") +
  annotate("rect", xmin = ps_min_treated, xmax = ps_max_control,
           ymin = -Inf, ymax = Inf, fill = "#16A34A", alpha = 0.10) +
  scale_fill_manual(
    values = c("0" = "#2563EB", "1" = "#DC2626"),
    labels = c("Control", "Treated"),
    name = NULL
  ) +
  labs(
    x = "Propensity score",
    y = "Density",
    title = sprintf("PS Distribution: Overlap Check (common support %.0f%%)", pct_support)
  )

# Effective sample size vs trimming threshold
trim_thresholds <- seq(0.05, 0.40, length.out = 30)
ess_vals <- vapply(trim_thresholds, function(t) {
  df_trim <- df[df$ps >= t & df$ps <= 1 - t, ]
  n <- nrow(df_trim)
  w_t <- df_trim$statin * 1 / pmin(pmax(df_trim$ps, t), 1 - t)
  (n^2) / ((sum(w_t)^2) / n)
}, numeric(1))
ess_df <- tibble(threshold = trim_thresholds, ess = ess_vals)

p_ess <- ggplot(ess_df, aes(threshold, ess)) +
  geom_line(colour = "#16A34A", linewidth = 1) +
  labs(
    x = "PS trimming threshold",
    y = "Effective sample size",
    title = "Effective Sample Size vs Trimming"
  )

p_overlap <- p_ps + p_w + p_ess

4. Placebo Outcome & Refutation Tests

# Test 1: Placebo outcome — an outcome that should NOT be affected by treatment
# Eye color should not be caused by statin use!

  as.formula(paste("eye_color ~ statin +", paste(COVARIATES, collapse = " + "))),
  data = df, family = binomial()
)
placebo_or <- exp(unname(coef(placebo_mod)[["statin"]]))

6. Knowledge Check

  1. E-value = 2.1 for your statin analysis. Is this large or small? What known confounders (e.g. healthy user bias) have RRs in this range?
  2. Your PS distribution shows poor overlap for elderly patients (age >80). What are your options?
  3. The permutation p-value is 0.001. What does this tell you about the observed association?
  4. A placebo outcome (baseline albumin from before statin initiation) shows OR=1.12. Should you be concerned?
  5. Calculate Rosenbaum’s gamma (how much unmeasured confounding would change the conclusion) using a simple sign test.
# Exercise 5 — simplified Rosenbaum gamma bound
# For a binary outcome, test how much gamma (odds of differential unmeasured confounding)
# would be needed to make the result non-significant

# Use matched pairs analysis (simplified)
# In matched pairs, under gamma=1 (no hidden bias), we expect binomial with p=0.5
set.seed(42)
# Find pseudo-matched pairs (PS matching)
simple_match <- function(df, ps_col = "ps", treatment = "statin") {
  treated <- df[df[[treatment]] == 1, ]
  control <- df[df[[treatment]] == 0, ]
  treated <- treated[order(treated[[ps_col]]), ]
  control <- control[order(control[[ps_col]]), ]
  n_pairs <- min(nrow(treated), nrow(control))
  list(
    y_t = treated$mi[seq_len(n_pairs)],
    y_c = control$mi[seq_len(n_pairs)]
  )
}

pairs <- simple_match(df)
y_t <- pairs$y_t
y_c <- pairs$y_c
# Discordant pairs (pairs where outcome differs)
disc <- y_t != y_c
n_disc <- sum(disc)
n_treated_wins <- sum((y_t < y_c) & disc)  # treated has LOWER MI rate
cat(sprintf("Discordant pairs: %d | Treated outcome better: %d\n",
            n_disc, n_treated_wins))
Discordant pairs: 80 | Treated outcome better: 40
# Under gamma=1: test statistic ~ Binomial(n_disc, 0.5)
p_gamma1 <- pbinom(n_treated_wins, n_disc, 0.5)  # one-sided
cat(sprintf("p-value at gamma=1 (no hidden bias): %.6f\n", p_gamma1))
p-value at gamma=1 (no hidden bias): 0.544464
# Find gamma that makes p = 0.05
p_at_gamma <- function(gamma) {
  p_plus <- gamma / (1 + gamma)
  pbinom(n_treated_wins, n_disc, p_plus) - 0.05
}

gamma_threshold <- tryCatch(
  uniroot(p_at_gamma, interval = c(1, 10))$root,
  error = function(e) NA_real_
)
if (!is.na(gamma_threshold)) {
  cat(sprintf("Gamma threshold (Rosenbaum): %.2f\n", gamma_threshold))
  cat(sprintf("  Unmeasured confounding must change treatment odds by >%.2f-fold\n",
              gamma_threshold))
  cat("  to make the result non-significant at alpha=0.05\n")
} else {
  cat("Result is robust to extreme levels of unmeasured confounding (gamma > 10)\n")
}

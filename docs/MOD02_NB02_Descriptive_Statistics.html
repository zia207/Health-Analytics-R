Module 02 — Descriptive Statistics for Clinical Variables

Descriptive Statistics for Clinical Variables

Learning objectives - Compute prevalence, incidence, and rates with correct denominators - Build stratified summaries by age group, sex, and payer - Detect and handle right-skewed clinical distributions (LOS, charges) - Apply age-standardisation (direct method) to remove age confounding - Produce a publication-ready Table 1 with confidence intervals

Estimated time: 1.5 hours | Level: Beginner → Intermediate | Libraries: dplyr / tibble, stats

1. Setup & Dataset

library(tidyverse)

skewness <- function(x) {
  x <- x[!is.na(x)]
  n <- length(x)
  m <- mean(x)
  s <- sd(x)
  if (n < 3 || s == 0) return(NA_real_)
  mean((x - m)^3) / (s^3)
}

2. Prevalence, Incidence & Rates

# ── 2.1  Comorbidity prevalence ──────────────────────────────────────────────
comorbidities <- c("diabetes", "hypertension", "ckd", "heart_failure")

wilson_ci <- function(k, n, alpha = 0.05) {
  z   <- qnorm(1 - alpha / 2)
  p   <- k / n
  den <- 1 + z^2 / n
  mid <- (p + z^2 / (2 * n)) / den
  hw  <- (z * sqrt(p * (1 - p) / n + z^2 / (4 * n^2))) / den
  c(round((mid - hw) * 100, 1), round((mid + hw) * 100, 1))
}

5. Publication-Ready Table 1

table1 <- function(df, stratify, continuous, categorical) {
  groups <- sort(unique(na.omit(df[[stratify]])))
  records <- list()

  for (col in continuous) {
    sk <- abs(skewness(df[[col]]))
    use_median <- !is.na(sk) && sk > 1
    stat_label <- if (use_median) "median [IQR]" else "mean ± SD"
    row <- list(Variable = sprintf("%s (%s)", col, stat_label))
    for (g in groups) {
      vals <- df[[col]][df[[stratify]] == g]
      vals <- vals[!is.na(vals)]
      key  <- paste0(stratify, "=", g)
      if (use_median) {
        row[[key]] <- sprintf("%.1f [%.1f–%.1f]",
                              median(vals), quantile(vals, 0.25), quantile(vals, 0.75))
      } else {
        row[[key]] <- sprintf("%.1f ± %.1f", mean(vals), sd(vals))
      }
    }
    records <- append(records, list(row))
  }

  for (col in categorical) {
    records <- append(records, list(list(Variable = col)))
    cats <- sort(unique(na.omit(df[[col]])))
    for (cat in cats) {
      row <- list(Variable = paste0("  ", as.character(cat)))
      for (g in groups) {
        sub <- df[df[[stratify]] == g, ]
        n   <- sum(sub[[col]] == cat, na.rm = TRUE)
        pct <- n / nrow(sub) * 100
        row[[paste0(stratify, "=", g)]] <- sprintf("%d (%.1f%%)", n, pct)
      }
      records <- append(records, list(row))
    }
  }

  bind_rows(lapply(records, as_tibble))
}

6. Statistical Tests for Group Differences

cat("Statistical tests: readmitted vs not readmitted\n\n")

# Mann-Whitney U — skewed continuous
for (pair in list(c("los_days", "LOS"), c("age", "Age"))) {
  col   <- pair[[1]]
  label <- pair[[2]]
  g0 <- df[[col]][df$readmitted_30 == 0]
  g1 <- df[[col]][df$readmitted_30 == 1]
  g0 <- g0[!is.na(g0)]
  g1 <- g1[!is.na(g1)]
  sk <- skewness(df[[col]])
  if (abs(sk) > 0.5) {
    wt <- wilcox.test(g0, g1, alternative = "two.sided", exact = FALSE)
    cat(sprintf("%s (skewed → Mann-Whitney U): U=%.0f, p=%.4f\n",
                label, wt$statistic, wt$p.value))
  } else {
    tt <- t.test(g0, g1)
    cat(sprintf("%s (normal → t-test): t=%.3f, p=%.4f\n",
                label, unname(tt$statistic), tt$p.value))
  }
}

2. Test Selection Framework

Outcome type?
├── Categorical (proportions) ─────────────────────────────────────────────────
│     ├── 2 groups, expected counts ≥5 → Chi-square test
│     ├── 2 groups, small N or sparse cells → Fisher's exact test
│     └── >2 groups → Chi-square (or log-linear model)
│
├── Continuous, 2 groups ──────────────────────────────────────────────────────
│     ├── Check normality (Shapiro-Wilk n<50, D'Agostino n≥50)
│     │     ├── Normal + equal variance → Independent samples t-test
│     │     ├── Normal + unequal variance → Welch's t-test
│     │     └── Non-normal (LOS, charges) → Mann-Whitney U
│     └── Paired / matched → Paired t-test (normal) or Wilcoxon signed-rank
│
├── Continuous, >2 groups ─────────────────────────────────────────────────────
│     ├── Normal + equal variance → One-way ANOVA + post-hoc Tukey HSD
│     └── Non-normal → Kruskal-Wallis + Dunn's test
│
└── Count / rate outcome → Poisson test or negative binomial

3. Normality Assessment

# D'Agostino-Pearson analogue: moments test via Pearson's chi-square of skew + kurtosis
dagostino_pearson <- function(x) {
  x <- x[!is.na(x)]
  n <- length(x)
  m3 <- mean((x - mean(x))^3)
  m4 <- mean((x - mean(x))^4)
  s  <- sd(x)
  skew <- m3 / s^3
  kurt <- m4 / s^4 - 3
  z_skew <- skew / sqrt(6 / n)
  z_kurt <- kurt / sqrt(24 / n)
  stat <- z_skew^2 + z_kurt^2
  p <- pchisq(stat, df = 2, lower.tail = FALSE)
  list(statistic = stat, p.value = p)
}

normality_report <- function(series, name) {
  s <- series[!is.na(series)]
  skewness <- mean((s - mean(s))^3) / sd(s)^3
  if (length(s) < 50) {
    sw <- shapiro.test(s)
    stat <- unname(sw$statistic)
    p <- sw$p.value
    test_name <- "Shapiro-Wilk"
  } else {
    dp <- dagostino_pearson(s)
    stat <- dp$statistic
    p <- dp$p.value
    test_name <- "D'Agostino"
  }
  normal <- p > 0.05
  cat(sprintf(
    "  %-20s: skew=%+.2f | %s p=%.4f → %s\n",
    name, skewness, test_name, p,
    if (normal) "NORMAL" else "NON-NORMAL"
  ))
  normal
}

# ── 4.3  Relative Risk and Odds Ratio ─────────────────────────────────────────
rr_or_correct <- function(exposed, outcome, label = "") {
  a <- sum(exposed == 1 & outcome == 1)
  b <- sum(exposed == 1 & outcome == 0)
  c <- sum(exposed == 0 & outcome == 1)
  d <- sum(exposed == 0 & outcome == 0)
  n_exp <- a + b; n_unexp <- c + d
  risk_e <- a / n_exp; risk_u <- c / n_unexp
  RR <- risk_e / risk_u
  OR <- if (b * c > 0) (a * d) / (b * c) else NA_real_
  se_log_rr <- sqrt((1 - risk_e) / a + (1 - risk_u) / c)
  se_log_or <- sqrt(1 / a + 1 / b + 1 / c + 1 / d)
  rr_lo <- exp(log(RR) - 1.96 * se_log_rr)
  rr_hi <- exp(log(RR) + 1.96 * se_log_rr)
  or_lo <- exp(log(OR) - 1.96 * se_log_or)
  or_hi <- exp(log(OR) + 1.96 * se_log_or)
  cat(sprintf("\n%s\n", label))
  cat(sprintf("  RR = %.2f (95%% CI: %.2f–%.2f)\n", RR, rr_lo, rr_hi))
  cat(sprintf("  OR = %.2f (95%% CI: %.2f–%.2f)\n", OR, or_lo, or_hi))
}

m_sbp <- df$sbp[df$sex == "M"]

# Brown-Forsythe (median-centred Levene) for equal variance
levene_bf <- function(x, g) {
  med <- tapply(x, g, median)
  z <- abs(x - med[as.character(g)])
  summary(aov(z ~ factor(g)))[[1]][["Pr(>F)"]][1]
}
lev_p <- levene_bf(c(m_sbp, f_sbp), c(rep("M", length(m_sbp)), rep("F", length(f_sbp))))
equal_var <- lev_p > 0.05

tt <- t.test(m_sbp, f_sbp, var.equal = equal_var)
pooled_sd <- sqrt(
  ((length(m_sbp) - 1) * sd(m_sbp)^2 + (length(f_sbp) - 1) * sd(f_sbp)^2) /
    (length(m_sbp) + length(f_sbp) - 2)
)
cohen_d <- (mean(m_sbp) - mean(f_sbp)) / pooled_sd

7. Multiple Comparisons Correction

# Test all 10 comorbidities against readmission — naive and corrected
COMORBS <- c("diabetes", "hypertension", "heart_failure", "ckd", "copd",
             "obesity", "afib", "depression", "anxiety", "sleep_apnea")
# Simulate flags for missing comorbidities
set.seed(55)
for (c in c("ckd", "copd", "obesity", "afib", "depression", "anxiety", "sleep_apnea")) {
  if (!c %in% names(df)) df[[c]] <- rbinom(N, 1, 0.15)
}

raw_p <- vapply(COMORBS, function(c) {
  chisq.test(table(df[[c]], df$readmitted_30))$p.value
}, numeric(1))

p_bonf <- p.adjust(raw_p, method = "bonferroni")
p_bh   <- p.adjust(raw_p, method = "BH")
reject_bonf <- p_bonf < 0.05
reject_bh   <- p_bh < 0.05

mc_df <- tibble(
  Comorbidity    = COMORBS,
  `Raw p`        = round(raw_p, 4),
  `Sig (α=0.05)` = raw_p < 0.05,
  `Bonferroni p` = round(p_bonf, 4),
  `Sig (Bonf)`   = reject_bonf,
  `BH FDR p`     = round(p_bh, 4),
  `Sig (BH)`     = reject_bh
)

8. Statistical vs Clinical Significance

# Power analysis: what N is needed to detect a clinically meaningful difference?
# Two-sided two-sample t-test power (equal n, equal variance)
t_power <- function(d, n, alpha = 0.05) {
  nu  <- 2 * n - 2
  ncp <- d * sqrt(n / 2)
  crit <- qt(1 - alpha / 2, nu)
  pt(-crit, nu, ncp = ncp) + (1 - pt(crit, nu, ncp = ncp))
}

solve_n_power <- function(d, alpha = 0.05, power = 0.80) {
  n <- 10
  while (t_power(d, n, alpha) < power && n < 1e5) n <- n + 1
  n
}

Health Analytics with R · Module 02: EDA in Healthcare

Learning objectives - Build histograms and KDE plots for lab values and clinical outcomes - Annotate plots with clinical reference ranges and thresholds - Compare distributions across groups with box plots and violin plots - Combine multiple plot types into publication-ready multi-panel figures - Export figures at print resolution (300 dpi)

Estimated time: 2 hours | Level: Beginner → Intermediate | Libraries: dplyr / tibble, ggplot2, patchwork, ggridges, scales

1. Setup

library(tidyverse)
library(ggplot2)
library(patchwork)
library(ggridges)
library(scales)

mod02_dir <- file.path(tempdir(), "mod02")
dir.create(mod02_dir, showWarnings = FALSE, recursive = TRUE)

theme_health <- function(base_size = 12) {
  theme_minimal(base_size = base_size) +
    theme(
      panel.grid.minor  = element_blank(),
      plot.background   = element_rect(fill = "white", colour = NA),
      panel.background  = element_rect(fill = "white", colour = NA),
      legend.background = element_rect(fill = "white", colour = NA)
    )
}
theme_set(theme_health())

skewness <- function(x) {
  x <- x[!is.na(x)]
  n <- length(x)
  m <- mean(x)
  s <- sd(x)
  if (n < 3 || s == 0) return(NA_real_)
  mean((x - m)^3) / (s^3)
}

# ── Dataset ───────────────────────────────────────────────────────────────────
set.seed(42)
N <- 800
ages      <- as.integer(pmin(pmax(rnorm(N, 62, 16), 18), 95))
sexes     <- sample(c("M", "F"), N, replace = TRUE, prob = c(0.48, 0.52))
payers    <- sample(c("Medicare", "Medicaid", "Private", "Self-pay", "Other"),
                    N, replace = TRUE, prob = c(0.40, 0.22, 0.28, 0.07, 0.03))
drg_codes <- sample(c(470, 291, 392, 690, 871, 193, 247, 603), N, replace = TRUE)
drg_labels_map <- c(
  `470` = "Major joint replacement", `291` = "Heart failure & shock",
  `392` = "Esophagitis/misc GI",     `690` = "Kidney/UTI",
  `871` = "Septicemia",              `193` = "Simple pneumonia",
  `247` = "Perc cardiovasc w stent", `603` = "Cellulitis"
)
los_days   <- as.integer(pmin(pmax(rgamma(N, shape = 2.5, scale = 1.8), 1), 30))
charges    <- round(los_days * runif(N, 1800, 4200), 2)
charges[runif(N) < 0.08] <- NA_real_
primary_dx <- sample(c("E11.9", "I10", "N18.3", "I50.9", "J18.9", "A41.9", "M16.11", "N39.0"),
                     N, replace = TRUE)
admit_type <- sample(c("Emergency", "Elective", "Urgent", "Newborn"),
                     N, replace = TRUE, prob = c(0.52, 0.30, 0.16, 0.02))
set.seed(99)
df <- tibble(
  patient_id    = sprintf("PT-%05d", seq_len(N)),
  age           = ages,
  sex           = sexes,
  payer         = payers,
  drg_code      = drg_codes,
  drg_label     = unname(drg_labels_map[as.character(drg_codes)]),
  primary_dx    = primary_dx,
  admit_type    = admit_type,
  los_days      = los_days,
  total_charge  = charges,
  diabetes      = rbinom(N, 1, 0.32),
  hypertension  = rbinom(N, 1, 0.45),
  ckd           = rbinom(N, 1, 0.15),
  heart_failure = rbinom(N, 1, 0.18),
  readmitted_30 = rbinom(N, 1, 0.14)
)
df$age_group <- cut(df$age, breaks = c(17, 44, 64, 74, 95),
                    labels = c("18-44", "45-64", "65-74", "75+"))

# Synthetic lab values
set.seed(21)
df$glucose    <- round(pmin(pmax(rnorm(N, 105, 28), 50), 400), 1)
df$creatinine <- round(pmin(pmax(rgamma(N, shape = 1.6, scale = 0.6), 0.4), 10), 2)
df$hba1c      <- round(pmin(pmax(rnorm(N, 6.8, 1.4), 4), 14), 1)
df$wbc        <- round(pmin(pmax(rgamma(N, shape = 4, scale = 2), 1), 30), 1)
df$hemoglobin <- round(pmin(pmax(rnorm(N, 12.8, 2.1), 5), 18), 1)

2. Histograms with Clinical Reference Ranges

# ── Glucose ───────────────────────────────────────────────────────────────────
p_glu <- ggplot(df, aes(x = glucose)) +
  annotate("rect", xmin = 70, xmax = 100, ymin = -Inf, ymax = Inf,
           alpha = 0.12, fill = "#16A34A") +
  geom_histogram(aes(y = after_stat(density)), bins = 35,
                 fill = "#2563EB", colour = "white", alpha = 0.85) +
  geom_density(colour = "#1e40af", linewidth = 0.8) +
  geom_vline(xintercept = 126, colour = "#DC2626", linetype = "dashed", linewidth = 0.8) +
  geom_vline(xintercept = median(df$glucose), colour = "navy",
             linetype = "dotted", linewidth = 0.8) +
  labs(x = "Glucose (mg/dL)", y = "Density", title = "Glucose",
       subtitle = sprintf("Median (%.0f)  |  Diabetes threshold (126)  |  Normal 70–100",
                          median(df$glucose))) +
  theme_health()

# ── Creatinine ────────────────────────────────────────────────────────────────
p_cr <- ggplot(df, aes(x = creatinine)) +
  annotate("rect", xmin = 0.6, xmax = 1.2, ymin = -Inf, ymax = Inf,
           alpha = 0.12, fill = "#16A34A") +
  geom_histogram(aes(y = after_stat(density)), bins = 35,
                 fill = "#DC2626", colour = "white", alpha = 0.85) +
  geom_density(colour = "#991b1b", linewidth = 0.8) +
  geom_vline(xintercept = 1.5, colour = "darkorange", linetype = "dashed", linewidth = 0.8) +
  geom_vline(xintercept = median(df$creatinine), colour = "darkred",
             linetype = "dotted", linewidth = 0.8) +
  labs(x = "Creatinine (mg/dL)", y = NULL,
       title = sprintf("Creatinine  (skew=%.2f)", skewness(df$creatinine)),
       subtitle = sprintf("Median (%.2f)  |  Mild AKI threshold  |  Normal 0.6–1.2",
                          median(df$creatinine))) +
  theme_health()

# ── HbA1c ─────────────────────────────────────────────────────────────────────
p_a1c <- ggplot(df, aes(x = hba1c)) +
  annotate("rect", xmin = 4,   xmax = 5.6, ymin = -Inf, ymax = Inf,
           alpha = 0.12, fill = "#16A34A") +
  annotate("rect", xmin = 5.7, xmax = 6.4, ymin = -Inf, ymax = Inf,
           alpha = 0.12, fill = "orange") +
  annotate("rect", xmin = 6.4, xmax = 14,  ymin = -Inf, ymax = Inf,
           alpha = 0.08, fill = "#DC2626") +

                 fill = "#16A34A", colour = "white", alpha = 0.85) +

  geom_vline(xintercept = median(df$hba1c), colour = "darkgreen",

       subtitle = sprintf("Median (%.1f%%)  |  Normal / pre-diabetes / diabetes bands",
                          median(df$hba1c))) +
  theme_health()

p_labs <- p_glu + p_cr + p_a1c +

3. Box Plots — LOS by Diagnosis Group

order_drg <- df %>%
  group_by(drg_label) %>%
  summarise(med = median(los_days), .groups = "drop") %>%
  arrange(desc(med)) %>%
  pull(drg_label)

p_los <- df %>%
  mutate(drg_label = factor(drg_label, levels = rev(order_drg))) %>%
  ggplot(aes(x = los_days, y = drg_label, fill = drg_label)) +
  geom_boxplot(outlier.size = 1, outlier.alpha = 0.4, show.legend = FALSE) +
  geom_vline(xintercept = median(df$los_days), colour = "#DC2626",
             linetype = "dashed", linewidth = 0.7) +
  scale_fill_brewer(palette = "Blues", direction = -1) +
  labs(x = "Length of stay (days)", y = NULL, title = "LOS distribution by DRG",
       caption = sprintf("Overall median (%.0f days)", median(df$los_days))) +
  theme_health()

payer_order <- df %>%
  filter(!is.na(total_charge)) %>%
  group_by(payer) %>%
  summarise(med = median(total_charge), .groups = "drop") %>%
  arrange(desc(med)) %>%
  pull(payer)

p_chg <- df %>%
  filter(!is.na(total_charge)) %>%
  mutate(payer = factor(payer, levels = payer_order)) %>%
  ggplot(aes(x = payer, y = total_charge, fill = payer)) +
  geom_boxplot(outlier.size = 1, outlier.alpha = 0.4, show.legend = FALSE) +
  scale_fill_brewer(palette = "Oranges", direction = -1) +
  scale_y_continuous(labels = label_dollar(scale = 1e-3, suffix = "k", accuracy = 1)) +
  labs(x = "Payer", y = "Total charge", title = "Charge distribution by payer") +
  theme_health()

p_box <- p_los + p_chg
print(p_box)
ggsave(file.path(mod02_dir, "boxplots_los_charge.png"), p_box,
       width = 15, height = 5, dpi = 300)

4. Violin Plots — Age by Payer

payer_age_order <- df %>%
  group_by(payer) %>%
  summarise(med = median(age), .groups = "drop") %>%
  arrange(desc(med)) %>%
  pull(payer)

p_age <- df %>%
  mutate(payer = factor(payer, levels = payer_age_order)) %>%
  ggplot(aes(x = payer, y = age, fill = payer)) +
  geom_violin(trim = FALSE, colour = NA, alpha = 0.85, show.legend = FALSE) +
  geom_boxplot(width = 0.12, outlier.shape = NA, fill = "white", alpha = 0.6,
               show.legend = FALSE) +
  geom_hline(yintercept = 65, colour = "#DC2626", linetype = "dashed", linewidth = 0.7) +
  scale_fill_brewer(palette = "Set2") +
  labs(x = "Payer", y = "Age (years)", title = "Age distribution by payer",
       caption = "Medicare eligibility (65)") +
  theme_health()

admit_order <- c("Emergency", "Urgent", "Elective", "Newborn")
p_split <- df %>%
  mutate(admit_type = factor(admit_type, levels = admit_order)) %>%
  ggplot(aes(x = admit_type, y = los_days, fill = sex)) +
  geom_violin(trim = FALSE, colour = NA, alpha = 0.85,
              position = position_dodge(0.8)) +
  geom_boxplot(width = 0.12, outlier.shape = NA, colour = "grey30",
               position = position_dodge(0.8), show.legend = FALSE) +
  scale_fill_manual(values = c(M = "#2563EB", F = "#ED7D31")) +

p_viol <- p_age + p_split

6. Publication-Ready Multi-Panel Figure

# Panel A — Glucose histogram
p_a <- ggplot(df, aes(x = glucose)) +
  annotate("rect", xmin = 70, xmax = 100, ymin = -Inf, ymax = Inf,
           alpha = 0.12, fill = "#16A34A") +
  geom_histogram(aes(y = after_stat(density)), bins = 30,
                 fill = "#2563EB", colour = "white") +
  geom_density(colour = "#1e40af", linewidth = 0.7) +
  geom_vline(xintercept = 126, colour = "#DC2626", linetype = "dashed", linewidth = 0.7) +
  labs(title = "A  Glucose distribution", x = "Glucose (mg/dL)", y = "Density") +
  theme_health() +
  theme(plot.title = element_text(face = "bold"))

# Panel B — LOS by payer (box)
po <- df %>%
  group_by(payer) %>%
  summarise(med = median(los_days), .groups = "drop") %>%
  arrange(desc(med)) %>%
  pull(payer)
p_b <- df %>%
  mutate(payer = factor(payer, levels = po)) %>%
  ggplot(aes(x = payer, y = los_days, fill = payer)) +
  geom_boxplot(outlier.size = 0.7, show.legend = FALSE) +
  scale_fill_brewer(palette = "Blues", direction = -1) +
  labs(title = "B  LOS by payer", x = NULL, y = "LOS (days)") +
  theme_health() +
  theme(plot.title = element_text(face = "bold"),
        axis.text.x = element_text(angle = 20, hjust = 1))

# Panel C — Readmission rate bar
rates <- df %>%
  group_by(payer) %>%
  summarise(rate = mean(readmitted_30) * 100, .groups = "drop") %>%
  arrange(desc(rate)) %>%
  mutate(payer = factor(payer, levels = payer),
         col   = if_else(rate > 15, "#DC2626", "#2563EB"))
p_c <- ggplot(rates, aes(x = payer, y = rate, fill = col)) +
  geom_col(colour = "white", show.legend = FALSE) +
  geom_hline(yintercept = mean(rates$rate), colour = "black", linetype = "dotted") +
  geom_text(aes(label = sprintf("%.1f%%", rate)), vjust = -0.3, size = 3) +
  scale_fill_identity() +
  labs(title = "C  Readmission by payer", x = NULL, y = "30-day readmission rate (%)",
       caption = sprintf("Mean (%.1f%%)", mean(rates$rate))) +
  theme_health() +
  theme(plot.title = element_text(face = "bold"),
        axis.text.x = element_text(angle = 20, hjust = 1))

# Panel D — Age violin by admit type
p_d <- df %>%
  mutate(admit_type = factor(admit_type, levels = c("Emergency", "Urgent", "Elective", "Newborn"))) %>%
  ggplot(aes(x = admit_type, y = age, fill = admit_type)) +
  geom_violin(trim = FALSE, colour = NA, show.legend = FALSE) +
  geom_boxplot(width = 0.12, outlier.shape = NA, fill = "white", alpha = 0.5,
               show.legend = FALSE) +
  geom_hline(yintercept = 65, colour = "#DC2626", linetype = "dashed", linewidth = 0.7) +
  scale_fill_brewer(palette = "Set3") +
  labs(title = "D  Age distribution by admission type", x = NULL, y = "Age (years)",
       caption = "Medicare threshold (65)") +
  theme_health() +
  theme(plot.title = element_text(face = "bold"))

# Panel E — Comorbidity prevalence
comorbs <- c("diabetes", "hypertension", "ckd", "heart_failure")
rates_c <- tibble(
  cond = comorbs,
  prev = map_dbl(comorbs, ~ mean(df[[.x]]) * 100)
) %>%
  mutate(cond = factor(cond, levels = rev(comorbs)))
p_e <- ggplot(rates_c, aes(x = prev, y = cond, fill = cond)) +
  geom_col(colour = "white", show.legend = FALSE) +
  geom_text(aes(label = sprintf("%.1f%%", prev)), hjust = -0.1, size = 3.2) +
  scale_fill_manual(values = c(diabetes = "#16A34A", hypertension = "#2563EB",
                               ckd = "#DC2626", heart_failure = "#B47CC7")) +
  coord_cartesian(xlim = c(0, 65)) +
  labs(title = "E  Comorbidity prevalence", x = "Prevalence (%)", y = NULL) +
  theme_health() +
  theme(plot.title = element_text(face = "bold"))

p_multi <- (p_a | p_b | p_c) / (p_d + p_e + plot_layout(widths = c(2, 1))) +
  plot_annotation(title = "Hospital Discharge Cohort — EDA Summary (N=800)")
print(p_multi)
ggsave(file.path(mod02_dir, "multi_panel_eda.png"), p_multi,
       width = 16, height = 10, dpi = 300)

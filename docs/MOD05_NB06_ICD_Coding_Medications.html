2. TF-IDF ICD-10 Code Suggester

icd_texts <- paste(icd10_ref$kw, icd10_ref$desc)
toks_icd <- quanteda::tokens(icd_texts, remove_punct = TRUE) |>
  quanteda::tokens_tolower() |>
  quanteda::tokens_ngrams(n = 1:2)
dfm_icd <- quanteda::dfm_tfidf(quanteda::dfm(toks_icd))
if (nfeat(dfm_icd) > 500) {
  keep <- names(quanteda::topfeatures(dfm_icd, n = 500))
  dfm_icd <- quanteda::dfm_select(dfm_icd, pattern = keep)
}
icd_matrix <- as.matrix(dfm_icd)

cosine_rows <- function(vec, mat) {
  nrms <- sqrt(rowSums(mat^2)); nrms[nrms == 0] <- 1
  vn <- sqrt(sum(vec^2)); if (vn == 0) vn <- 1
  as.numeric((mat / nrms) %*% (vec / vn))
}

suggest_icd <- function(text, top_k = 5, threshold = 0.12) {
  toks <- quanteda::tokens(tolower(text), remove_punct = TRUE) |>
    quanteda::tokens_ngrams(n = 1:2)
  d <- quanteda::dfm_match(quanteda::dfm(toks), features = colnames(icd_matrix))
  vec <- as.numeric(as.matrix(d))
  scores <- cosine_rows(vec, icd_matrix)
  top_idx <- order(scores, decreasing = TRUE)[seq_len(top_k)]
  keep <- top_idx[scores[top_idx] >= threshold]
  tibble::tibble(
    code  = icd10_ref$code[keep],
    desc  = icd10_ref$desc[keep],
    score = round(scores[keep], 3)
  )
}

drug_list <- drug_list[order(nchar(drug_list), decreasing = TRUE)]

dose_unit <- "(?:mg|mcg|g|mL|units?|IU|mEq)"
route_pat <- "(?:PO|IV|IM|SQ|SC|SL|inhaled|topical|PR)"
freq_pat  <- "(?:QD|BID|TID|QID|QAM|QHS|q\\d+h|PRN|once daily|twice daily|daily|weekly)"
drug_pat  <- paste0("\\b(", paste(stringr::str_replace_all(drug_list, "([\\W])", "\\\\\\1"),
                                  collapse = "|"), ")\\b")
med_pat <- paste0(
  drug_pat,
  "(?:[\\s,]+(\\d+(?:\\.\\d+)?)\\s*(", dose_unit, "))?",
  "(?:[\\s,]+(", route_pat, "))?",
  "(?:[\\s,]+(", freq_pat, "))?"
)

extract_meds <- function(text) {
  m <- stringr::str_match_all(text, stringr::regex(med_pat, ignore_case = TRUE))[[1]]
  if (nrow(m) == 0) {
    return(tibble::tibble(drug = character(), dose = character(), unit = character(),
                          route = character(), freq = character()))
  }
  out <- tibble::tibble(
    drug  = tolower(m[, 2]),
    dose  = ifelse(is.na(m[, 3]), "", stringr::str_squish(m[, 3])),
    unit  = ifelse(is.na(m[, 4]), "", stringr::str_squish(m[, 4])),
    route = toupper(ifelse(is.na(m[, 5]), "", m[, 5])),
    freq  = toupper(ifelse(is.na(m[, 6]), "", m[, 6]))
  )
  dplyr::distinct(out, drug, .keep_all = TRUE)
}

4. Medication Reconciliation

reconcile_medications <- function(admit_text, dc_text) {
  admit_meds <- extract_meds(admit_text)
  dc_meds    <- extract_meds(dc_text)
  admit_set  <- admit_meds$drug
  dc_set     <- dc_meds$drug
  continued    <- sort(intersect(admit_set, dc_set))
  discontinued <- sort(setdiff(admit_set, dc_set))
  new_at_dc    <- sort(setdiff(dc_set, admit_set))
  dose_changed <- continued[purrr::map_lgl(continued, function(d) {
    a <- admit_meds$dose[admit_meds$drug == d]
    b <- dc_meds$dose[dc_meds$drug == d]
    nzchar(a) && nzchar(b) && a != b
  })]
  list(
    continued = continued, discontinued = discontinued,
    new_at_dc = new_at_dc, dose_changed = dose_changed,
    admit_meds = admit_meds, dc_meds = dc_meds
  )
}

6. ICD-10 Code Validation

validate_icd <- function(code) {
  code <- toupper(stringr::str_squish(as.character(code)))
  valid <- stringr::str_detect(code, "^[A-Z]\\d{2}(\\.\\d{0,4})?$")
  match <- icd10_ref[icd10_ref$code == code, ]
  list(
    code = code,
    valid_format = valid,
    in_reference = nrow(match) > 0,
    desc = if (nrow(match) > 0) match$desc[1] else "Not in reference"
  )
}

7. Medication Completeness Audit

completeness_audit <- function(text) {
  meds <- extract_meds(text)
  complete   <- meds[nzchar(meds$freq) | nzchar(meds$route), ]
  incomplete <- meds[!nzchar(meds$freq) & !nzchar(meds$route), ]
  cat(sprintf("Total medications : %d\n", nrow(meds)))
  cat(sprintf("Complete records  : %d\n", nrow(complete)))
  cat(sprintf("Incomplete records: %d\n", nrow(incomplete)))
  if (nrow(incomplete) > 0) {
    cat("\nMedications missing route AND frequency:\n")
    purrr::pwalk(incomplete, function(drug, dose, unit, ...) {
      cat(sprintf("  %-25s dose=%s %s -- INCOMPLETE\n", drug, dose, unit))
    })
  }
  incomplete
}

# Exercise 5 — dose range validation
dose_limits_oral <- list(
  metformin    = c(250, 2550),
  furosemide   = c(20, 600),
  lisinopril   = c(2.5, 40),
  warfarin     = c(1, 15),
  aspirin      = c(81, 325),
  atorvastatin = c(10, 80)
)

validate_doses <- function(text) {
  meds <- extract_meds(text)
  flags <- list()
  for (i in seq_len(nrow(meds))) {
    m <- meds[i, ]
    if (m$drug %in% names(dose_limits_oral) && nzchar(m$dose) &&
        m$route %in% c("PO", "")) {
      dose <- suppressWarnings(as.numeric(m$dose))
      if (is.na(dose)) next
      lim <- dose_limits_oral[[m$drug]]
      if (!(lim[1] <= dose && dose <= lim[2])) {
        flags[[length(flags) + 1]] <- tibble::tibble(
          drug = m$drug, dose = dose, unit = m$unit,
          expected = sprintf("%s-%s mg", lim[1], lim[2]),
          flag = if (dose > lim[2]) "ABOVE MAX" else "BELOW MIN"
        )
      }
    }
  }
  if (length(flags) == 0) {
    return(tibble::tibble(drug = character(), dose = numeric(), unit = character(),
                          expected = character(), flag = character()))
  }
  dplyr::bind_rows(flags)
}

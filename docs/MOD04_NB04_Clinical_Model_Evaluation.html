2. AUC-ROC and Precision-Recall with Bootstrap CIs

bootstrap_ci <- function(y_true, y_prob, metric_fn, n_boot = 1000, seed = 42) {
  set.seed(seed)
  n <- length(y_true)
  scores <- replicate(n_boot, {
    idx <- sample.int(n, n, replace = TRUE)
    metric_fn(y_true[idx], y_prob[idx])
  })
  quantile(scores, c(0.025, 0.975))
}

summary_rows <- list()
roc_all <- list(); pr_all <- list()
for (name in names(probs)) {
  prob <- probs[[name]]
  auc  <- auc_prob(y_te, prob)
  ap   <- ap_prob(y_te, prob)
  brier <- brier_prob(y_te, prob)
  auc_ci <- bootstrap_ci(y_te, prob, auc_prob)
  ap_ci  <- bootstrap_ci(y_te, prob, ap_prob)
  summary_rows[[name]] <- tibble(
    Model = name, AUC = auc,
    `AUC 95%CI` = sprintf("%.3f–%.3f", auc_ci[1], auc_ci[2]),
    `Avg Prec` = ap,
    `AP 95%CI` = sprintf("%.3f–%.3f", ap_ci[1], ap_ci[2]),
    Brier = brier
  )
  roc_all[[name]] <- yardstick::roc_curve(
    tibble(truth = factor(y_te, levels = c(0, 1)), .pred = prob),
    truth, .pred, event_level = "second"

3. Calibration Assessment & Recalibration

calib_bins <- function(y, p, n_bins = 10) {
  tibble(y = y, p = p) %>%
    mutate(bin = ntile(p, n_bins)) %>%
    group_by(bin) %>%
    summarise(mean_pred = mean(p), frac_pos = mean(y), .groups = "drop")
}

platt_calibrate <- function(p_fit, y_fit, p_new) {
  cal <- glm(y ~ p, data = tibble(y = y_fit, p = p_fit), family = binomial)
  as.numeric(predict(cal, newdata = tibble(p = p_new), type = "response"))
}
iso_calibrate <- function(p_fit, y_fit, p_new) {
  ord <- order(p_fit)
  iso <- isoreg(p_fit[ord], y_fit[ord])
  sf  <- as.stepfun(iso)
  pmin(pmax(sf(p_new), 0), 1)
}

calib_plots <- imap(probs, function(prob, name) {
  orig   <- calib_bins(y_te, prob)
  plat_p <- platt_calibrate(prob, y_te, prob)
  iso_p  <- iso_calibrate(prob, y_te, prob)
  d <- bind_rows(
    orig %>% mutate(method = "Uncalibrated"),
    calib_bins(y_te, plat_p) %>% mutate(method = "Platt (sigmoid)"),
    calib_bins(y_te, iso_p)  %>% mutate(method = "Isotonic")
  )
  brier_orig  <- brier_prob(y_te, prob)
  brier_platt <- brier_prob(y_te, plat_p)
  ggplot(d, aes(mean_pred, frac_pos, colour = method, shape = method)) +
    geom_abline(slope = 1, intercept = 0, linetype = "dashed", colour = "grey30") +
    geom_point(size = 2) + geom_line() +
    scale_colour_manual(values = c("Uncalibrated" = colors_m[[name]],
                                   "Platt (sigmoid)" = "#D97706",
                                   "Isotonic" = "#7C3AED")) +
    annotate("text", x = 0.05, y = 0.95,
             label = sprintf("Brier: %.3f→%.3f", brier_orig, brier_platt),
             hjust = 0, size = 3) +
    labs(x = "Mean predicted", y = "Fraction positives", title = name, colour = NULL, shape = NULL) +
    coord_equal(xlim = c(0, 1), ylim = c(0, 1)) +
    theme(legend.position = "bottom")
})

4. Decision Curve Analysis (DCA) — Net Benefit

net_benefit <- function(y_true, y_prob, threshold) {
  y_pred <- as.integer(y_prob >= threshold)
  tp <- sum(y_pred == 1 & y_true == 1)
  fp <- sum(y_pred == 1 & y_true == 0)
  n  <- length(y_true)
  odds <- threshold / (1 - threshold)
  (tp - fp * odds) / n
}

thresholds <- seq(0.01, 0.50, length.out = 100)
dca <- imap_dfr(probs, function(prob, name) {
  tibble(threshold = thresholds,
         nb = map_dbl(thresholds, ~ net_benefit(y_te, prob, .x)),
         Model = name)
})
nb_all <- tibble(

5. Threshold Selection for Clinical Deployment

best_prob <- probs[["Grad Boosting"]]
roc_df <- pROC::roc(y_te, best_prob, quiet = TRUE)
fpr <- 1 - roc_df$specificities

# pROC thresholds are in decreasing score order; replace Inf/-Inf
thrs[is.infinite(thrs)] <- ifelse(thrs[is.infinite(thrs)] > 0, 1, 0)

pr_df <- yardstick::pr_curve(
  tibble(truth = factor(y_te, levels = c(0, 1)), .pred = best_prob),
  truth, .pred, event_level = "second"
)
precision_v <- pr_df$precision
recall_v    <- pr_df$recall
thrs_pr     <- pr_df$.threshold
thrs_pr[is.infinite(thrs_pr)] <- 1

# Three threshold strategies

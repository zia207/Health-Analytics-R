Module 03 — Relative Risk, Odds Ratios & Confidence Intervals

Relative Risk, Odds Ratios & Confidence Intervals

Learning objectives - Calculate RR, OR, ARR, NNT from 2×2 contingency tables - Construct bootstrap and asymptotic 95% CIs for all four measures - Build and interpret forest plots for multiple exposure-outcome pairs - Understand when OR approximates RR (and when it does not) - Apply the Mantel-Haenszel method for stratified analysis

2. Core Measures from a 2×2 Table

measures_2x2 <- function(exposed, outcome, exposure_label = "Exposed", ci_method = "asymptotic") {
  # Compute RR, OR, ARR, and NNT with 95% CI from binary exposure and outcome.
  # Returns a tibble suitable for forest plots.
  a <- as.integer(sum(exposed == 1 & outcome == 1))  # exposed + event
  b <- as.integer(sum(exposed == 1 & outcome == 0))  # exposed + no event
  c <- as.integer(sum(exposed == 0 & outcome == 1))  # unexposed + event
  d <- as.integer(sum(exposed == 0 & outcome == 0))  # unexposed + no event
  n_e <- a + b; n_u <- c + d
  risk_e <- a / n_e; risk_u <- c / n_u

  # Point estimates
  ARR <- risk_e - risk_u          # absolute risk reduction (negative = harm)
  NNT <- if (ARR != 0) 1 / abs(ARR) else Inf
  RR  <- risk_e / risk_u
  OR  <- if (b * c > 0) (a * d) / (b * c) else NA_real_

  # 95% CI (delta method / log)
  z <- qnorm(0.975)
  se_log_rr <- sqrt((b / (a * n_e)) + (d / (c * n_u)))
  rr_lo <- exp(log(RR) - z * se_log_rr)
  rr_hi <- exp(log(RR) + z * se_log_rr)
  se_log_or <- sqrt(1 / a + 1 / b + 1 / c + 1 / d)
  or_lo <- exp(log(OR) - z * se_log_or)
  or_hi <- exp(log(OR) + z * se_log_or)
  se_arr <- sqrt(risk_e * (1 - risk_e) / n_e + risk_u * (1 - risk_u) / n_u)
  arr_lo <- ARR - z * se_arr
  arr_hi <- ARR + z * se_arr

  p_val <- chisq.test(matrix(c(a, b, c, d), nrow = 2, byrow = TRUE))$p.value

  tibble(
    Exposure     = exposure_label,
    N_exposed    = n_e,
    N_unexposed  = n_u,
    Risk_exposed = round(risk_e * 100, 1),
    Risk_unexpos = round(risk_u * 100, 1),
    ARR_pct      = round(ARR * 100, 2),
    NNT          = round(NNT, 1),
    RR           = round(RR, 2),
    RR_lo        = round(rr_lo, 2),
    RR_hi        = round(rr_hi, 2),
    OR           = round(OR, 2),
    OR_lo        = round(or_lo, 2),
    OR_hi        = round(or_hi, 2),
    p_value      = round(p_val, 4)
  )
}

3. Bootstrap Confidence Intervals

bootstrap_rr_or <- function(exposed, outcome, n_boot = 2000, seed = 42) {
  # Bootstrap 95% CI for RR and OR — distribution-free.
  set.seed(seed)
  n <- length(exposed)
  rrs <- numeric(0); ors <- numeric(0)
  for (i in seq_len(n_boot)) {
    boot_idx <- sample.int(n, n, replace = TRUE)
    e <- exposed[boot_idx]
    o <- outcome[boot_idx]
    a <- sum(e == 1 & o == 1)
    b <- sum(e == 1 & o == 0)
    c <- sum(e == 0 & o == 1)
    d <- sum(e == 0 & o == 0)
    n_e <- a + b; n_u <- c + d
    if (n_e > 0 && n_u > 0 && a > 0 && c > 0 && b > 0 && d > 0) {
      rrs <- c(rrs, (a / n_e) / (c / n_u))
      ors <- c(ors, (a * d) / (b * c))
    }
  }
  rr_ci <- quantile(rrs, c(0.025, 0.975))
  or_ci <- quantile(ors, c(0.025, 0.975))
  list(rr_ci = rr_ci, or_ci = or_ci, rrs = rrs, ors = ors)
}

6. Mantel-Haenszel Stratified Analysis

# Stratified analysis: control for age confounding in diabetes → readmission
age_strata <- c("18-44", "45-64", "65-74", "75+")

# Stratum-specific 2×2 tables
stratum_results <- lapply(age_strata, function(stratum) {
  sub <- df |> filter(age_group == stratum)
  a <- sum(sub$diabetes == 1 & sub$readmitted_30 == 1)
  b <- sum(sub$diabetes == 1 & sub$readmitted_30 == 0)
  c <- sum(sub$diabetes == 0 & sub$readmitted_30 == 1)
  d <- sum(sub$diabetes == 0 & sub$readmitted_30 == 0)
  n <- a + b + c + d
  or_s <- if (b * c > 0) (a * d) / (b * c) else NA_real_

  tibble(Stratum = stratum, a = a, b = b, c = c, d = d, n = n,
         OR_stratum = round(or_s, 2), RR_stratum = round(rr_s, 2))
})

7. Knowledge Check

  1. You observe OR=3.1 for smoking → lung cancer in a case-control study (case prevalence 45%). Should you report RR? Why or why not?
  2. ARR = -2.3% for a treatment vs control. What is the NNT, and how do you interpret it?
  3. In the forest plot, CKD has a wide CI. What does this tell you?
  4. When would you prefer bootstrap CI over the asymptotic log method?
  5. Diabetes OR = 1.8 (crude). After stratifying by age the MH OR = 2.2. What does this imply?
# Exercise 5 — simulate confounding and test MH vs crude
set.seed(77)
n_ex <- 2000
age_ex  <- rbinom(n_ex, 1, 0.5)            # older=1
# Older → more diabetes AND more readmission (positive confounding)

read_ex <- rbinom(n_ex, 1, logistic(-2.5 + 0.4 * diab_ex + 1.0 * age_ex))

crude <- measures_2x2(diab_ex, read_ex, "Crude")
for (pair in list(list("Young (age=0)", age_ex == 0), list("Old (age=1)", age_ex == 1))) {
  r <- measures_2x2(diab_ex[pair[[2]]], read_ex[pair[[2]]], pair[[1]])
  cat(sprintf("%-20s: OR=%s\n", pair[[1]], r$OR))
}

cat(sprintf("Crude OR: %s\n", crude$OR))

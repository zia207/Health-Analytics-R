Descriptive Statistics & Health Metrics

Learning objectives - Compute prevalence and incidence rates with proper denominators - Produce stratified summaries by age group, sex, and payer - Interpret skewed distributions for LOS and hospital charges - Apply age-standardisation (direct method) to compare populations - Build a publication-ready descriptive “Table 1”

# --- 2.3  95% confidence interval for a proportion (Wilson method) ---
wilson_ci <- function(k, n, alpha = 0.05) {
  # Wilson score interval — preferred for proportions near 0 or 1.
  z         <- qnorm(1 - alpha / 2)
  p_hat     <- k / n
  denom     <- 1 + z^2 / n
  centre    <- (p_hat + z^2 / (2 * n)) / denom
  half_width <- (z * sqrt(p_hat * (1 - p_hat) / n + z^2 / (4 * n^2))) / denom
  c(
    lo = round((centre - half_width) * 100, 1),
    hi = round((centre + half_width) * 100, 1)
  )
}

ci <- wilson_ci(readmissions, total_discharges)
cat(sprintf("Readmission rate: %.1f%% (95%% CI: %s%% – %s%%)\n",
            rate_per_100, ci["lo"], ci["hi"]))
Readmission rate: 13.8% (95% CI: 11.1% – 17.1%)

4. Interpreting Skewed Clinical Distributions

# --- 4.1  LOS skewness and kurtosis ---
skewness <- function(x) {
  x <- x[!is.na(x)]
  m <- mean(x)
  s <- sd(x)
  mean((x - m)^3) / (s^3)
}
kurtosis_ex <- function(x) {
  x <- x[!is.na(x)]
  m <- mean(x)
  s <- sd(x)
  mean((x - m)^4) / (s^4) - 3
}

los <- df$los_days[!is.na(df$los_days)]
chg <- df$total_charge[!is.na(df$total_charge)]

for (pair in list(list("LOS (days)", los), list("Total charge ($)", chg))) {

              if (sk > 0.5) "(right-skewed — median is preferred)" else ""))
  cat(sprintf("  Kurtosis: %.3f\n\n", kurtosis_ex(series)))
}

5. Age-Standardisation (Direct Method)

# Direct age-standardisation: applies study rates to a standard population
# to remove confounding by age structure differences between groups.

# US 2000 Standard Population weights by age group
std_pop <- tibble(
  age_group  = factor(c("18-44", "45-64", "65-74", "75+"),
                      levels = c("18-44", "45-64", "65-74", "75+")),
  std_weight = c(0.3761, 0.3399, 0.1267, 0.0573)   # proportions sum to ~1
)

# Crude rates by age group × payer (Medicare vs Private as example)
for (pay in c("Medicare", "Private")) {
  sub <- df %>% filter(payer == pay)
  rates <- sub %>%
    group_by(age_group) %>%
    summarise(
      crude_rate = mean(readmitted_30),
      n          = n(),
      .groups    = "drop"
    ) %>%
    left_join(std_pop, by = "age_group") %>%
    mutate(weighted_rate = crude_rate * std_weight)
  age_adj_rate  <- sum(rates$weighted_rate, na.rm = TRUE) * 100
  crude_overall <- mean(sub$readmitted_30) * 100
  cat(sprintf("%-10s — Crude: %.1f%%  |  Age-adjusted: %.1f%%\n",
              pay, crude_overall, age_adj_rate))
}

6. Table 1 — Publication-Ready Descriptive Table

A “Table 1” in clinical research summarises the baseline characteristics of a cohort, often stratified by a key variable (e.g. readmitted vs not).

table1 <- function(df, stratify_col, continuous_cols, categorical_cols) {
  # Build a basic Table 1.
  # Continuous: mean ± SD (or median [IQR] if skewed)
  # Categorical: N (%)
  groups <- sort(unique(df[[stratify_col]]))
  group_label <- function(g) {
    n <- sum(df[[stratify_col]] == g, na.rm = TRUE)
    pretty <- if (identical(g, 0) || identical(g, 0L)) {
      "Not readmitted"
    } else if (identical(g, 1) || identical(g, 1L)) {
      "Readmitted"
    } else {
      as.character(g)
    }
    sprintf("%s (N=%d)", pretty, n)
  }

  rows <- list()
  header <- c(Variable = "Variable")
  for (g in groups) header[[as.character(g)]] <- group_label(g)
  rows[[length(rows) + 1]] <- header

  for (col in continuous_cols) {
    sk  <- abs(skewness(df[[col]]))
    row <- c(Variable = paste0(col, if (sk > 1) " [median, IQR]" else " [mean ± SD]"))
    for (g in groups) {
      vals <- df[[col]][df[[stratify_col]] == g]
      vals <- vals[!is.na(vals)]
      row[[as.character(g)]] <- if (sk > 1) {
        sprintf("%.1f [%.1f–%.1f]", median(vals), quantile(vals, 0.25), quantile(vals, 0.75))
      } else {
        sprintf("%.1f ± %.1f", mean(vals), sd(vals))
      }
    }
    rows[[length(rows) + 1]] <- row
  }

  for (col in categorical_cols) {
    row <- c(Variable = col)
    for (g in groups) row[[as.character(g)]] <- ""
    rows[[length(rows) + 1]] <- row
    cats <- unique(df[[col]][!is.na(df[[col]])])
    for (cat in cats) {
      row <- c(Variable = paste0("  ", as.character(cat)))
      for (g in groups) {
        sub   <- df[df[[stratify_col]] == g, ]
        n_cat <- sum(sub[[col]] == cat, na.rm = TRUE)
        pct   <- n_cat / nrow(sub) * 100
        row[[as.character(g)]] <- sprintf("%d (%.1f%%)", n_cat, pct)
      }
      rows[[length(rows) + 1]] <- row
    }
  }

  bind_rows(lapply(rows, function(r) as_tibble(as.list(r))))
}

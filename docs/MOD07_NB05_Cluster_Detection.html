spatial_smooth <- function(values, row_idx, col_idx, nrow, ncol, sigma = 2) {
  grid <- matrix(0, nrow = nrow, ncol = ncol)
  for (i in seq_along(values)) grid[row_idx[i], col_idx[i]] <- values[i]
  k <- ceiling(3 * sigma)
  xs <- (-k):k
  kern <- outer(dnorm(xs, sd = sigma), dnorm(xs, sd = sigma))
  kern <- kern / sum(kern)
  pad <- k
  gp <- matrix(0, nrow + 2 * pad, ncol + 2 * pad)
  gp[(pad + 1):(pad + nrow), (pad + 1):(pad + ncol)] <- grid
  out <- matrix(0, nrow, ncol)
  for (r in seq_len(nrow)) for (c in seq_len(ncol)) {
    out[r, c] <- sum(gp[r:(r + 2 * k), c:(c + 2 * k)] * kern)
  }
  mapply(function(r, c) out[r, c], row_idx, col_idx)
}

2. Kulldorff Spatial Scan Statistic

The spatial scan statistic tests whether the number of cases within a circular window Z is higher than expected:

LLR(Z) = log[ (O_Z/E_Z)^O_Z * ((O-O_Z)/(E-E_Z))^(O-O_Z) ]

Where: - O_Z, E_Z = observed and expected cases inside window Z - O, E = total observed and expected - The window that maximises LLR is the most likely cluster - Significance is assessed via Monte Carlo permutation (randomise cases under H₀)

This is equivalent to the Poisson spatial scan statistic implemented in SaTScan software.

scan_statistic_poisson <- function(df_in, max_pop_fraction = 0.5, n_perm = 199, seed = 42) {
  coords  <- cbind(df_in$lon, df_in$lat)
  O_total <- sum(df_in$observed)
  E_total <- sum(df_in$expected)
  n       <- nrow(df_in)
  max_pop <- sum(df_in$population) * max_pop_fraction
  dmat    <- as.matrix(dist(coords))

  llr_poisson <- function(O_z, E_z) {
    if (O_z == 0 || E_z == 0 || O_z <= E_z) return(0)
    O_out <- O_total - O_z
    E_out <- E_total - E_z
    if (O_out <= 0 || E_out <= 0 || O_out <= E_out) return(0)
    max(0, O_z * log(O_z / E_z) + O_out * log(O_out / E_out))
  }

  best_llr <- 0; best_center <- 1; best_radius <- 0; best_inside <- integer(0)

  for (ci in seq_len(n)) {
    ord <- order(dmat[ci, ])
    O_z <- 0; E_z <- 0; pop_z <- 0
    for (k in 2:n) {
      j <- ord[k]
      pop_z <- pop_z + df_in$population[j]
      if (pop_z > max_pop) break
      O_z <- O_z + df_in$observed[j]
      E_z <- E_z + df_in$expected[j]
      llr <- llr_poisson(O_z, E_z)
      if (llr > best_llr) {
        best_llr <- llr
        best_center <- ci
        best_radius <- dmat[ci, j]
        best_inside <- ord[2:k]
      }
    }
  }

  set.seed(seed)
  mc_llrs <- numeric(n_perm)
  E_arr <- df_in$expected
  p_mult <- E_arr / E_total
  for (b in seq_len(n_perm)) {
    O_perm <- as.numeric(rmultinom(1, O_total, p_mult))
    best_b <- 0
    for (ci in seq_len(n)) {
      ord <- order(dmat[ci, ])
      Oz <- 0; Ez <- 0; popz <- 0
      for (k in 2:n) {
        j <- ord[k]
        popz <- popz + df_in$population[j]
        if (popz > max_pop) break
        Oz <- Oz + O_perm[j]
        Ez <- Ez + E_arr[j]
        l <- llr_poisson(Oz, Ez)
        if (l > best_b) best_b <- l
      }
    }
    mc_llrs[b] <- best_b
  }

  list(
    best_llr = best_llr, best_center = best_center,
    best_radius = best_radius, best_inside = best_inside,
    p_value = mean(mc_llrs >= best_llr), mc_llrs = mc_llrs
  )
}

llrs_by_size <- sapply(cluster_sizes, function(max_in) {

  if (O_z > 0 && E_z > 0 && O_z > E_z && O_out > 0 && E_out > 0 && O_out > E_out) {

nb <- vector("list", N)
for (i in seq_len(N)) {
  nbrs <- which(pmax(abs(df$row - df$row[i]), abs(df$col - df$col[i])) <= 1)
  nb[[i]] <- setdiff(nbrs, i)

class(nb) <- "nb"
attr(nb, "region.id") <- as.character(seq_len(N))

Learning objectives - Understand spatial autocorrelation: Tobler’s First Law of Geography - Build spatial weights matrices (queen contiguity, rook, distance-based, k-NN) - Compute Global Moran’s I and its significance via permutation - Calculate Local Moran’s I (LISA) to identify spatial clusters - Produce cluster maps: High-High, Low-Low, High-Low, Low-High quadrants - Interpret Moran scatterplots for health outcomes

spatial_smooth <- function(values, row_idx, col_idx, nrow, ncol, sigma = 2) {
  grid <- matrix(0, nrow = nrow, ncol = ncol)
  for (i in seq_along(values)) grid[row_idx[i], col_idx[i]] <- values[i]
  k <- ceiling(3 * sigma)
  xs <- (-k):k
  kern <- outer(dnorm(xs, sd = sigma), dnorm(xs, sd = sigma))
  kern <- kern / sum(kern)
  pad <- k
  gp <- matrix(0, nrow + 2 * pad, ncol + 2 * pad)
  gp[(pad + 1):(pad + nrow), (pad + 1):(pad + ncol)] <- grid
  out <- matrix(0, nrow, ncol)
  for (r in seq_len(nrow)) {
    for (c in seq_len(ncol)) {
      out[r, c] <- sum(gp[r:(r + 2 * k), c:(c + 2 * k)] * kern)
    }
  }
  mapply(function(r, c) out[r, c], row_idx, col_idx)
}

2. Spatial Weights Matrix

A spatial weights matrix W encodes the neighbourhood structure: - Queen contiguity: shares any boundary point or corner - Rook contiguity: shares an edge only - Distance-based: w_ij = 1 if distance < threshold - k-NN: each unit connected to its k nearest neighbours - Inverse distance: w_ij = 1/d_ij (continuous)

Row-standardisation: W* where each row sums to 1 (so W*x = spatial lag of x)

# Queen / rook from the regular grid indices
build_grid_nb <- function(row_idx, col_idx, type = c("queen", "rook")) {
  type <- match.arg(type)
  n <- length(row_idx)
  nb <- vector("list", n)
  for (i in seq_len(n)) {
    dr <- abs(row_idx - row_idx[i])
    dc <- abs(col_idx - col_idx[i])
    if (type == "queen") {
      nbrs <- which(dr <= 1 & dc <= 1 & !(dr == 0 & dc == 0))
    } else {
      nbrs <- which(((dr == 1 & dc == 0) | (dr == 0 & dc == 1)))
    }
    nb[[i]] <- if (length(nbrs)) as.integer(nbrs) else 0L
  }
  class(nb) <- "nb"
  attr(nb, "region.id") <- as.character(seq_len(n))
  attr(nb, "type") <- type
  nb
}

3. Global Moran’s I

print_moran <- function(x, lw, label) {
  mt <- moran.test(x, lw, zero.policy = TRUE, alternative = "two.sided")
  mp <- moran.mc(x, lw, nsim = 999, zero.policy = TRUE, alternative = "two.sided")
  tibble(
    variable = label,
    I        = unname(mt$estimate["Moran I statistic"]),
    E_I      = unname(mt$estimate["Expectation"]),
    z_score  = unname(mt$statistic),
    p_value  = mp$p.value
  )
}

5. Getis-Ord G* Hot Spot Analysis

# G* includes self in the neighbourhood
lw_gstar <- nb2listw(include.self(nb_queen), style = "B", zero.policy = TRUE)
g_star <- localG(df$cvd_rate, lw_gstar, zero.policy = TRUE)
gdf$g_star   <- as.numeric(g_star)
gdf$g_star_p <- 2 * pnorm(-abs(gdf$g_star))

gdf$hotspot_class <- case_when(
  gdf$g_star >  2.58 & gdf$g_star_p < 0.01 ~ "Hot spot 99%",
  gdf$g_star >  1.96 & gdf$g_star_p < 0.05 ~ "Hot spot 95%",
  gdf$g_star < -2.58 & gdf$g_star_p < 0.01 ~ "Cold spot 99%",
  gdf$g_star < -1.96 & gdf$g_star_p < 0.05 ~ "Cold spot 95%",

Health Analytics with R · Module 03: Statistical Inference

Learning objectives - Fit and interpret logistic regression for binary outcomes (readmission, mortality) - Report adjusted ORs with 95% CIs from model output - Assess model fit: Hosmer-Lemeshow test, calibration plot, ROC-AUC - Detect and handle multicollinearity (VIF) - Build a multivariable risk score with predicted probabilities

Estimated time: 2.5 hours | Level: Intermediate → Advanced | Libraries: stats, broom, ggplot2, pROC, dplyr

                        prob = c(0.52, 0.30, 0.18))
n_comorb      <- diabetes + heart_failure + copd + hypertension

readmit_logit <- (-3.0 + 0.7 * heart_failure + 0.5 * diabetes + 0.4 * copd
                  + 0.02 * (ages - 60) + 0.3 * as.integer(admit_type == "Emergency")
                  + 0.2 * as.integer(los_days > 7) + 0.3 * as.integer(payers == "Medicaid")
                  + rnorm(N, 0, 0.3))
readmitted_30 <- rbinom(N, 1, logistic(readmit_logit))

4. Multicollinearity — Variance Inflation Factor

# VIF for continuous and binary predictors
vif_from_lm <- function(data, vars) {
  X <- data[, vars, drop = FALSE]
  vapply(vars, function(v) {
    others <- setdiff(vars, v)
    r2 <- summary(lm(as.formula(paste(v, "~", paste(others, collapse = "+"))),
                     data = X))$r.squared
    1 / (1 - r2)
  }, numeric(1))
}

vif_vals <- vif_from_lm(df, c("age", "diabetes", "heart_failure", "copd", "hypertension", "los_gt7"))

6. Hosmer-Lemeshow Goodness-of-Fit

hosmer_lemeshow <- function(y_true, y_pred, n_groups = 10) {
  # Hosmer-Lemeshow test for logistic regression calibration.
  df_hl <- tibble(y = y_true, p = y_pred) |>
    mutate(decile = ntile(p, n_groups))
  grouped <- df_hl |>
    group_by(decile) |>
    summarise(
      obs_pos = sum(y),
      obs_neg = sum(y == 0),
      exp_pos = sum(p),
      exp_neg = sum(1 - p),
      .groups = "drop"
    )
  hl_stat <- sum((grouped$obs_pos - grouped$exp_pos)^2 / grouped$exp_pos +
                   (grouped$obs_neg - grouped$exp_neg)^2 / grouped$exp_neg)
  p_val <- 1 - pchisq(hl_stat, df = n_groups - 2)
  list(stat = hl_stat, p = p_val)
}

hl <- hosmer_lemeshow(df$readmitted_30, df$pred_prob)

Libraries: sf, spdep, spatialreg, ggplot2, MASS

spatial_smooth <- function(values, row_idx, col_idx, nrow, ncol, sigma = 2) {
  grid <- matrix(0, nrow = nrow, ncol = ncol)
  for (i in seq_along(values)) grid[row_idx[i], col_idx[i]] <- values[i]
  k <- ceiling(3 * sigma)
  xs <- (-k):k
  kern <- outer(dnorm(xs, sd = sigma), dnorm(xs, sd = sigma))
  kern <- kern / sum(kern)
  pad <- k
  gp <- matrix(0, nrow + 2 * pad, ncol + 2 * pad)
  gp[(pad + 1):(pad + nrow), (pad + 1):(pad + ncol)] <- grid
  out <- matrix(0, nrow, ncol)
  for (r in seq_len(nrow)) for (c in seq_len(ncol)) {
    out[r, c] <- sum(gp[r:(r + 2 * k), c:(c + 2 * k)] * kern)
  }
  mapply(function(r, c) out[r, c], row_idx, col_idx)
}

2. The Small-Area Estimation Problem

Problem: Observed SMR = O_i / E_i is unstable for areas with small populations.

A county with expected deaths E=2 and observed O=4 has SMR=2.0, suggesting 100% excess mortality — but this could easily be random variation.

Key insight: Raw rates have high variance in low-population areas. Bayesian methods “borrow strength” from neighbouring areas and the overall mean.

Method Shrinkage target Spatially informed?
Raw SMR None No
Empirical Bayes (EB) Overall mean No
BYM/CAR Spatial neighbours Yes
Full Bayesian MCMC Prior + data Yes (BYM)

3. Empirical Bayes Smoothing

# Gamma-Poisson (conjugate) Empirical Bayes
# Model: O_i | theta_i ~ Poisson(E_i * theta_i)
#        theta_i ~ Gamma(alpha, beta)
# Posterior mean: E[theta_i|O_i] = (O_i + alpha) / (E_i + beta)
# Shrinkage toward overall rate: alpha/beta

smr_mean <- mean(df$smr)
smr_var  <- var(df$smr)
extra_var <- max(smr_var - smr_mean / mean(df$expected), 0.01)
alpha_hat <- smr_mean^2 / extra_var
beta_hat  <- smr_mean / extra_var

cat(sprintf("EB Hyperparameters: alpha=%.3f, beta=%.3f\n", alpha_hat, beta_hat))
EB Hyperparameters: alpha=100.937, beta=100.468
cat(sprintf("Prior mean (overall rate): %.3f\n", alpha_hat / beta_hat))
Prior mean (overall rate): 1.005
gdf$smr_eb <- (df$observed + alpha_hat) / (df$expected + beta_hat)
gdf$smr_shrinkage <- 1 - (df$expected / (df$expected + beta_hat))
df$smr_eb <- gdf$smr_eb
df$smr_shrinkage <- gdf$smr_shrinkage

4. CAR/BYM Model — Conceptual Framework

The Besag-York-Mollié (BYM) model decomposes the random effect into:

Y_i | theta_i ~ Poisson(E_i * theta_i)
log(theta_i) = mu + u_i + v_i

Where: - u_i = spatially structured random effect (ICAR prior — neighbours pull toward each other) - v_i = unstructured random effect (i.i.d. normal — pure noise) - mu = overall log-relative risk

ICAR prior for u_i:

u_i | u_{-i} ~ Normal(mean(u_j for j in neighbours_i), sigma²_u / n_i)

This is equivalent to a joint Gaussian distribution with precision matrix Q = D - W
where D is the diagonal matrix of neighbour counts and W is the adjacency matrix.

# Spatial empirical Bayes (Marshall) via spdep::EBlocal
nb <- vector("list", N)
W_bin <- matrix(0, N, N)
for (i in seq_len(N)) {
  nbrs <- which(pmax(abs(df$row - df$row[i]), abs(df$col - df$col[i])) <= 1)
  nbrs <- setdiff(nbrs, i)
  nb[[i]] <- if (length(nbrs)) nbrs else 0L
  W_bin[i, nbrs] <- 1
}
class(nb) <- "nb"
attr(nb, "region.id") <- as.character(seq_len(N))
lw <- nb2listw(nb, style = "W", zero.policy = TRUE)

eb_loc <- EBlocal(df$observed, df$expected, nb)
gdf$smr_eb_local <- eb_loc$est

# Approximate BYM via Moran eigenvectors of the graph Laplacian (INLA/Stan-free)
D_diag <- rowSums(W_bin)
Q <- diag(D_diag) - W_bin
eig <- eigen(Q, symmetric = TRUE)
# Smallest non-zero eigenvalues = smoothest spatial patterns
ord <- order(eig$values)
spatial_basis <- eig$vectors[, ord[2:11], drop = FALSE]  # skip the zero eigenvalue

log_smr <- log(pmin(pmax(df$smr, 0.01), 5))
ridge_fit <- MASS::lm.ridge(log_smr ~ spatial_basis, lambda = 1)
# Reconstruct fitted values from ridge
X_bym <- cbind(1, spatial_basis)
# lm.ridge coefs
b_hat <- coef(ridge_fit)
log_smr_hat <- as.numeric(X_bym %*% b_hat)
gdf$smr_bym_approx <- exp(log_smr_hat)

# CAR-like spatial error on log(SMR) as a second Stan-free smoother
sem_bym <- errorsarlm(log_smr ~ 1, data = df, listw = lw, zero.policy = TRUE)
gdf$smr_car <- exp(fitted(sem_bym))
This method assumes the response is known - see manual page

6. Knowledge Check

  1. A county has O=1 death, E=0.8, so SMR=1.25. The EB-smoothed estimate is 1.02. Why does EB shrink this so aggressively?
  2. BYM model sigma_u >> sigma_v. What does this imply about the nature of spatial clustering?
  3. A researcher uses raw SMR to rank counties for intervention. What are the consequences of not smoothing?
  4. Moran’s I for raw SMR = 0.41 but for EB-smoothed SMR = 0.55. Why does smoothing increase spatial autocorrelation?

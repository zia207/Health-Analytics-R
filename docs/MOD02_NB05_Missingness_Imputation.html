Module 02 — Missingness Analysis & Imputation

Missingness Analysis & Imputation

Learning objectives - Visualise missing data patterns with naniar - Test MCAR, MAR, and MNAR mechanisms with statistical tests - Apply median, LOCF, and multiple imputation (MICE) strategies - Compare imputed vs original distributions with overlapping KDE plots - Document imputation decisions for a reproducible analysis plan

Estimated time: 2 hours | Level: Intermediate | Libraries: dplyr / tibble, naniar, visdat, mice, ggplot2

# ── 4.3  KNN Imputation ───────────────────────────────────────────────────────
num_cols <- c("age", "los_days", "glucose", "creatinine", "hba1c", "total_charge")
df_num   <- df %>% select(all_of(num_cols))

knn_impute <- function(dat, k = 5) {
  x <- as.matrix(dat)
  mus <- apply(x, 2, mean, na.rm = TRUE)
  sds <- apply(x, 2, sd,   na.rm = TRUE)
  sds[sds == 0 | is.na(sds)] <- 1
  xs <- sweep(sweep(x, 2, mus, "-"), 2, sds, "/")
  xs_fill <- xs
  for (j in seq_len(ncol(xs_fill))) {
    xs_fill[is.na(xs_fill[, j]), j] <- 0
  }
  miss <- which(is.na(x), arr.ind = TRUE)
  out <- x
  for (i in unique(miss[, 1])) {
    cols_miss <- miss[miss[, 1] == i, 2, drop = TRUE]
    d <- sqrt(rowSums((xs_fill -
                         matrix(xs_fill[i, ], nrow = nrow(xs_fill),
                                ncol = ncol(xs_fill), byrow = TRUE))^2))
    d[i] <- Inf
    nn <- order(d)[seq_len(min(k, length(d) - 1))]
    w  <- 1 / pmax(d[nn], 1e-8)
    w  <- w / sum(w)
    for (j in cols_miss) {
      out[i, j] <- sum(w * (xs_fill[nn, j] * sds[j] + mus[j]))
    }
  }
  as.data.frame(out)
}

5. Comparing Imputed Distributions

plot_kde_compare <- function(col, title) {
  orig <- tibble(value = df[[col]], source = "Original (complete cases)") %>%
    filter(!is.na(value))
  knn  <- tibble(value = df_knn[[col]], source = "KNN imputed")
  mice <- tibble(value = df_mice[[col]], source = "MICE imputed")
  bind_rows(orig, knn, mice) %>%
    ggplot(aes(x = value, colour = source, linetype = source)) +
    geom_density(linewidth = 0.9) +
    scale_colour_manual(values = c(
      "Original (complete cases)" = "black",
      "KNN imputed"  = "#2563EB",
      "MICE imputed" = "#DC2626"
    )) +
    scale_linetype_manual(values = c(
      "Original (complete cases)" = "solid",
      "KNN imputed"  = "dashed",
      "MICE imputed" = "dotted"
    )) +
    labs(title = title, y = "Density", x = NULL, colour = NULL, linetype = NULL) +
    theme_health() +
    theme(legend.position = "bottom", legend.text = element_text(size = 8))
}

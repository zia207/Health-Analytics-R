Health Analytics with R · Module 04: Machine Learning for Clinical Prediction

This capstone integrates NB-01 through NB-07 into a single production-ready ML pipeline.

Pipeline stages: 1. Data assembly, EDA, and train/val/test temporal splits 2. Preprocessing pipeline with leakage-safe transforms 3. Baseline and candidate model comparison (LR, RF, GBM, XGBoost) 4. Hyperparameter optimisation with tune_bayes (best model) 5. Comprehensive evaluation: AUC, AP, calibration, DCA, EPV 6. SHAP-style global and local explanations (DALEX / fastshap) 7. Risk stratification and patient dashboard 8. MLOps-ready: model serialisation + drift detection template 9. Publication-quality summary figure + methods narrative

Estimated time: 4 hours | Level: Advanced

pred_pack <- function(p_val, p_test, name) {
  tibble(
    Model      = name,
    `Val AUC`  = round(auc_prob(y_val,  p_val),  4),
    `Test AUC` = round(auc_prob(y_test, p_test), 4),
    `Val AP`   = round(ap_prob(y_val, p_val), 4),
    `Val Brier` = round(brier_prob(y_val, p_val), 4)
  )
}
race_df <- bind_rows(
  pred_pack(as.numeric(predict(lr, Xval_df, type = "response")),
            as.numeric(predict(lr, Xte_df,  type = "response")), "Logistic Reg (L2)"),
  pred_pack(predict(rf, data = Xval_df)$predictions[, "1"],

) %>% arrange(desc(`Val AUC`))

4. Bayesian Fine-Tuning of Best Model

set.seed(SEED)
# tune_bayes on boost_tree (xgboost engine) — analogue of Optuna TPE
xgb_spec <- boost_tree(
  trees = tune(), tree_depth = tune(), learn_rate = tune(),
  sample_size = tune(), min_n = tune()
) %>% set_engine("xgboost") %>% set_mode("classification")

tune_data <- Xtr_df %>% mutate(readmitted = factor(y_train, levels = c(0, 1)))
xgb_wf <- workflow() %>% add_model(xgb_spec) %>% add_formula(readmitted ~ .)
skf <- vfold_cv(tune_data, v = 3, strata = readmitted)

xgb_param <- extract_parameter_set_dials(xgb_wf) %>%
  update(
    trees      = trees(range = c(100L, 400L)),
    tree_depth = tree_depth(range = c(3L, 8L)),
    learn_rate = learn_rate(range = c(-2, log10(0.3))),
    sample_size = sample_prop(range = c(0.5, 1.0)),
    min_n      = min_n(range = c(5L, 30L))
  )

study <- tune_bayes(
  xgb_wf, resamples = skf, param_info = xgb_param,

risk_tier <- function(prob) {
  case_when(
    prob < 0.10 ~ "Low (<10%)",
    prob < 0.20 ~ "Moderate (10-20%)",
    prob < 0.35 ~ "High (20-35%)",
    TRUE        ~ "Very high (>35%)"
  )
}

net_benefit <- function(yt, yp, t) {
  yp_b <- as.integer(yp >= t)
  tp <- sum(yp_b == 1 & yt == 1); fp <- sum(yp_b == 1 & yt == 0)
  (tp - fp * (t / (1 - t))) / length(yt)
}

)

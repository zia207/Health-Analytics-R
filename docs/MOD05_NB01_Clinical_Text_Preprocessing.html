Module 05 — Clinical Text Preprocessing

Clinical Text Preprocessing

expand_abbreviations <- function(text, abbrev_dict = clinical_abbrevs) {
  # Longest abbreviations first so "DM2" is not split by "DM"
  ord <- order(nchar(names(abbrev_dict)), decreasing = TRUE)
  for (i in ord) {
    abbrev <- names(abbrev_dict)[i]
    expansion <- unname(abbrev_dict[i])
    pattern <- paste0("\\b", stringr::str_replace_all(abbrev, "([\\W])", "\\\\\\1"), "\\b")
    text <- stringr::str_replace_all(text, stringr::regex(pattern), expansion)
  }
  text
}

4. PHI Detection & De-identification

phi_patterns <- c(
  PATIENT_NAME  = "(?:Patient|Pt)\\s*:\\s*[A-Z][a-z]+\\s+[A-Z][a-z]+",
  PROVIDER_NAME = "(?:Dr\\.|Attending)\\s*:\\s*[A-Z][a-z]+\\s+[A-Z][a-z]+",
  MRN           = "MRN\\s*:\\s*\\d{6,10}",
  DATE_OF_BIRTH = "DOB\\s*:\\s*\\d{1,2}/\\d{1,2}/\\d{4}",
  DATE          = "\\b\\d{1,2}/\\d{1,2}/\\d{4}\\b",
  AGE           = "\\b\\d{2,3}\\s*(?:y/o|year[-\\s]old|yo)\\b"
)

detect_phi <- function(text) {
  found <- purrr::imap_dfr(phi_patterns, function(pattern, phi_type) {
    m <- gregexpr(pattern, text, perl = TRUE, ignore.case = TRUE)[[1]]
    if (m[1] == -1) {
      return(tibble::tibble(type = character(), text = character(),
                            start = integer(), end = integer()))
    }
    matched <- regmatches(text, list(m))[[1]]
    tibble::tibble(
      type  = phi_type,
      text  = matched,
      start = as.integer(m),
      end   = as.integer(m + attr(m, "match.length"))
    )
  })
  dplyr::arrange(found, start)
}

deidentify <- function(text) {
  result <- text
  spans <- purrr::imap_dfr(phi_patterns, function(pattern, phi_type) {
    m <- gregexpr(pattern, result, perl = TRUE, ignore.case = TRUE)[[1]]
    if (m[1] == -1) {
      return(tibble::tibble(start = integer(), end = integer(),
                            replacement = character()))
    }
    tibble::tibble(
      start       = as.integer(m),
      end         = as.integer(m + attr(m, "match.length")),
      replacement = paste0("[", phi_type, "]")
    )
  })
  if (nrow(spans) == 0) return(result)
  spans <- dplyr::arrange(spans, dplyr::desc(start))
  for (i in seq_len(nrow(spans))) {
    result <- paste0(
      substr(result, 1, spans$start[i] - 1),
      spans$replacement[i],
      substr(result, spans$end[i], nchar(result))
    )
  }
  result
}

5. Section Extraction

section_patterns <- c(
  HPI         = "(?:HISTORY OF PRESENT ILLNESS|HPI)\\s*:?",
  MEDICATIONS = "MEDICATIONS?\\s*:?",
  ASSESSMENT  = "(?:ASSESSMENT(?:\\s+AND\\s+PLAN)?|A/?P|IMPRESSION)\\s*:?",
  LABS        = "LABS?\\s*:?",
  PLAN        = "PLAN\\s*:?",
  DISCHARGE   = "DISCHARGE CONDITION\\s*:?"
)

extract_sections <- function(text) {
  headers <- purrr::imap_dfr(section_patterns, function(pattern, name) {
    m <- gregexpr(pattern, text, perl = TRUE, ignore.case = TRUE)[[1]]
    if (m[1] == -1) {
      return(tibble::tibble(start = integer(), end = integer(), name = character()))
    }
    tibble::tibble(
      start = as.integer(m),
      end   = as.integer(m + attr(m, "match.length")),
      name  = name
    )
  })
  if (nrow(headers) == 0) return(list())
  headers <- dplyr::arrange(headers, start)
  sections <- list()
  for (i in seq_len(nrow(headers))) {
    next_start <- if (i < nrow(headers)) headers$start[i + 1] else nchar(text) + 1
    content <- substr(text, headers$end[i], next_start - 1)
    content <- stringr::str_squish(stringr::str_replace_all(content, "\\n+", " "))
    if (nzchar(content)) sections[[headers$name[i]]] <- content
  }
  sections
}

6. Vital Signs & Lab Value Extraction

vital_patterns <- c(
  temperature    = "T\\s*(?:emp)?\\s*:?\\s*(\\d{2,3}\\.?\\d?)\\s*[CF]?",
  heart_rate     = "HR\\s*:?\\s*(\\d{2,3})\\s*(?:bpm)?",
  blood_pressure = "BP\\s*:?\\s*(\\d{2,3})\\s*/\\s*(\\d{2,3})",
  resp_rate      = "RR\\s*:?\\s*(\\d{1,2})",
  spo2           = "SpO2\\s*:?\\s*(\\d{2,3})\\s*%"
)
lab_patterns <- c(
  sodium        = "Na\\s*(\\d{3})",
  potassium     = "K\\s*(\\d\\.\\d)",
  creatinine    = "Cr\\s*(\\d\\.\\d)",
  bnp           = "BNP\\s*(\\d{2,5})",
  troponin_i    = "Troponin\\s*I\\s*:?\\s*(\\d+\\.\\d+)",
  ejection_frac = "EF\\s*(\\d{2})\\s*%"
)

extract_vitals_labs <- function(text) {
  results <- list()
  all_pats <- c(vital_patterns, lab_patterns)
  for (name in names(all_pats)) {
    m <- stringr::str_match(text, stringr::regex(all_pats[[name]], ignore_case = TRUE))
    if (!is.na(m[1, 1])) {
      if (name == "blood_pressure") {
        results$systolic  <- as.numeric(m[1, 2])
        results$diastolic <- as.numeric(m[1, 3])
      } else {
        val <- suppressWarnings(as.numeric(m[1, 2]))
        results[[name]] <- if (!is.na(val)) val else m[1, 2]
      }
    }
  }
  results
}

7. Negation Detection

negation_triggers <- c(
  "\\bno\\b", "\\bnot\\b", "\\bdenies?\\b", "\\bwithout\\b",
  "\\bw/o\\b", "\\bnever\\b", "\\babsent\\b", "\\bnegative\\b"
)

is_negated <- function(text, entity, window = 5) {
  tokens <- stringr::str_split(tolower(text), "\\s+")[[1]]
  entity_lower <- tolower(entity)
  for (i in seq_along(tokens)) {
    if (stringr::str_detect(tokens[i], stringr::fixed(entity_lower))) {
      lo <- max(1, i - window)
      context <- paste(tokens[lo:max(1, i - 1)], collapse = " ")
      if (i == 1) context <- ""
      for (trigger in negation_triggers) {
        if (stringr::str_detect(context, stringr::regex(trigger, ignore_case = TRUE))) {
          return(TRUE)
        }
      }
    }
  }
  FALSE
}

8. Full Preprocessing Pipeline

preprocess_clinical_note <- function(text, note_id = "") {
  result <- list(note_id = note_id, original_length = nchar(text))
  result$phi_detected  <- detect_phi(text)
  clean <- deidentify(text)
  result$expanded_text <- expand_abbreviations(clean)
  result$sections      <- extract_sections(text)
  result$vitals_labs   <- extract_vitals_labs(text)
  lines <- stringr::str_split(result$expanded_text, "\n")[[1]]
  sentences <- lines[nchar(stringr::str_squish(lines)) > 0]
  sentences <- sentences[purrr::map_int(stringr::str_split(sentences, "\\s+"), length) > 5]
  result$sentences      <- sentences
  result$sentence_count <- length(sentences)
  result
}

9. Medication Extraction Exercise

med_pattern <- paste0(
  "(\\d+\\.\\s+)?([A-Z][a-zA-Z]+(?:\\s+[a-zA-Z]+)?)",
  "\\s+(\\d+(?:\\.\\d+)?\\s*(?:mg|mcg|g|mL|units?))",
  "(?:\\s+((?:PO|IV|SQ|IM)\\s+(?:BID|QD|TID|QAM|q\\d+h)))?"
)

extract_medications <- function(text) {
  m <- stringr::str_match_all(text, stringr::regex(med_pattern, ignore_case = TRUE))[[1]]
  if (nrow(m) == 0) {
    return(tibble::tibble(drug = character(), dose = character(), route_freq = character()))
  }
  tibble::tibble(
    drug       = stringr::str_squish(m[, 3]),
    dose       = stringr::str_squish(m[, 4]),
    route_freq = stringr::str_squish(ifelse(is.na(m[, 5]), "", m[, 5]))
  )
}

Health Analytics with R · Module 02: EDA in Healthcare

Learning objectives - Aggregate daily, weekly, and monthly admission counts with lubridate floor dates - Detect seasonality and long-term trends in hospital admissions - Overlay policy/intervention events as annotated vertical markers - Compute rolling averages and Bollinger-style control limits - Decompose a time-series into trend, seasonality, and residual - Build an interactive Plotly time-series chart

Estimated time: 1.5 hours | Level: Intermediate | Libraries: dplyr / tibble, ggplot2, stats, plotly, lubridate

1. Setup & Time-Series Dataset

library(tidyverse)
library(ggplot2)
library(patchwork)
library(lubridate)
library(scales)
library(plotly)

mod02_dir <- file.path(tempdir(), "mod02")
dir.create(mod02_dir, showWarnings = FALSE, recursive = TRUE)

theme_health <- function(base_size = 12) {
  theme_minimal(base_size = base_size) +
    theme(
      panel.grid.minor = element_blank(),
      plot.background  = element_rect(fill = "white", colour = NA),
      panel.background = element_rect(fill = "white", colour = NA)
    )
}
theme_set(theme_health())

# ── Reconstruct dataset ───────────────────────────────────────────────────────
set.seed(42)
N <- 800
ages      <- as.integer(pmin(pmax(rnorm(N, 62, 16), 18), 95))
sexes     <- sample(c("M", "F"), N, replace = TRUE, prob = c(0.48, 0.52))
payers    <- sample(c("Medicare", "Medicaid", "Private", "Self-pay", "Other"),
                    N, replace = TRUE, prob = c(0.40, 0.22, 0.28, 0.07, 0.03))
drg_codes <- sample(c(470, 291, 392, 690, 871, 193, 247, 603), N, replace = TRUE)
drg_labels_map <- c(
  `470` = "Major joint replacement", `291` = "Heart failure & shock",
  `392` = "Esophagitis/misc GI",     `690` = "Kidney/UTI",
  `871` = "Septicemia",              `193` = "Simple pneumonia",
  `247` = "Perc cardiovasc w stent", `603` = "Cellulitis"
)
los_days   <- as.integer(pmin(pmax(rgamma(N, shape = 2.5, scale = 1.8), 1), 30))
charges    <- round(los_days * runif(N, 1800, 4200), 2)
charges[runif(N) < 0.08] <- NA_real_
primary_dx <- sample(c("E11.9", "I10", "N18.3", "I50.9", "J18.9", "A41.9", "M16.11", "N39.0"),
                     N, replace = TRUE)
admit_type <- sample(c("Emergency", "Elective", "Urgent", "Newborn"),
                     N, replace = TRUE, prob = c(0.52, 0.30, 0.16, 0.02))
set.seed(99)
df <- tibble(
  patient_id    = sprintf("PT-%05d", seq_len(N)),
  age           = ages,
  sex           = sexes,
  payer         = payers,
  drg_code      = drg_codes,
  drg_label     = unname(drg_labels_map[as.character(drg_codes)]),
  primary_dx    = primary_dx,
  admit_type    = admit_type,
  los_days      = los_days,
  total_charge  = charges,
  diabetes      = rbinom(N, 1, 0.32),
  hypertension  = rbinom(N, 1, 0.45),
  ckd           = rbinom(N, 1, 0.15),
  heart_failure = rbinom(N, 1, 0.18),
  readmitted_30 = rbinom(N, 1, 0.14)
)
df$age_group <- cut(df$age, breaks = c(17, 44, 64, 74, 95),
                    labels = c("18-44", "45-64", "65-74", "75+"))
start <- as.Date("2019-01-01")
set.seed(5)
n_days <- as.integer(as.Date("2023-12-31") - start)
df$admit_date     <- as.Date(start + sample(0:(n_days - 1), N, replace = TRUE))
df$discharge_date <- df$admit_date + df$los_days
df <- df %>% arrange(admit_date)

# ── Add COVID-style shock to 2020 ─────────────────────────────────────────────
# (already embedded in data via seed; we add a synthetic elective drop marker)
cat(sprintf("Admissions: %d  |  Date range: %s → %s\n",
            nrow(df), min(df$admit_date), max(df$admit_date)))

2. Resampling — Daily, Weekly, Monthly Counts

daily <- df %>%
  count(date = admit_date, name = "admissions") %>%
  complete(date = seq(min(date), max(date), by = "day"), fill = list(admissions = 0))

weekly <- df %>%
  count(date = floor_date(admit_date, "week"), name = "admissions") %>%
  complete(date = seq(min(date), max(date), by = "week"), fill = list(admissions = 0))

monthly <- df %>%
  count(date = floor_date(admit_date, "month"), name = "admissions") %>%
  complete(date = seq(min(date), max(date), by = "month"), fill = list(admissions = 0))

plot_resolution <- function(ts, title, colour) {
  ggplot(ts, aes(x = date, y = admissions)) +
    geom_ribbon(aes(ymin = 0, ymax = admissions), fill = colour, alpha = 0.20) +
    geom_line(colour = colour, linewidth = if (title == "Daily admissions") 0.4 else 0.8) +
    scale_x_date(date_breaks = "6 months", date_labels = "%Y-%m") +
    labs(title = title, x = NULL, y = "N admissions") +
    theme_health() +
    theme(axis.text.x = element_text(angle = 30, hjust = 1))
}

p_res <- plot_resolution(daily,   "Daily admissions",   "#AEC6CF") /
  plot_resolution(weekly,  "Weekly admissions",  "#2563EB") /
  plot_resolution(monthly, "Monthly admissions", "#1F4E79") +
  plot_annotation(title = "Admission counts at three temporal resolutions")
print(p_res)
ggsave(file.path(mod02_dir, "ts_resolutions.png"), p_res,
       width = 14, height = 10, dpi = 300)
cat(sprintf("Monthly — mean: %.1f  std: %.1f\n",
            mean(monthly$admissions), sd(monthly$admissions)))

3. Rolling Averages & Control Limits

roll_k <- 7
daily <- daily %>%
  mutate(
    roll_mean = as.numeric(stats::filter(admissions, rep(1 / roll_k, roll_k), sides = 2)),
    roll_std  = as.numeric(sqrt(stats::filter(admissions^2, rep(1 / roll_k, roll_k), sides = 2) -
                                  roll_mean^2)),
    upper_cl  = roll_mean + 2 * roll_std,
    lower_cl  = pmax(roll_mean - 2 * roll_std, 0)
  )

events <- tibble(
  dt    = as.Date(c("2020-03-15", "2020-06-01", "2021-01-01", "2022-09-01")),
  label = c("COVID-19 onset", "Elective resumption", "Vaccine rollout", "Policy change"),
  colour = c("#DC2626", "#16A34A", "purple", "orange")
)

p_roll <- ggplot(daily, aes(x = date)) +
  geom_line(aes(y = admissions, colour = "Daily"), linewidth = 0.35, alpha = 0.8) +
  geom_ribbon(aes(ymin = lower_cl, ymax = upper_cl, fill = "±2 SD control band"),
              alpha = 0.15, colour = NA) +
  geom_line(aes(y = roll_mean, colour = "7-day rolling mean"), linewidth = 1.0) +
  geom_vline(data = events, aes(xintercept = dt, colour = label),
             linetype = "dashed", linewidth = 0.7, show.legend = FALSE) +
  geom_text(data = events, aes(x = dt, y = max(daily$admissions, na.rm = TRUE) * 0.92,
                               label = label, colour = label),
            size = 2.8, vjust = 1, show.legend = FALSE) +
  scale_colour_manual(
    values = c("Daily" = "#B0C4DE", "7-day rolling mean" = "#1F4E79",
               setNames(events$colour, events$label)),
    breaks = c("Daily", "7-day rolling mean")
  ) +
  scale_fill_manual(values = c("±2 SD control band" = "#1F4E79")) +
  scale_x_date(date_breaks = "3 months", date_labels = "%Y-%m") +
  labs(y = "Admissions per day", x = NULL, colour = NULL, fill = NULL,
       title = "Daily admissions with 7-day rolling mean and ±2 SD control band") +
  theme_health() +
  theme(axis.text.x = element_text(angle = 30, hjust = 1),
        legend.position = "bottom")
print(p_roll)

5. Seasonal Decomposition

# Use monthly series (requires ≥2 full periods)
ts_monthly <- ts(monthly$admissions, frequency = 12,
                 start = c(year(min(monthly$date)), month(min(monthly$date))))
decomp <- decompose(ts_monthly, type = "additive")

decomp_df <- tibble(
  date     = monthly$date,
  Observed = as.numeric(decomp$x),
  Trend    = as.numeric(decomp$trend),
  Seasonal = as.numeric(decomp$seasonal),
  Residual = as.numeric(decomp$random)
) %>%
  pivot_longer(-date, names_to = "component", values_to = "value") %>%
  mutate(component = factor(component, levels = c("Observed", "Trend", "Seasonal", "Residual")))

comp_cols <- c(Observed = "#1F4E79", Trend = "#DC2626",
               Seasonal = "#16A34A", Residual = "#B47CC7")

p_decomp <- ggplot(decomp_df, aes(x = date, y = value, colour = component)) +
  geom_line(linewidth = 0.8, na.rm = TRUE) +
  geom_hline(data = tibble(component = factor(c("Seasonal", "Residual"),
                                              levels = levels(decomp_df$component)),
                           yint = 0),
             aes(yintercept = yint), colour = "gray", linetype = "dotted", linewidth = 0.5) +
  scale_colour_manual(values = comp_cols, guide = "none") +
  scale_x_date(date_breaks = "6 months", date_labels = "%Y-%m") +
  facet_wrap(~ component, ncol = 1, scales = "free_y") +
  labs(title = "Seasonal decomposition of monthly admissions (additive model)",
       x = NULL, y = NULL) +
  theme_health() +
  theme(axis.text.x = element_text(angle = 30, hjust = 1))
print(p_decomp)
ggsave(file.path(mod02_dir, "ts_decomposition.png"), p_decomp,
       width = 14, height = 9, dpi = 300)

# Seasonal strength index
seasonal_strength <- 1 - var(decomp$random, na.rm = TRUE) /
  var(decomp$seasonal + decomp$random, na.rm = TRUE)
cat(sprintf("Seasonal strength index: %.3f  (>0.64 = strong)\n", seasonal_strength))
Seasonal strength index: 0.235  (>0.64 = strong)

6. Admission Trends by Diagnosis Group

top_drgs <- df %>%
  count(drg_label, sort = TRUE) %>%
  slice_head(n = 4) %>%
  pull(drg_label)

monthly_drg <- df %>%
  filter(drg_label %in% top_drgs) %>%
  count(date = floor_date(admit_date, "month"), drg_label) %>%
  complete(date, drg_label, fill = list(n = 0)) %>%
  group_by(drg_label) %>%
  mutate(roll = as.numeric(stats::filter(n, rep(1 / 3, 3), sides = 2))) %>%
  ungroup()

colors_drg <- c("#1F4E79", "#DC2626", "#16A34A", "#B47CC7")
names(colors_drg) <- unique(monthly_drg$drg_label)

p_drg <- ggplot(monthly_drg, aes(x = date, y = roll, colour = drg_label, fill = drg_label)) +
  geom_ribbon(aes(ymin = 0, ymax = roll), alpha = 0.08, colour = NA) +
  geom_line(linewidth = 1.0, na.rm = TRUE) +
  scale_colour_manual(values = colors_drg) +
  scale_fill_manual(values = colors_drg) +

       title = "Admission trends by DRG — top 4 diagnoses") +

print(p_drg)

  arrange(date) %>%

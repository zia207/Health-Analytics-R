Health Analytics with R · Module 02: EDA in Healthcare

Learning objectives - Join patient ZIP codes to census tracts using synthetic spatial data - Compute age-adjusted hospitalisation rates and readmission rates by area - Create choropleth maps with sf and ggplot2 - Overlay a social vulnerability index (SVI) and compute spatial correlations - Export interactive maps with leaflet

Estimated time: 3 hours | Level: Advanced | Libraries: dplyr / tibble, sf, ggplot2, leaflet, stats

2. Synthetic Geospatial Dataset — 50 ZIP Code Areas

set.seed(42)

# ── Simulate 50 ZIP code areas (modelled on a mid-size US metropolitan area) ──
N_ZIPS <- 50

# Geographic centroids (roughly spanning a metro area, lat/lon)
lat_center <- 43.0
lon_center <- -76.1   # ~ Upstate NY
zip_df <- tibble(
  zip           = sprintf("%05d", 10000 + seq_len(N_ZIPS) - 1),
  tract_name    = sprintf("Tract %d", 100 + seq_len(N_ZIPS) - 1),
  lat           = runif(N_ZIPS, lat_center - 0.5, lat_center + 0.5),
  lon           = runif(N_ZIPS, lon_center - 0.5, lon_center + 0.5),
  population    = sample(3000:25000, N_ZIPS, replace = TRUE),
  pct_poverty   = rbeta(N_ZIPS, 2, 8) * 50,     # 0-50%
  pct_uninsured = rbeta(N_ZIPS, 1.5, 8) * 30,
  median_income = pmin(pmax(rnorm(N_ZIPS, 52000, 18000), 15000), 120000),
  svi_score     = rbeta(N_ZIPS, 2, 3),          # CDC SVI 0–1, higher=worse
  pct_elderly   = rbeta(N_ZIPS, 2, 6) * 40,     # % aged 65+
  urban_rural   = sample(c("Urban", "Suburban", "Rural"), N_ZIPS,
                         replace = TRUE, prob = c(0.50, 0.35, 0.15))
)

# Hospitalisations inversely correlated with income and SVI
zip_df$hosp_rate <- pmin(pmax(
  200
  + zip_df$svi_score * 180
  - zip_df$median_income / 1000 * 1.5
  + zip_df$pct_elderly * 4.5
  + rnorm(N_ZIPS, 0, 25),
  50), 600)

# Readmission rate
zip_df$readmit_rate <- pmin(pmax(
  12
  + zip_df$svi_score * 10
  + zip_df$pct_poverty * 0.18
  + rnorm(N_ZIPS, 0, 1.5),
  5), 35)

# Age-adjusted rate (add noise to crude to simulate adjustment)
zip_df$hosp_rate_adj <- pmin(pmax(
  zip_df$hosp_rate - zip_df$pct_elderly * 3.2 + rnorm(N_ZIPS, 0, 15),
  50), 600)

3. Patient-to-ZIP Assignment & Aggregation

# ── Generate patient discharge records with ZIP codes ─────────────────────────
set.seed(99)
N <- 800
# Assign patients to ZIPs weighted by population
zip_weights <- zip_df$population / sum(zip_df$population)
patient_zips <- sample(zip_df$zip, size = N, replace = TRUE, prob = zip_weights)

patients <- tibble(
  patient_id    = sprintf("PT-%05d", seq_len(N)),
  zip           = patient_zips,
  age           = as.integer(pmin(pmax(rnorm(N, 62, 16), 18), 95)),
  sex           = sample(c("M", "F"), N, replace = TRUE, prob = c(0.48, 0.52)),
  los_days      = as.integer(pmin(pmax(rgamma(N, shape = 2.5, scale = 1.8), 1), 30)),
  readmitted_30 = rbinom(N, 1, 0.14),
  total_charge  = round(pmin(pmax(rgamma(N, shape = 2.5, scale = 1.8), 1), 30) * 2500, 2)
)

# Aggregate to ZIP level
zip_agg <- patients %>%
  group_by(zip) %>%
  summarise(
    n_discharges = n(),
    n_readmitted = sum(readmitted_30),
    mean_los     = mean(los_days),
    mean_charge  = mean(total_charge),
    .groups = "drop"
  ) %>%
  mutate(observed_readmit_rate = n_readmitted / n_discharges * 100)

# Merge with ZIP reference table
zip_merged <- zip_df %>%
  left_join(zip_agg, by = "zip") %>%
  mutate(
    n_discharges = replace_na(n_discharges, 0),
    n_readmitted = replace_na(n_readmitted, 0)
  )
cat(sprintf("Merged ZIP dataset: %d x %d\n", nrow(zip_merged), ncol(zip_merged)))

# ── Scatter plots: SVI vs health outcomes ────────────────────────────────────
color_urban <- c(Urban = "#2563EB", Suburban = "#16A34A", Rural = "#DC2626")

plot_svi <- function(metric, ylabel) {
  fit <- lm(zip_merged[[metric]] ~ zip_merged$svi_score)
  r   <- cor.test(zip_merged$svi_score, zip_merged[[metric]], use = "complete.obs")
  ggplot(zip_merged, aes(x = svi_score, y = .data[[metric]],
                         colour = urban_rural, size = population)) +
    geom_point(alpha = 0.7) +
    geom_abline(intercept = coef(fit)[1], slope = coef(fit)[2],
                colour = "black", linetype = "dashed", linewidth = 0.7) +
    annotate("label", x = 0.04, y = Inf, hjust = 0, vjust = 1.4,
             label = sprintf("r = %.2f, p = %.3f",
                             unname(r$estimate), r$p.value),
             size = 3.3, fill = "white", label.size = 0) +
    scale_colour_manual(values = color_urban, name = "Area type") +
    scale_size_continuous(range = c(2, 9), guide = "none") +
    labs(x = "Social Vulnerability Index (SVI)", y = ylabel,
         title = paste(ylabel, "vs SVI")) +
    theme_health()
}

6. Quintile Disparity Analysis

# Stratify ZIP codes into SVI quintiles and compare outcomes
zip_merged <- zip_merged %>%
  mutate(svi_quintile = cut(
    svi_score,
    breaks = quantile(svi_score, probs = seq(0, 1, 0.2), na.rm = TRUE),
    labels = c("Q1 (Least)", "Q2", "Q3", "Q4", "Q5 (Most)"),
    include.lowest = TRUE
  ))

disparity_table <- zip_merged %>%
  group_by(svi_quintile) %>%
  summarise(
    n_zips          = n(),
    mean_population = mean(population),
    hosp_rate       = mean(hosp_rate_adj),
    readmit_rate    = mean(readmit_rate),
    pct_uninsured   = mean(pct_uninsured),
    median_income   = mean(median_income),
    .groups = "drop"
  ) %>%
  mutate(across(where(is.numeric), ~ round(.x, 1)))

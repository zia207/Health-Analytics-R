Health Analytics with R · Module 04: Machine Learning for Clinical Prediction

Learning objectives - Understand Shapley values and the SHAP framework conceptually - Generate global explanations (summary, bar, beeswarm plots) - Produce local explanations (waterfall, force plots) for individual patients - Build SHAP dependence plots to reveal interaction effects - Create a clinical explanation report for a high-risk patient

Estimated time: 2.5 hours | Level: Advanced | Libraries: DALEX, fastshap, vip, tidymodels, ggplot2

set.seed(42); N <- 3000
clip <- function(x, lo, hi) pmin(pmax(x, lo), hi)
b <- matrix(rnorm(N * 4), ncol = 4); set.seed(99)
age <- as.integer(clip(rnorm(N, 62, 15), 18, 92))
sex <- sample(c("M", "F"), N, replace = TRUE, prob = c(0.48, 0.52))
payer <- sample(c("Medicare", "Medicaid", "Private", "Self-pay", "Other"),
                N, replace = TRUE, prob = c(0.40, 0.22, 0.28, 0.07, 0.03))
admit_type <- sample(c("Emergency", "Elective", "Urgent"),
                     N, replace = TRUE, prob = c(0.52, 0.30, 0.18))
los_days <- as.integer(clip(round(rgamma(N, shape = 2.5, scale = 1.8)), 1, 30))
diabetes <- rbinom(N, 1, sig(0.6 * b[, 1] - 0.5))
hypertension <- rbinom(N, 1, sig(0.7 * b[, 1] - 0.2))
hf <- rbinom(N, 1, sig(0.4 * b[, 2] + 0.5 * hypertension - 1.5))
copd <- rbinom(N, 1, sig(0.5 * b[, 3] - 1.0))
ckd <- rbinom(N, 1, sig(0.5 * b[, 1] + 0.6 * diabetes - 1.2))
obesity <- rbinom(N, 1, sig(0.5 * b[, 1] - 0.8))
depression <- rbinom(N, 1, sig(0.3 * b[, 4] - 1.4))
n_comorb <- diabetes + hypertension + hf + copd + ckd
set.seed(21)
glucose <- round(clip(rnorm(N, 105 + 15 * diabetes, 28), 50, 400), 1)
creatinine <- round(clip(rgamma(N, shape = 1.6 + 0.8 * hf, scale = 0.6), 0.4, 10), 2)
hba1c <- round(clip(rnorm(N, 6.5 + 0.8 * diabetes, 1.4), 4, 14), 1)
sbp <- round(clip(rnorm(N, 128 + 12 * hypertension, 18), 80, 200), 0)
bmi <- round(clip(rnorm(N, 28 + 4 * obesity, 6), 15, 55), 1)
logit <- (-3.2 + 0.6 * hf + 0.5 * diabetes + 0.5 * ckd + 0.3 * copd +
            0.02 * (age - 62) / 15 +
            0.3 * as.integer(admit_type == "Emergency") +
            0.25 * as.integer(payer == "Medicaid") +
            0.2 * as.integer(los_days > 7) +
            0.5 * hf * diabetes + rnorm(N, 0, 0.25))
readmitted <- rbinom(N, 1, sig(logit))
df <- tibble(
  age, sex, payer, admit_type, los_days,
  diabetes, hypertension, hf, copd, ckd, obesity, depression, n_comorb,
  glucose, creatinine, hba1c, sbp, bmi, readmitted
)
df$los_gt7 <- as.integer(df$los_days > 7)
NUMERIC <- c("age", "los_days", "n_comorb", "glucose", "creatinine", "hba1c", "sbp", "bmi")
BINARY <- c("diabetes", "hypertension", "hf", "copd", "ckd", "obesity", "depression", "los_gt7")
CATEGORIC <- c("sex", "payer", "admit_type")
FEATURES <- c(NUMERIC, BINARY, CATEGORIC); TARGET <- "readmitted"

set.seed(42)
df_ml <- df %>%
  select(all_of(c(FEATURES, TARGET))) %>%
  mutate(readmitted = factor(readmitted, levels = c(0, 1)))
split <- initial_split(df_ml, prop = 0.80, strata = readmitted)
df_tr <- training(split); df_te <- testing(split)
rec <- recipe(readmitted ~ ., data = df_tr) %>%
  step_impute_median(all_of(NUMERIC)) %>%
  step_normalize(all_of(NUMERIC)) %>%
  step_impute_median(all_of(BINARY)) %>%
  step_impute_mode(all_of(CATEGORIC)) %>%
  step_dummy(all_of(CATEGORIC), one_hot = TRUE)
rec_prep <- prep(rec, training = df_tr)
Xtr_tbl <- bake(rec_prep, new_data = df_tr) %>% select(-readmitted)
Xte_tbl <- bake(rec_prep, new_data = df_te) %>% select(-readmitted)
Xtr <- as.matrix(Xtr_tbl); Xte <- as.matrix(Xte_tbl)
y_tr <- as.integer(as.character(df_tr$readmitted))
y_te <- as.integer(as.character(df_te$readmitted))
feat_nm <- colnames(Xtr)
X_te_named <- Xte_tbl
X_tr_named <- Xtr_tbl

model <- xgboost(
  data = Xtr, label = factor(y_tr, levels = c(0, 1)), nrounds = 200, max_depth = 4,
  eta = 0.05, subsample = 0.8, objective = "binary:logistic",
  verbose = 0
)
cat(sprintf("Model AUC: %.4f\n", auc_prob(y_te, predict(model, Xte))))

2. SHAP — Conceptual Foundation

SHAP value for feature \(j\) on observation \(i\): the average marginal contribution of feature \(j\) to the prediction, weighted over all possible subsets of features (from game theory’s Shapley values).

Properties: - Additivity: sum of all SHAP values = prediction − base rate - Consistency: if a feature contributes more in model A than B, its SHAP value is larger in A - Local accuracy: SHAP values exactly decompose each individual prediction

# ── Compute SHAP values (fastshap) + DALEX explainer ──────────────────────────
pred_fun <- function(object, newdata) {
  predict(object, as.matrix(newdata))
}

explainer <- DALEX::explain(
  model, data = as.data.frame(X_tr_named), y = y_tr,
  predict_function = pred_fun, label = "xgboost", verbose = FALSE
)

set.seed(42)
# Kernel-style SHAP via Monte Carlo (fastshap); nsim kept modest for tutorial speed
shap_vals <- fastshap::explain(
  model, X = as.data.frame(X_tr_named), newdata = as.data.frame(X_te_named),
  pred_wrapper = pred_fun, nsim = 30, adjust = TRUE
)
shap_mat <- as.matrix(shap_vals)
shap_vals_sample <- shap_mat[seq_len(min(300, nrow(shap_mat))), , drop = FALSE]
X_sample <- X_te_named[seq_len(min(300, nrow(X_te_named))), , drop = FALSE]

cat(sprintf("SHAP values shape: %d x %d\n", nrow(shap_mat), ncol(shap_mat)))
SHAP values shape: 600 x 26
base_rate <- mean(predict(model, Xtr))
cat(sprintf("Expected value (base rate): %.4f\n", base_rate))

cat(sprintf("  (corresponds to: %.1f%% readmission probability)\n", base_rate * 100))
  (corresponds to: 10.4% readmission probability)
# Verify additivity: prediction ≈ base + sum(SHAP)  (adjust=TRUE enforces this)
idx <- 1

# ── Global summary plot ───────────────────────────────────────────────────────
mean_abs <- colMeans(abs(shap_mat))

p_bar <- ggplot(imp_df, aes(importance, reorder(feature, importance))) +
  geom_col(fill = col_blue, alpha = 0.85, colour = "white") +
  labs(x = "Mean |SHAP value|", y = NULL,
       title = "Global Feature Importance (mean |SHAP value| = average impact on model output)")

# Beeswarm plot (manual version)
set.seed(42)
top15_idx <- order(mean_abs)[(length(mean_abs) - 14):length(mean_abs)]
bee <- map_dfr(seq_along(top15_idx), function(rank) {
  feat_idx <- top15_idx[rank]
  shap_col <- shap_vals_sample[, feat_idx]
  feat_col <- X_sample[[feat_idx]]
  rng <- diff(range(feat_col)) + 1e-8
  tibble(
    shap = shap_col,
    y = rank + rnorm(length(shap_col), 0, 0.08),
    feat = feat_nm[feat_idx],
    rank = rank,
    norm_val = (feat_col - min(feat_col)) / rng
  )
})

p_bee <- ggplot(bee, aes(shap, y, colour = norm_val)) +
  geom_point(alpha = 0.4, size = 0.8) +
  geom_vline(xintercept = 0, colour = "black", linewidth = 0.4) +
  scale_colour_gradient2(low = col_blue, mid = "white", high = col_red, midpoint = 0.5) +
  scale_y_continuous(breaks = seq_along(top15_idx),
                     labels = feat_nm[top15_idx]) +
  labs(x = "SHAP value (impact on predicted probability)",
       y = NULL, colour = "Feature value (normalised)",
       title = "Beeswarm Plot (red=high feature value, blue=low)")

probs_te <- predict(model, Xte)

# Manual waterfall plot
top_k <- 12
top_local_idx <- order(abs(shap_patient), decreasing = TRUE)[1:top_k]
local_df <- tibble(
  label = sprintf("%s = %.2f", feat_nm[top_local_idx], feat_vals[top_local_idx]),
  val   = shap_patient[top_local_idx]
) %>%
  mutate(
    label = fct_reorder(label, val),
    direction = if_else(val > 0, "increases risk", "decreases risk")
  )

actual <- if (y_te[high_risk_idx] == 1) "Readmitted" else "Not readmitted"
ggplot(local_df, aes(val, label, fill = direction)) +
  geom_col(alpha = 0.85, colour = "white") +
  geom_vline(xintercept = 0, colour = "black", linewidth = 0.4) +
  geom_text(aes(label = sprintf("%+.3f → %s", val, direction)),
            hjust = if_else(local_df$val > 0, -0.05, 1.05), size = 2.7) +

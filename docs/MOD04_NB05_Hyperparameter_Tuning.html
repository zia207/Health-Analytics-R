Module 04 — Machine Learning for Clinical Prediction

Hyperparameter Tuning & Cross-Validation

Learning objectives - Compare grid search, random search, and Bayesian optimisation (tune_grid / tune_bayes) - Implement nested cross-validation for unbiased performance estimation - Use successive-halving search for efficient large-scale tuning - Build learning curves to diagnose underfitting and overfitting - Apply regularisation paths (Ridge, LASSO, ElasticNet logistic regression)

Estimated time: 2 hours | Level: Advanced | Libraries: tidymodels, tune, dials, ggplot2

# Successive halving (factor = 3) on the same grid
# Evaluate all configs on a small subsample, keep top 1/3, expand data, repeat
halving_search <- function(grid, data, factor_h = 3, seed = 42) {
  set.seed(seed)
  n <- nrow(data)
  sizes <- unique(pmin(n, round(n * (factor_h^(0:2)) / (factor_h^2))))
  sizes[length(sizes)] <- n
  remain <- grid
  last_scores <- NULL
  for (s in sizes) {
    idx <- sample.int(n, s)
    sub <- data[idx, ]
    folds_h <- vfold_cv(sub, v = 3, strata = readmitted)
    tun <- tune_grid(
      xgb_wf, resamples = folds_h, grid = remain,
      metrics = metric_set(roc_auc),
      control = control_grid(verbose = FALSE)
    )
    ranked <- collect_metrics(tun) %>%
      filter(.metric == "roc_auc") %>%
      arrange(desc(mean))
    last_scores <- ranked
    n_keep <- max(1, floor(nrow(ranked) / factor_h))
    keep_cols <- names(remain)
    remain <- ranked %>% select(all_of(keep_cols)) %>% slice_head(n = n_keep)
  }
  last_scores
}
t0 <- Sys.time()
hs <- halving_search(grid, train_baked, factor_h = 3, seed = 42)
t_halv <- as.numeric(difftime(Sys.time(), t0, units = "secs"))
cat(sprintf("Halving grid   : auto reduce | best AUC=%.4f | time=%.1fs\n",
            hs$mean[1], t_halv))

3. Nested Cross-Validation — Unbiased Evaluation

# Nested CV: inner loop tunes hyperparams, outer loop evaluates generalisation
# This avoids the optimistic bias of tuning and evaluating on the same data
set.seed(42)
outer_cv <- vfold_cv(train_baked, v = 5, strata = readmitted)
param_grid_small <- expand_grid(
  trees = c(100, 200), tree_depth = c(4, 6),
  learn_rate = c(0.05, 0.10), sample_size = 0.8
)

nested_auc <- map_dbl(seq_along(outer_cv$splits), function(fold) {
  sp <- outer_cv$splits[[fold]]
  d_in  <- analysis(sp)
  d_out <- assessment(sp)
  inner <- vfold_cv(d_in, v = 3, strata = readmitted)
  gs_inner <- tune_grid(
    xgb_wf, resamples = inner, grid = param_grid_small,
    metrics = metric_set(roc_auc),
    control = control_grid(verbose = FALSE)
  )
  best <- select_best(gs_inner, metric = "roc_auc")
  fit  <- finalize_workflow(xgb_wf, best) %>% fit(d_in)
  p    <- predict(fit, new_data = d_out, type = "prob")$.pred_1
  auc_out <- auc_prob(as.integer(as.character(d_out$readmitted)), p)
  cat(sprintf("  Outer fold %d: best_params=%s | AUC=%.4f\n",
              fold, paste(names(best), best, sep = "=", collapse = ", "), auc_out))
  auc_out
})

4. Learning Curves

best_params <- select_best(gs, metric = "roc_auc")
best_wf <- finalize_workflow(xgb_wf, best_params)

train_fracs <- seq(0.10, 1.0, length.out = 10)
lc <- map_dfr(train_fracs, function(frac) {
  set.seed(42)
  n_use <- max(50, floor(frac * nrow(train_baked)))
  folds_lc <- vfold_cv(train_baked, v = 3, strata = readmitted)
  scores <- map(folds_lc$splits, function(sp) {
    tr <- analysis(sp)
    te <- assessment(sp)
    tr <- tr[sample.int(nrow(tr), min(n_use, nrow(tr))), ]
    fit <- best_wf %>% fit(tr)
    p_tr <- predict(fit, new_data = tr, type = "prob")$.pred_1
    p_te <- predict(fit, new_data = te, type = "prob")$.pred_1
    tibble(
      train = auc_prob(as.integer(as.character(tr$readmitted)), p_tr),
      val   = auc_prob(as.integer(as.character(te$readmitted)), p_te)
    )
  }) %>% bind_rows()
  tibble(
    n = n_use,
    train_mean = mean(scores$train), train_sd = sd(scores$train),
    val_mean   = mean(scores$val),   val_sd   = sd(scores$val)
  )
})

gap <- lc$train_mean[nrow(lc)] - lc$val_mean[nrow(lc)]
ggplot(lc, aes(n)) +
  geom_ribbon(aes(ymin = train_mean - train_sd, ymax = train_mean + train_sd),
              fill = col_red, alpha = 0.15) +

5. Regularisation Paths for Logistic Regression

Cs <- 10^seq(-4, 4, length.out = 30)
# glmnet lambda is inverse of C (inverse regularisation strength); use lambda = 1/C
path <- map_dfr(Cs, function(C) {
  lam <- 1 / C
  w <- ifelse(y_tr == 1, 0.5 / mean(y_tr == 1), 0.5 / mean(y_tr == 0))
  fit <- glmnet(Xtr, y_tr, family = "binomial", alpha = 0, lambda = lam, weights = w)
  p_tr <- as.numeric(predict(fit, Xtr, type = "response"))
  p_te <- as.numeric(predict(fit, Xte, type = "response"))
  tibble(C = C, train = auc_prob(y_tr, p_tr), test = auc_prob(y_te, p_te))
})
best_C <- path$C[which.max(path$test)]

ggplot(path, aes(C)) +
  geom_line(aes(y = train, colour = "Train AUC"), linewidth = 1) +
  geom_line(aes(y = test,  colour = "Test AUC"),  linewidth = 1) +
  geom_vline(xintercept = best_C, colour = col_green, linetype = "dashed") +
  annotate("rect", xmin = min(Cs), xmax = 1e-1, ymin = -Inf, ymax = Inf,
           alpha = 0.05, fill = col_blue) +

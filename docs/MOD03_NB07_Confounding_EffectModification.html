Health Analytics with R · Module 03: Statistical Inference

Learning objectives - Distinguish confounding from effect modification (interaction) conceptually and empirically - Use DAGs to identify adjustment sets - Test for effect modification with interaction terms - Apply stratification and regression adjustment to control confounding - Compute E-values as a sensitivity analysis for unmeasured confounding

# ── Age is a confounder: older → more diabetes AND more readmission ────────────
age <- as.integer(pmin(pmax(rnorm(N, 62, 15), 18), 95))
age_std <- (age - 62) / 15

diabetes <- rbinom(N, 1, logistic(-1.0 + 0.05 * age_std))
sex      <- sample(c("M", "F"), N, replace = TRUE, prob = c(0.48, 0.52))
payer    <- sample(c("Medicare", "Medicaid", "Private"), N, replace = TRUE,
                   prob = c(0.42, 0.22, 0.36))
los_days <- as.integer(pmin(pmax(rgamma(N, shape = 2.5, scale = 1.8), 1), 30))
hf       <- rbinom(N, 1, logistic(-1.8 + 0.6 * diabetes + 0.04 * age_std))

# Readmission: diabetes → readmission but age is a confounder
readmit_logit <- (-2.8 + 0.5 * diabetes + 0.03 * age_std + 0.6 * hf
                  + 0.2 * as.integer(payer == "Medicaid")
                  + 0.15 * as.integer(los_days > 7)
                  + rnorm(N, 0, 0.2))
readmitted <- rbinom(N, 1, logistic(readmit_logit))

# Effect modifier: diabetes effect on readmission is STRONGER in patients with HF
readmit_em_logit <- (-2.8 + 0.3 * diabetes + 0.03 * age_std + 0.6 * hf
                     + 0.9 * diabetes * hf              # interaction term
                     + 0.2 * as.integer(payer == "Medicaid")
                     + rnorm(N, 0, 0.2))
readmitted_em <- rbinom(N, 1, logistic(readmit_em_logit))

df <- tibble(

  age           = age,
  age_std       = age_std,
  sex           = sex,
  payer         = relevel(factor(payer), ref = "Private"),
  los_days      = los_days,
  diabetes      = diabetes,
  hf            = hf,
  readmitted    = readmitted,
  readmitted_em = readmitted_em
) |>
  mutate(
    age_group = cut(age, breaks = c(17, 44, 64, 74, 95),

2. What is Confounding?

A confounder is a variable that is associated with both the exposure and the outcome, but is not on the causal pathway between them.

        Age
       ↙    ↘
Diabetes → Readmission

Age → more diabetes (older patients have more diabetes)
Age → more readmission (older patients readmit more)
Age is NOT caused by diabetes

Effect: crude OR for diabetes → readmission is inflated because age is “lurking” in the background.

# ── Demonstrate confounding: crude vs adjusted estimate ──────────────────────
crude_or <- function(exposed, outcome) {
  a <- sum(exposed == 1 & outcome == 1)
  b <- sum(exposed == 1 & outcome == 0)
  c <- sum(exposed == 0 & outcome == 1)
  d <- sum(exposed == 0 & outcome == 0)
  OR <- if (b * c > 0) (a * d) / (b * c) else NA_real_
  se <- sqrt(1 / a + 1 / b + 1 / c + 1 / d)
  c(OR, exp(log(OR) - 1.96 * se), exp(log(OR) + 1.96 * se))
}

# Crude
cr <- crude_or(df$diabetes, df$readmitted)

# Adjusted (logistic regression)
model_adj <- glm(
  readmitted ~ diabetes + age_std + sex + payer + hf + los_days,
  data = df, family = binomial()
)
tid_dm <- broom::tidy(model_adj, conf.int = TRUE, exponentiate = TRUE) |>
  filter(term == "diabetes")
adj_or_val <- tid_dm$estimate
adj_lo <- tid_dm$conf.low
adj_hi <- tid_dm$conf.high

cat("Diabetes → 30-day readmission\n")

4. Effect Modification (Interaction)

# Test whether HF MODIFIES the effect of diabetes on readmission
# (as opposed to HF merely being a confounder)

# Model without interaction
model_no_int <- glm(
  readmitted_em ~ diabetes + hf + age_std + payer,
  data = df, family = binomial()
)

# Model with interaction
model_int <- glm(
  readmitted_em ~ diabetes * hf + age_std + payer,
  data = df, family = binomial()
)

# Likelihood ratio test for interaction term
lr <- anova(model_no_int, model_int, test = "LRT")
lr_stat <- lr$Deviance[2]
lr_p    <- lr$`Pr(>Chi)`[2]

cat("Likelihood ratio test for diabetes × HF interaction:\n")
Likelihood ratio test for diabetes × HF interaction:
cat(sprintf("  LR statistic = %.3f, df=1, p = %.4f\n\n", lr_stat, lr_p))
  LR statistic = 18.258, df=1, p = 0.0000
if (!is.na(lr_p) && lr_p < 0.05) {
  cat("  ✓ Significant interaction — effect modification is present\n")
  cat("  → Report STRATUM-SPECIFIC effects, not a single pooled estimate\n")
} else {
  cat("  ✗ No significant interaction — no effect modification\n")
  cat("  → A single adjusted estimate is appropriate\n")
}

  → Report STRATUM-SPECIFIC effects, not a single pooled estimate

5. The 10% Change Rule for Confounding

cat("Systematic 10% change rule across all predictors\n\n")
Systematic 10% change rule across all predictors
cat(sprintf("%-20s %10s %10s %10s %12s\n",
            "Predictor", "Crude OR", "Adj OR", "% Change", "Confounder?"))
Predictor              Crude OR     Adj OR   % Change  Confounder?
cat(paste(rep("-", 65), collapse = ""), "\n")
----------------------------------------------------------------- 
outcome_col <- "readmitted"
for (pair in list(
  c("diabetes", "Diabetes"),
  c("hf", "Heart failure"),
  c("age_std", "Age (std)")
)) {
  exposure <- pair[[1]]; label <- pair[[2]]
  # Crude
  if (exposure == "age_std") {
    m_crude <- glm(as.formula(paste(outcome_col, "~", exposure)), data = df, family = binomial())
    cr <- exp(coef(m_crude)[[exposure]])
  } else {
    a <- sum(df[[exposure]] == 1 & df[[outcome_col]] == 1)
    b <- sum(df[[exposure]] == 1 & df[[outcome_col]] == 0)
    c <- sum(df[[exposure]] == 0 & df[[outcome_col]] == 1)
    d <- sum(df[[exposure]] == 0 & df[[outcome_col]] == 0)
    cr <- if (b * c > 0) (a * d) / (b * c) else NA_real_
  }

  # Adjusted (controlling for the other two)
  covariates <- setdiff(c("diabetes", "hf", "age_std"), exposure)
  formula_a  <- as.formula(paste(outcome_col, "~", exposure, "+", paste(covariates, collapse = " + ")))
  m_adj <- glm(formula_a, data = df, family = binomial())
  adj   <- exp(coef(m_adj)[[exposure]])

  pct <- (adj - cr) / cr * 100
  is_conf <- if (abs(pct) > 10) "✗ Yes (>10%)" else "✓ No (<10%)"
  cat(sprintf("%-20s %10.3f %10.3f %+9.1f%% %12s\n", label, cr, adj, pct, is_conf))
}

6. E-Value: Sensitivity to Unmeasured Confounding

e_value <- function(RR) {
  # E-value: minimum strength of association an unmeasured confounder would need
  # to fully explain an observed RR, on both the exposure-confounder and
  # confounder-outcome pathways.
  # VanderWeele & Ding (2017) formula.
  if (RR < 1) RR <- 1 / RR   # flip if protective
  RR + sqrt(RR * (RR - 1))
}

# Convert OR to approximate RR using prevalence correction
outcome_prev <- mean(df$readmitted)
or_to_rr <- function(OR, prev) {
  # Zhang & Yu (1998) approximation.
  OR / ((1 - prev) + prev * OR)
}

cat("E-value analysis — sensitivity to unmeasured confounding\n")
E-value analysis — sensitivity to unmeasured confounding
cat(paste(rep("=", 65), collapse = ""), "\n")
================================================================= 
hf_or <- exp(coef(model_adj)[["hf"]])
for (pair in list(
  list("Diabetes → readmission (adj OR)", adj_or_val),
  list("HF → readmission", hf_or)
)) {
  label <- pair[[1]]; OR_val <- pair[[2]]
  RR_approx <- or_to_rr(OR_val, outcome_prev)
  ev <- e_value(RR_approx)
  ev_ci <- e_value(max(1.0, RR_approx / exp(1.96 * 0.15)))  # rough CI lower
  cat(sprintf("\n%s\n", label))
  cat(sprintf("  Adjusted OR ≈ RR  : %.3f\n", RR_approx))
  cat(sprintf("  E-value (point)   : %.2f\n", ev))
  cat(sprintf("  E-value (CI lower): %.2f\n", ev_ci))
  cat("  → An unmeasured confounder would need to be associated with BOTH\n")
  cat(sprintf("    exposure AND outcome by a factor of at least %.1f× to explain away\n", ev))
  cat("    the observed association entirely.\n")
  if (ev > 2.0) {
    cat("  → E-value > 2 suggests robustness to weak confounding.\n")
  } else {
    cat("  → E-value ≤ 2 — association could be explained by a moderate confounder.\n")
  }
}

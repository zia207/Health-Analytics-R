Module 05 — Clinical Text Preprocessing

Word Embeddings & Word2Vec

tokenize_note <- function(text) {
  text <- tolower(text)
  text <- stringr::str_replace_all(text, "[^a-z0-9\\s-]", " ")
  stringr::str_split(stringr::str_squish(text), "\\s+")[[1]]
}

3. Train Clinical Word2Vec

text2vec trains GloVe embeddings from a skip-gram co-occurrence matrix — the standard R analogue of gensim Word2Vec. Skip-gram window = 5 matches the Python demo.

train_glove <- function(token_list, window = 5L, rank = 30L, n_iter = 15L,
                        min_count = 2L, x_max = 100, learning_rate = 0.01) {
  it <- text2vec::itoken(token_list, progressbar = FALSE)
  vocab <- text2vec::create_vocabulary(it) |>
    text2vec::prune_vocabulary(term_count_min = min_count)
  vectorizer <- text2vec::vocab_vectorizer(vocab)
  it2 <- text2vec::itoken(token_list, progressbar = FALSE)
  tcm <- text2vec::create_tcm(it2, vectorizer, skip_grams_window = window)
  rank_use <- max(2L, min(rank, nrow(tcm) - 1L))
  tryCatch({
    glove <- text2vec::GlobalVectors$new(rank = rank_use, x_max = x_max)
    wv_main <- glove$fit_transform(
      tcm, n_iter = n_iter, convergence_tol = 0.05,
      n_threads = 1, learning_rate = learning_rate
    )
    wv_main + t(glove$components)
  }, error = function(e) {
    cat("GloVe optimiser unstable; using SVD of the co-occurrence matrix.\n")
    mat <- as.matrix(tcm)
    mat[!is.finite(mat)] <- 0
    s <- svd(mat, nu = rank_use, nv = rank_use)
    out <- s$u %*% diag(sqrt(pmax(s$d[seq_len(rank_use)], 0)), rank_use, rank_use)
    rownames(out) <- rownames(tcm)
    out
  })
}

cosine_sim <- function(a, b) {
  sum(a * b) / (sqrt(sum(a^2)) * sqrt(sum(b^2)) + 1e-12)
}

most_similar <- function(word_vectors, word, topn = 4) {
  if (!word %in% rownames(word_vectors)) return(NULL)
  target <- word_vectors[word, ]
  sims <- apply(word_vectors, 1, cosine_sim, b = target)
  sims <- sims[names(sims) != word]
  sims <- sort(sims, decreasing = TRUE)
  head(sims, topn)
}

4. Clinical Analogy Tasks

analogy <- function(word_vectors, a, b, c, topn = 3) {
  vec <- word_vectors[a, ] - word_vectors[b, ] + word_vectors[c, ]
  sims <- apply(word_vectors, 1, cosine_sim, b = vec)
  sims <- sims[!names(sims) %in% c(a, b, c)]
  head(sort(sims, decreasing = TRUE), topn)
}

if (nrow(wv) > 50) {
  analogy_tests <- list(
    c("diabetes", "insulin", "infection"),
    c("heart", "failure", "kidney"),
    c("acute", "kidney", "heart"),
    c("septic", "shock", "cardiogenic")
  )
  cat("Analogy tasks (A - B + C ≈ ?):\n\n")
  for (triple in analogy_tests) {
    A <- triple[1]; B <- triple[2]; C <- triple[3]
    if (all(c(A, B, C) %in% rownames(wv))) {

doc_data <- purrr::imap_dfr(condition_templates, function(templates, cond) {
  purrr::map_dfr(seq_len(80), function(i) {
    base <- tokenize_note(sample(templates, 1))
    noise <- sample(c("patient", "treatment", "management", "hospital"), 2, replace = TRUE)
    tokens <- sample(c(base, noise))
    tibble::tibble(tokens = list(tokens), condition = cond)
  })
})

get_doc_vector <- function(tokens, word_vectors, dim = 100) {
  keep <- tokens[tokens %in% rownames(word_vectors)]
  if (length(keep) == 0) return(rep(0, dim))
  colMeans(word_vectors[keep, , drop = FALSE])
}

X_w2v <- t(vapply(doc_data$tokens, get_doc_vector,
                  numeric(ncol(wv)), word_vectors = wv, dim = ncol(wv)))
colnames(X_w2v) <- paste0("e", seq_len(ncol(X_w2v)))
y_docs <- factor(doc_data$condition)

set.seed(42)
split <- rsample::initial_split(
  tibble::tibble(y = y_docs, id = seq_len(nrow(X_w2v))),
  prop = 0.8, strata = y
)
tr_idx <- rsample::training(split)$id
te_idx <- rsample::testing(split)$id

tr_df <- data.frame(y = y_docs[tr_idx], X_w2v[tr_idx, , drop = FALSE], check.names = FALSE)
te_df <- data.frame(X_w2v[te_idx, , drop = FALSE], check.names = FALSE)
fit_lr <- nnet::multinom(y ~ ., data = tr_df, trace = FALSE, decay = 0.1, maxit = 200)
pred <- predict(fit_lr, newdata = te_df)
acc <- mean(pred == as.character(y_docs[te_idx]))
cat(sprintf("Word2Vec doc-vec accuracy: %.4f\n", acc))

7. Skip-gram vs CBOW Comparison

Skip-gram (window = 5, predicting context from the target) vs a CBOW-like narrower window (window = 2). Both are trained as GloVe on skip-gram co-occurrence matrices — the text2vec analogue of gensim’s sg=1 vs sg=0.

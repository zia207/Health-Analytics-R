3. ICD-10-CM Code Structure

ICD-10-CM (International Classification of Diseases, 10th Revision, Clinical Modification) is the standard for diagnosis coding in US healthcare.

E   1 1 . 9
│   │ │   └── 5th-7th characters: clinical detail / laterality / episode
│   │ └────── 4th character: subcategory / anatomic site
│   └──────── 2nd-3rd characters: condition / etiology
└──────────── 1st character: chapter letter (A–Z)

# --- 3.2  ICD-10-CM parser ---
parse_icd10cm <- function(code) {
  # Parse an ICD-10-CM code into its structural components.
  # Returns a named list with chapter, category, subcategory, detail, and formatted code.
  code <- toupper(gsub("\\.", "", trimws(as.character(code))))
  if (!grepl("^[A-Z][0-9]{2}", code)) {
    return(list(
      valid = FALSE, formatted = NA_character_, chapter = NA_character_,
      category = NA_character_, subcategory = NA_character_,
      detail = NA_character_, specificity = NA_character_, raw = code
    ))
  }

  chapter     <- unname(icd10_chapters[substr(code, 1, 1)])
  if (is.na(chapter)) chapter <- "Unknown"
  category    <- substr(code, 1, 3)
  subcategory <- if (nchar(code) > 3) substr(code, 4, 4) else ""
  detail      <- if (nchar(code) > 4) substr(code, 5, nchar(code)) else ""
  formatted   <- paste0(category, if (nchar(code) > 3) paste0(".", substr(code, 4, nchar(code))) else "")
  spec_idx    <- min(nchar(code) - 3, 4) + 1
  specificity <- c("3-char", "4-char", "5-char", "6-char", "7-char")[spec_idx]

  list(
    valid       = TRUE,
    formatted   = formatted,
    chapter     = chapter,
    category    = category,
    subcategory = subcategory,
    detail      = detail,
    specificity = specificity,
    raw         = code
  )
}

# --- 4.2  CPT validator ---
validate_cpt <- function(code) {
  # Check if a CPT code is a valid 5-digit numeric string.
  code <- trimws(as.character(code))
  is_numeric_5 <- grepl("^[0-9]{5}$", code)
  code_int <- if (is_numeric_5) as.integer(code) else NA_integer_

  if (!is_numeric_5) {
    return(list(valid = FALSE, code = code, section = "Unknown", notes = "Not 5 digits"))
  }

  section <- "Unknown"
  if      (code_int >= 100   && code_int <= 1999)  section <- "Anaesthesia"
  else if (code_int >= 10004 && code_int <= 69990) section <- "Surgery"
  else if (code_int >= 70010 && code_int <= 79999) section <- "Radiology"
  else if (code_int >= 80047 && code_int <= 89398) section <- "Pathology/Lab"
  else if (code_int >= 90281 && code_int <= 99607) section <- "Medicine/E&M"

  list(valid = TRUE, code = code, section = section, notes = "")
}

for (cpt in c("99213", "71046", "27447", "00100", "ABCDE", "9921")) {

  status <- if (isTRUE(r$valid)) "VALID" else "INVALID"
  cat(sprintf("  %-8s → %-8s | section: %s\n", cpt, status, r$section))
}

5. LOINC Code Structure (Lab & Clinical Observations)

LOINC (Logical Observation Identifiers Names and Codes) standardises laboratory and clinical observations. Each code identifies a specific measurement.

Structure: NNNNN-C (5–6 digit numeric) + (1 digit check digit)

LOINC names follow a 6-part axis: 1. Component — what was measured (e.g. Glucose) 2. Property — type of quantity (e.g. MCnc = mass concentration) 3. Time aspect — Pt (point in time) vs 24H (24-hour) 4. System — specimen (e.g. Ser/Plas = serum/plasma) 5. Scale — Qn (quantitative), Ord (ordinal), Nom (nominal) 6. Method — analytical method (if relevant)

# --- 5.2  Simulate lab results and flag abnormals ---
set.seed(7)
n_patients <- 200

# For each patient, generate realistic lab values
sim_lab <- function(lo, hi, n) {
  mid  <- (lo + hi) / 2
  span <- (hi - lo) / 2
  vals <- rnorm(n, mean = mid, sd = span * 0.6)
  round(pmin(pmax(vals, lo * 0.4), hi * 2.5), 2)
}

lab_records <- purrr::pmap_dfr(loinc_ref, function(loinc, component, property, system,
                                                   unit, normal_low, normal_high) {
  vals <- sim_lab(normal_low, normal_high, n_patients)
  tibble(
    patient_id  = sprintf("PT-%05d", seq_len(n_patients)),
    loinc       = loinc,
    component   = component,
    value       = vals,
    unit        = unit,
    flag        = ifelse(vals > normal_high, "H", ifelse(vals < normal_low, "L", "N")),
    normal_low  = normal_low,
    normal_high = normal_high
  )
})

6. FHIR-Like JSON Parsing

# FHIR (Fast Healthcare Interoperability Resources) is the modern HL7 standard
# for exchanging health data. Resources are JSON objects. Here we simulate
# a small FHIR Bundle containing Patient + Observation resources.

make_fhir_bundle <- function(n = 5) {
  # Generate a minimal FHIR-like bundle for n patients.
  loinc_sample <- loinc_ref %>% select(loinc, component, unit) %>% head(3)
  entries <- list()

  for (i in seq_len(n)) {
    pt_id <- paste0("patient-", i)
    entries[[length(entries) + 1]] <- list(
      resourceType = "Patient",
      id           = pt_id,
      gender       = sample(c("male", "female"), 1),
      birthDate    = sprintf("%d-%02d-15", sample(1940:1984, 1), sample(1:12, 1)),
      address      = list(list(postalCode = as.character(sample(10000:99999, 1))))
    )
    for (j in seq_len(nrow(loinc_sample))) {
      lab <- loinc_sample[j, ]
      lo  <- loinc_ref$normal_low[loinc_ref$loinc == lab$loinc]
      hi  <- loinc_ref$normal_high[loinc_ref$loinc == lab$loinc]
      val <- round(runif(1, lo, hi), 2)
      entries[[length(entries) + 1]] <- list(
        resourceType  = "Observation",
        id            = paste0("obs-", pt_id, "-", lab$loinc),
        subject       = list(reference = paste0("Patient/", pt_id)),
        code          = list(coding = list(list(
          system  = "http://loinc.org",
          code    = lab$loinc,
          display = lab$component
        ))),
        valueQuantity = list(value = val, unit = lab$unit),
        status        = "final"
      )
    }
  }
  list(resourceType = "Bundle", type = "collection", entry = entries)
}

safe_harbour_check <- function(df) {
  # Flag columns that may contain PHI based on name heuristics.
  phi_keywords <- c("name", "phone", "fax", "email", "ssn", "mrn", "dob", "birth",
                    "address", "zip", "ip", "device", "url", "id_", "npi")
  flagged <- character()
  for (col in names(df)) {
    if (any(vapply(phi_keywords, function(kw) grepl(kw, tolower(col), fixed = TRUE), logical(1)))) {
      flagged <- c(flagged, col)
    }
  }
  flagged
}

# De-identify: drop PHI, generalise ZIP to 3-digit
deidentify <- function(df, phi_cols, zip_col = NULL) {
  keep <- setdiff(names(df), intersect(phi_cols, names(df)))
  df_clean <- df[keep]
  if (!is.null(zip_col) && zip_col %in% names(df_clean)) {
    # HIPAA: 3-digit ZIP if population > 20,000; else replace with 000
    df_clean[[zip_col]] <- paste0(substr(df_clean[[zip_col]], 1, 3), "XX")
  }
  df_clean
}

8. Code Quality Validation Pipeline

# Build a reusable validation function that checks ICD, CPT, and LOINC
# codes in a data frame and produces a quality report.

validate_code_column <- function(series, code_type) {
  # Validate a column of clinical codes. code_type: 'icd10', 'cpt', or 'loinc'
  purrr::map_dfr(series, function(code) {
    if (is.na(code) || identical(code, "")) {
      return(tibble(code = as.character(code), valid = FALSE, reason = "Missing/empty"))
    }
    code <- trimws(as.character(code))
    if (code_type == "icd10") {
      r <- parse_icd10cm(code)
      tibble(code = code, valid = isTRUE(r$valid),
             reason = if (isTRUE(r$valid)) "" else "Invalid ICD-10 format")
    } else if (code_type == "cpt") {
      r <- validate_cpt(code)
      tibble(code = code, valid = isTRUE(r$valid), reason = r$notes)
    } else if (code_type == "loinc") {
      valid <- grepl("^[0-9]{2,5}-[0-9]$", code)
      tibble(code = code, valid = valid,
             reason = if (valid) "" else "Invalid LOINC format (expected NNNNN-C)")
    } else {
      tibble(code = code, valid = FALSE, reason = "Unknown code_type")
    }
  })
}

Module 02 — Data Loading & Automated Profiling

Data Loading & Automated Profiling

Learning objectives - Load health data from CSV, Excel, and Parquet with production-grade options - Generate an automated profiling report with DataExplorer and skimr - Interpret profiling output sections: overview, variables, correlations, missing values - Customise and export the report for sharing with clinical collaborators

Estimated time: 1 hour | Level: Beginner | Libraries: dplyr / tibble, DataExplorer, skimr, readxl, arrow

2. Synthetic Hospital Discharge Dataset

set.seed(42)
N <- 800

ages      <- as.integer(pmin(pmax(rnorm(N, 62, 16), 18), 95))
sexes     <- sample(c("M", "F"), N, replace = TRUE, prob = c(0.48, 0.52))
payers    <- sample(c("Medicare", "Medicaid", "Private", "Self-pay", "Other"),
                    N, replace = TRUE, prob = c(0.40, 0.22, 0.28, 0.07, 0.03))
drg_codes <- sample(c(470, 291, 392, 690, 871, 193, 247, 603), N, replace = TRUE)
drg_labels_map <- c(
  `470` = "Major joint replacement", `291` = "Heart failure & shock",
  `392` = "Esophagitis/misc GI",     `690` = "Kidney/UTI",
  `871` = "Septicemia",              `193` = "Simple pneumonia",
  `247` = "Perc cardiovasc w stent", `603` = "Cellulitis"
)
los_days   <- as.integer(pmin(pmax(rgamma(N, shape = 2.5, scale = 1.8), 1), 30))
charges    <- round(los_days * runif(N, 1800, 4200), 2)
charges[runif(N) < 0.08] <- NA_real_
primary_dx <- sample(c("E11.9", "I10", "N18.3", "I50.9", "J18.9", "A41.9", "M16.11", "N39.0"),
                     N, replace = TRUE)
admit_type <- sample(c("Emergency", "Elective", "Urgent", "Newborn"),
                     N, replace = TRUE, prob = c(0.52, 0.30, 0.16, 0.02))
set.seed(99)
df <- tibble(
  patient_id    = sprintf("PT-%05d", seq_len(N)),
  age           = ages,
  sex           = sexes,
  payer         = payers,
  drg_code      = drg_codes,
  drg_label     = unname(drg_labels_map[as.character(drg_codes)]),
  primary_dx    = primary_dx,
  admit_type    = admit_type,
  los_days      = los_days,
  total_charge  = charges,
  diabetes      = rbinom(N, 1, 0.32),
  hypertension  = rbinom(N, 1, 0.45),
  ckd           = rbinom(N, 1, 0.15),
  heart_failure = rbinom(N, 1, 0.18),
  readmitted_30 = rbinom(N, 1, 0.14)
)
bins <- c(17, 44, 64, 74, 95)
df$age_group <- cut(df$age, breaks = bins, labels = c("18-44", "45-64", "65-74", "75+"))

start <- as.Date("2019-01-01")
set.seed(5)
n_days <- as.integer(as.Date("2023-12-31") - start)
df$admit_date     <- start + sample(0:(n_days - 1), N, replace = TRUE)
df$discharge_date <- df$admit_date + df$los_days
df <- df %>% arrange(admit_date)
cat(sprintf("Dataset: %d rows × %d columns\n", nrow(df), ncol(df)))

6. Interpreting the Profile Report

The report is divided into five sections. Here is what to look for in each.

Section What to check
Overview Number of variables, rows, missing cells, duplicate rows
Variables Distribution shape, % missing, distinct count, warnings (high cardinality, skewed, zeros)
Interactions Bivariate scatter plots — look for clusters or non-linear relationships
Correlations Pearson (linear), Spearman (monotonic) — flag high correlations (>0.8)
Missing values Bar chart + matrix — check if missingness clusters on specific patients or time periods
Sample First/last rows — sanity check on values and date formatting

=== Variable-level warnings ===
n <- nrow(df)
for (col in names(df)) {
  x <- df[[col]]
  n_miss <- sum(is.na(x))
  n_uniq <- n_distinct(x)
  alerts <- character(0)
  if (n_uniq == n) alerts <- c(alerts, "HIGH CARDINALITY")
  if (n_miss / n > 0.05) alerts <- c(alerts, "MISSING")
  if (is.numeric(x) && sum(x == 0, na.rm = TRUE) / n > 0.20) alerts <- c(alerts, "ZEROS")
  if (is.numeric(x) && sd(x, na.rm = TRUE) > 0) {
    sk <- mean((x - mean(x, na.rm = TRUE))^3, na.rm = TRUE) / sd(x, na.rm = TRUE)^3
    if (abs(sk) > 1) alerts <- c(alerts, "SKEWED")
  }
  if (length(alerts)) {
    for (a in alerts) cat(sprintf("  [%-20s] %s\n", a, col))
  }
}

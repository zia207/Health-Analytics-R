2. TextRank Extractive Summarisation

Sentence scoring with tidytext / quanteda TF-IDF and a PageRank-style iteration (TextRank analogue). lexRankr implements the same algorithm if you prefer a package wrapper.

stops <- c(quanteda::stopwords("en"),
           "patient", "mg", "daily", "was", "were", "the", "a", "an",
           "admission", "discharge", "history", "hospital", "noted")

split_sentences <- function(text) {
  sents <- unlist(quanteda::tokens(text, what = "sentence"))
  sents <- stringr::str_squish(sents)
  n_words <- vapply(strsplit(sents, "\\s+"), length, integer(1))
  sents[n_words > 5]
}

sentence_tfidf <- function(sents) {
  toks <- quanteda::tokens(sents, remove_punct = TRUE) |>
    quanteda::tokens_tolower() |>
    quanteda::tokens_remove(stops)
  quanteda::dfm_tfidf(quanteda::dfm(toks))
}

cosine_sim_mat <- function(mat) {
  nrms <- sqrt(rowSums(mat^2)); nrms[nrms == 0] <- 1
  Xn <- mat / nrms
  sim <- tcrossprod(Xn)
  diag(sim) <- 0
  sim
}

textrank_summarise <- function(text, n_sents = 5) {
  sents <- split_sentences(text)
  if (length(sents) <= n_sents) {
    return(list(sents = sents, scores = rep(1, length(sents))))
  }
  d <- sentence_tfidf(sents)
  mat <- as.matrix(d)
  if (nrow(mat) == 0 || ncol(mat) == 0) {
    return(list(sents = sents[seq_len(n_sents)], scores = rep(1, n_sents)))
  }
  sim <- cosine_sim_mat(mat)
  n <- nrow(sim)
  scores <- rep(1 / n, n)
  for (iter in seq_len(50)) {
    rs <- rowSums(sim); rs[rs == 0] <- 1
    trans <- t(sim / rs)
    scores <- 0.15 / n + 0.85 * as.numeric(trans %*% scores)
  }
  top_idx <- sort(order(scores, decreasing = TRUE)[seq_len(n_sents)])
  list(sents = sents[top_idx], scores = scores)
}

3. LSA-Based Summarisation

lsa_summarise <- function(text, n_sents = 5, n_comp = 4) {
  sents <- split_sentences(text)
  if (length(sents) <= n_sents) return(sents)
  d <- sentence_tfidf(sents)
  X <- as.matrix(d)
  n_comp <- min(n_comp, nrow(X) - 1, ncol(X) - 1)
  if (n_comp < 1) return(sents[seq_len(n_sents)])
  svd_fit <- svd(X, nu = n_comp, nv = n_comp)
  X_r <- svd_fit$u %*% diag(svd_fit$d[seq_len(n_comp)], n_comp, n_comp)
  scs <- sqrt(rowSums(X_r^2))
  top_idx <- sort(order(scs, decreasing = TRUE)[seq_len(n_sents)])
  sents[top_idx]
}

4. Section-Aware Summarisation

section_weights <- c(
  "DISCHARGE DIAGNOSES"   = 3.0,
  "DISCHARGE MEDICATIONS" = 2.5,
  "HOSPITAL COURSE"       = 2.0,
  "FOLLOW-UP"             = 1.5,
  "LABS"                  = 1.0,
  "ADMITTING DIAGNOSIS"   = 1.0,
  "DISCHARGE CONDITION"   = 1.5
)

get_section <- function(pos, text) {
  best <- "GENERAL"
  best_pos <- -1
  for (header in names(section_weights)) {
    hp <- regexpr(header, toupper(text), fixed = TRUE)[1]
    if (hp != -1 && hp < pos && hp > best_pos) {
      best <- header
      best_pos <- hp
    }
  }
  best
}

section_aware_summarise <- function(text, n_sents = 6) {
  sents <- split_sentences(text)
  if (length(sents) <= n_sents) {
    return(list(sents = sents, secs = rep("GENERAL", length(sents))))
  }
  d <- sentence_tfidf(sents)
  ts <- as.numeric(rowSums(as.matrix(d)))
  weighted <- numeric(length(sents))
  secs <- character(length(sents))
  for (i in seq_along(sents)) {
    pos <- regexpr(sents[i], text, fixed = TRUE)[1]
    if (pos == -1) pos <- 1
    sec <- get_section(pos, text)
    w <- if (sec %in% names(section_weights)) section_weights[[sec]] else 0.5
    weighted[i] <- ts[i] * w
    secs[i] <- sec
  }
  top_idx <- sort(order(weighted, decreasing = TRUE)[seq_len(n_sents)])
  list(sents = sents[top_idx], secs = secs[top_idx])
}

5. ROUGE Evaluation

get_ngrams <- function(text, n) {
  toks <- stringr::str_extract_all(tolower(text), "\\b\\w+\\b")[[1]]
  if (length(toks) < n) return(character())
  vapply(seq_len(length(toks) - n + 1), function(i) {
    paste(toks[i:(i + n - 1)], collapse = " ")
  }, character(1))
}

rouge_n <- function(hyp, ref, n = 1) {
  h <- get_ngrams(hyp, n)
  r <- get_ngrams(ref, n)
  if (length(r) == 0) return(list(p = 0, r = 0, f1 = 0))
  hc <- table(h); rc <- table(r)
  ov <- sum(purrr::map_dbl(names(rc), function(ng) {
    min(if (ng %in% names(hc)) as.numeric(hc[[ng]]) else 0, as.numeric(rc[[ng]]))
  }))
  p  <- if (length(h) > 0) ov / length(h) else 0
  re <- ov / length(r)
  f1 <- if ((p + re) > 0) 2 * p * re / (p + re) else 0
  list(p = round(p, 4), r = round(re, 4), f1 = round(f1, 4))
}

rouge_l <- function(hyp, ref) {
  h <- stringr::str_extract_all(tolower(hyp), "\\b\\w+\\b")[[1]]
  r <- stringr::str_extract_all(tolower(ref), "\\b\\w+\\b")[[1]]
  m <- length(h); n <- length(r)
  dp <- matrix(0, m + 1, n + 1)
  if (m > 0 && n > 0) {
    for (i in seq_len(m)) {
      for (j in seq_len(n)) {
        dp[i + 1, j + 1] <- if (h[i] == r[j]) dp[i, j] + 1 else max(dp[i, j + 1], dp[i + 1, j])
      }
    }
  }
  lcs <- dp[m + 1, n + 1]
  p  <- if (m > 0) lcs / m else 0
  re <- if (n > 0) lcs / n else 0
  f1 <- if ((p + re) > 0) 2 * p * re / (p + re) else 0
  list(p = round(p, 4), r = round(re, 4), f1 = round(f1, 4))
}

7. Zero-Shot TF-IDF Triage Classifier

Keyword / label similarity analogue of zero-shot NLI: cosine similarity between the note and triage-class descriptions.

triage_descs <- c(
  IMMEDIATE   = paste("cardiac arrest respiratory failure unresponsive SpO2 below 85 BP below 70",
                      "active hemorrhage STEMI severe respiratory distress cyanosis anaphylaxis"),
  URGENT      = paste("chest pain active angina dyspnea rest SpO2 92 altered mental status",
                      "new neurological deficit uncontrolled pain severe hypertension fever sepsis"),
  SEMI_URGENT = paste("moderate pain stable vital signs mild dyspnea worsening chronic condition",
                      "urinary symptoms mild infection stable arrhythmia elevated glucose"),
  ROUTINE     = paste("prescription refill follow-up chronic stable lab results review",
                      "medication adjustment preventive care no acute distress")
)

zero_shot_triage <- function(text) {
  all_texts <- c(unname(triage_descs), text)
  toks <- quanteda::tokens(all_texts, remove_punct = TRUE) |>
    quanteda::tokens_tolower() |>
    quanteda::tokens_ngrams(n = 1:2)
  d <- quanteda::dfm_tfidf(quanteda::dfm(toks))
  mat <- as.matrix(d)
  nrms <- sqrt(rowSums(mat^2)); nrms[nrms == 0] <- 1
  Xn <- mat / nrms
  scores <- as.numeric(Xn[nrow(Xn), ] %*% t(Xn[-nrow(Xn), , drop = FALSE]))
  names(scores) <- names(triage_descs)
  sort(scores, decreasing = TRUE)
}

8. Knowledge Check

  1. TextRank ROUGE-1 F1=0.42. What baseline would you compare against (e.g. lead-1)?
  2. The section-aware method weighted DISCHARGE DIAGNOSES at 3.0. What justifies this?
  3. Zero-shot triage classified chest pain as SEMI-URGENT. What safety implications?
  4. Abstractive LLM summarisation can hallucinate clinical facts. Describe two validation strategies.
  5. Implement a lead-1 baseline: use the first sentence of each major section.
lead_baseline <- function(text) {
  headers <- c("ADMITTING DIAGNOSIS", "HOSPITAL COURSE", "DISCHARGE DIAGNOSES",
               "DISCHARGE MEDICATIONS", "FOLLOW-UP", "DISCHARGE CONDITION")
  sents <- character()
  for (h in headers) {
    pos <- regexpr(h, toupper(text), fixed = TRUE)[1]
    if (pos == -1) next
    rest <- substr(text, pos, min(nchar(text), pos + 299))
    for (line in strsplit(rest, "\\.")[[1]]) {
      line <- stringr::str_squish(line)
      if (length(strsplit(line, "\\s+")[[1]]) > 5) {
        sents <- c(sents, paste0(line, "."))
        break
      }
    }
  }
  sents
}

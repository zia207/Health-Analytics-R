Module 08 - Reproducible Research & Deployment

Model Cards & Clinical AI Governance

Learning objectives - Build a comprehensive Model Card following Mitchell et al. (2019) standards - Conduct fairness analysis across demographic subgroups (sex, race, payer) - Apply the EU AI Act and FDA AI/ML framework to clinical models - Build a clinical AI governance checklist - Implement ongoing model monitoring for bias and drift - Write clinical staff education materials for AI model use

Estimated time: 2.5 hours | Level: Intermediate | Libraries: tidymodels, ggplot2, gt, jsonlite

set.seed(42)
N <- 3000
sigmoid <- function(x) 1 / (1 + exp(-x))
age  <- as.integer(pmin(pmax(rnorm(N, 60, 15), 18), 90))
sex  <- sample(c("M", "F"), N, replace = TRUE, prob = c(0.48, 0.52))
race <- sample(c("White", "Black", "Hispanic", "Asian", "Other"), N, replace = TRUE,
               prob = c(0.60, 0.18, 0.12, 0.06, 0.04))
payer <- sample(c("Medicare", "Medicaid", "Private", "Self-pay"), N, replace = TRUE,
                prob = c(0.40, 0.22, 0.28, 0.10))
los  <- as.integer(pmin(pmax(rgamma(N, 2.5, scale = 1.8), 1), 30))
dm   <- rbinom(N, 1, 0.35)
hf   <- rbinom(N, 1, 0.22)
copd <- rbinom(N, 1, 0.18)
logit <- -3.2 + 0.6 * hf + 0.5 * dm + 0.3 * copd + 0.02 * (age - 60) +
  0.2 * as.integer(los > 7)
readmitted <- rbinom(N, 1, sigmoid(logit))
df <- tibble(age = age, sex = sex, race = race, payer = payer,
             los_days = los, diabetes = dm, hf = hf, copd = copd, readmitted = readmitted)
FEATURES <- c("age", "los_days", "diabetes", "hf", "copd")
split <- initial_split(df, prop = 0.8, strata = readmitted)
train <- training(split)
test  <- testing(split)
df_te <- test
model <- xgb.train(
  params = list(objective = "binary:logistic", max_depth = 4, eta = 0.05,
                eval_metric = "auc"),
  data = xgb.DMatrix(as.matrix(train[FEATURES]), label = train$readmitted),
  nrounds = 200, verbose = 0
)
df_te$pred_prob <- predict(model, as.matrix(df_te[FEATURES]))
y_te <- df_te$readmitted
cat(sprintf("Dataset: %d | AUC=%.4f\n", N,
            as.numeric(pROC::auc(y_te, df_te$pred_prob, quiet = TRUE))))

2. Why Model Cards Matter in Healthcare

A Model Card (Mitchell et al., 2019) is a short document accompanying a trained ML model that: - Specifies the model’s intended use cases and out-of-scope uses - Documents training data, evaluation data, and known limitations - Reports performance across disaggregated subgroups (sex, race, payer) - Lists ethical considerations for clinical deployment

Regulatory context: - FDA (USA): Pre-Determined Change Control Plans; Software as a Medical Device (SaMD) - EU AI Act: High-risk AI in healthcare requires conformity assessment and documentation - Joint Commission: “Use of predictive analytics” must be monitored and disclosed - ONC HTI-1 Rule: Requires transparency about algorithms used in decision support

3. Model Card Generation

auc_overall <- as.numeric(pROC::auc(y_te, df_te$pred_prob, quiet = TRUE))
ap_overall  <- {
  ord <- order(df_te$pred_prob, decreasing = TRUE)
  y <- y_te[ord]; tp <- cumsum(y); prec <- tp / seq_along(y); rec <- tp / sum(y)
  sum(diff(c(0, rec)) * prec)
}
brier_overall <- mean((df_te$pred_prob - y_te)^2)

4. Fairness Analysis: AUC by Subgroup

fairness_results <- list()
for (spec in list(
  list(col = "sex", groups = c("M", "F")),
  list(col = "race", groups = c("White", "Black", "Hispanic", "Asian")),
  list(col = "payer", groups = c("Medicare", "Medicaid", "Private", "Self-pay"))
)) {
  for (group in spec$groups) {
    mask <- df_te[[spec$col]] == group
    if (sum(mask) < 20) next
    sub_y <- y_te[mask]
    sub_p <- df_te$pred_prob[mask]
    if (n_distinct(sub_y) < 2) next
    auc_g <- as.numeric(pROC::auc(sub_y, sub_p, quiet = TRUE))
    ord <- order(sub_p, decreasing = TRUE)
    yord <- sub_y[ord]; tp <- cumsum(yord); prec <- tp / seq_along(yord); rec <- tp / sum(yord)
    ap_g <- sum(diff(c(0, rec)) * prec)
    fairness_results[[length(fairness_results) + 1]] <- tibble(
      Subgroup = paste0(spec$col, "=", group),
      N = sum(mask),
      `Readmit%` = round(mean(sub_y) * 100, 1),
      AUC = round(auc_g, 4),
      AUC_delta = round(auc_g - auc_overall, 4),
      AP = round(ap_g, 4)
    )
  }
}

# Visualise fairness across subgroups
plot_fair <- function(col, title) {
  rows <- map_dfr(unique(df_te[[col]]), function(group) {
    mask <- df_te[[col]] == group
    if (sum(mask) < 20) return(tibble())
    sub_y <- y_te[mask]
    if (n_distinct(sub_y) < 2) return(tibble())
    tibble(
      group = group, n = sum(mask),
      auc = as.numeric(pROC::auc(sub_y, df_te$pred_prob[mask], quiet = TRUE))
    )
  })
  rows$col_fill <- if_else(abs(rows$auc - auc_overall) > 0.05, col_red, col_blue)
  ggplot(rows, aes(group, auc, fill = col_fill)) +
    geom_col(alpha = 0.85, colour = "white") +
    geom_hline(yintercept = auc_overall, linetype = "dashed") +
    geom_hline(yintercept = auc_overall - 0.05, colour = "orange", linetype = "dotted") +
    geom_text(aes(label = sprintf("%.3f\n(n=%d)", auc, n)), vjust = -0.2, size = 3) +
    scale_fill_identity() +
    coord_cartesian(ylim = c(0.5, 1.0)) +
    labs(title = title, y = "AUC-ROC", x = NULL) +
    theme(axis.text.x = element_text(angle = 15, hjust = 1))
}

fig_fair <- plot_fair("sex", "Performance by Sex") |

spatial_smooth <- function(values, row_idx, col_idx, nrow, ncol, sigma = 2) {
  grid <- matrix(0, nrow = nrow, ncol = ncol)
  for (i in seq_along(values)) grid[row_idx[i], col_idx[i]] <- values[i]
  k <- ceiling(3 * sigma)
  xs <- (-k):k
  kern <- outer(dnorm(xs, sd = sigma), dnorm(xs, sd = sigma))
  kern <- kern / sum(kern)
  pad <- k
  gp <- matrix(0, nrow + 2 * pad, ncol + 2 * pad)
  gp[(pad + 1):(pad + nrow), (pad + 1):(pad + ncol)] <- grid
  out <- matrix(0, nrow, ncol)
  for (r in seq_len(nrow)) for (c in seq_len(ncol)) {
    out[r, c] <- sum(gp[r:(r + 2 * k), c:(c + 2 * k)] * kern)
  }
  mapply(function(r, c) out[r, c], row_idx, col_idx)
}

2. Access Metrics Framework

Primary care access metrics:

Metric Formula Limitation
Provider:population ratio Providers / Population Ignores distance
Distance to nearest min(d_ij) Ignores capacity
2SFCA Sum of weighted provider ratios Most comprehensive
Gravity model Sum(S_j / d^p) Power-law decay
Floating catchment Within d miles of provider Binary, ignores gradients

Two-Step Floating Catchment Area (2SFCA): 1. Step 1: For each provider j, compute R_j = Beds_j / Σ(Pop_i for i within d) 2. Step 2: For each county i, A_i = Σ(R_j for j within d of i) Higher A_i = better access.

two_sfca <- function(county_pop, dist, provider_cap, catchment, decay_power = 1) {
  n_i <- length(county_pop)
  n_j <- length(provider_cap)
  R_j <- numeric(n_j)
  for (j in seq_len(n_j)) {
    in_zone <- dist[, j] <= catchment
    denom <- sum(county_pop[in_zone] / (dist[in_zone, j] + 0.1)^decay_power)
    R_j[j] <- if (denom > 0) provider_cap[j] / denom else 0
  }
  A_i <- numeric(n_i)
  for (i in seq_len(n_i)) {
    in_zone <- dist[i, ] <= catchment
    if (any(in_zone)) {
      weights <- 1 / (dist[i, in_zone] + 0.1)^decay_power
      A_i[i] <- sum(R_j[in_zone] * weights)
    }
  }
  A_i
}

gdf$access_2sfca <- two_sfca(
  df$population, dist_matrix, df_hosp$beds,
  catchment = CATCHMENT_DEG, decay_power = 2
)

concentration_index <- function(health, rank_var) {
  n <- length(health)
  rank <- rank(rank_var) / n
  2 * cov(rank, health) / mean(health)
}

ci_poverty_access <- concentration_index(gdf$access_2sfca, gdf$pct_poverty)

  5. Implement a Gini coefficient for healthcare access to quantify inequality across all counties.
gini_coefficient <- function(values) {
  sorted_v <- sort(values)
  n <- length(sorted_v)
  cumulative <- cumsum(sorted_v)
  total <- cumulative[n]
  lorenz <- c(0, cumulative / total)
  x <- seq(0, 1, length.out = n + 1)
  trapz <- sum(diff(x) * (head(lorenz, -1) + tail(lorenz, -1)) / 2)
  list(gini = 1 - 2 * trapz, lorenz = lorenz, x = x)
}

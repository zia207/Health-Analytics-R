augment <- function(text, noise = 0.15) {
  words <- strsplit(text, "\\s+")[[1]]
  extra <- c("assessment", "plan", "patient", "stable", "monitor", "admitted")
  result <- character()
  for (w in words) {
    result <- c(result, w)
    if (runif(1) < noise) result <- c(result, sample(extra, 1))
  }
  paste(result, collapse = " ")
}

clean_clinical_text <- function(text) {
  text <- tolower(text)
  text <- stringr::str_replace_all(text, "\\b\\d+\\.?\\d*\\b", " ")
  text <- stringr::str_replace_all(text, "[^a-z0-9\\s-]", " ")
  stringr::str_squish(text)
}

set.seed(42)

make_dfm <- function(texts, type = c("bow", "tfidf_uni", "tfidf_bi", "char"),
                     train_dfm = NULL) {
  type <- match.arg(type)
  texts <- as.character(texts)
  names(texts) <- paste0("doc", seq_along(texts))
  if (type == "char") {
    toks <- quanteda::tokens(texts, what = "character") |>
      quanteda::tokens_ngrams(n = 3:5, concatenator = "")
  } else {
    ngram <- if (type == "tfidf_bi") 1:2 else 1L
    toks <- quanteda::tokens(texts, remove_punct = TRUE) |>
      quanteda::tokens_remove(clinical_stop_words) |>
      quanteda::tokens_ngrams(n = ngram)
  }
  d <- quanteda::dfm(toks)
  if (!is.null(train_dfm)) {
    d <- quanteda::dfm_match(d, features = quanteda::featnames(train_dfm))
  }
  max_feat <- if (type %in% c("tfidf_bi", "char")) 5000L else 3000L
  if (is.null(train_dfm) && nfeat(d) > max_feat) {
    keep <- names(quanteda::topfeatures(d, n = max_feat))
    d <- quanteda::dfm_select(d, pattern = keep)
  }
  if (type != "bow") d <- quanteda::dfm_tfidf(d)
  d
}

# 5-fold stratified CV
folds <- rsample::vfold_cv(df, v = 5, strata = condition)

cv_one <- function(split, type, y_col) {
  tr <- rsample::analysis(split)
  te <- rsample::assessment(split)
  dtr <- make_dfm(tr$clean_text, type)
  dte <- make_dfm(te$clean_text, type, train_dfm = dtr)
  xtr <- as.matrix(quanteda::convert(dtr, to = "matrix"))
  xte <- as.matrix(quanteda::convert(dte, to = "matrix"))
  if (is.null(colnames(xtr))) colnames(xtr) <- paste0("f", seq_len(ncol(xtr)))
  if (is.null(colnames(xte))) colnames(xte) <- colnames(xtr)
  xtr[!is.finite(xtr)] <- 0
  xte[!is.finite(xte)] <- 0
  ytr <- tr[[y_col]]
  yte <- te[[y_col]]
  if (y_col == "condition") {
    ytr_f <- factor(ytr)
    yte_f <- factor(yte, levels = levels(ytr_f))
    levs <- levels(ytr_f)
    cents <- t(vapply(levs, function(lv) {
      rows <- xtr[ytr_f == lv, , drop = FALSE]
      if (!nrow(rows)) rep(0, ncol(xtr)) else colMeans(rows)
    }, numeric(ncol(xtr))))
    pred <- apply(xte, 1, function(v) {
      d <- sqrt(rowSums(sweep(cents, 2, v, "-")^2))
      levs[which.min(d)]
    })
    yardstick::f_meas_vec(yte_f, factor(pred, levels = levels(ytr_f)),
                          estimator = "macro_weighted")
  } else {
    ytr_n <- as.numeric(ytr)
    fit <- glmnet::glmnet(xtr, ytr_n, family = "binomial",
                          lambda = 0.01, maxit = 1000)
    proba <- as.numeric(predict(fit, newx = xte, type = "response"))
    yardstick::roc_auc_vec(factor(yte, levels = c(0, 1)), proba, event_level = "second")
  }
}

fit_and_score <- function(name, alpha, family = "binomial") {
  if (name == "Complement NB") {
    tmod <- quanteda.textmodels::textmodel_nb(dtr, factor(yr_tr), distribution = "Bernoulli")
    pred <- predict(tmod, newdata = dte)
    proba <- as.numeric(predict(tmod, newdata = dte, type = "probability")[, "1"])
    list(name = name, proba = proba, pred = as.integer(as.character(pred)))
  } else {
    fit <- glmnet::glmnet(xtr, yr_tr, family = family, alpha = alpha,
                          lambda = 0.01, maxit = 2000)
    proba <- as.numeric(predict(fit, newx = xte, type = "response"))
    pred  <- as.integer(proba >= 0.5)
    list(name = name, proba = proba, pred = pred)
  }
}

# Exercise 5 — regex numeric features
get_bnp <- function(text) {
  m <- stringr::str_match(text, stringr::regex("BNP\\s*(\\d{2,5})", ignore_case = TRUE))
  if (is.na(m[1, 1])) 0 else as.numeric(m[1, 2])
}
get_cr <- function(text) {
  m <- stringr::str_match(text, stringr::regex("Cr\\s*(\\d\\.\\d)", ignore_case = TRUE))
  if (is.na(m[1, 1])) 0 else as.numeric(m[1, 2])
}

x_aug_te[!is.finite(x_aug_te)] <- 0

lr_aug <- glmnet::glmnet(x_aug_tr, yr_tr, family = "binomial",
                         lambda = 0.01, maxit = 1000)
auc_aug <- yardstick::roc_auc_vec(
  factor(yr_te, levels = c(0, 1)),
  as.numeric(predict(lr_aug, newx = x_aug_te, type = "response")),
  event_level = "second"
)
cat(sprintf("Text only     : %.4f\n", auc_text))
Text only     : 0.4483
cat(sprintf("Text+BNP+Cr   : %.4f  (gain: %+.4f)\n", auc_aug, auc_aug - auc_text))
Text+BNP+Cr   : 0.4483  (gain: +0.0000)

2. G-Computation (Standardisation)

G-computation uses the g-formula:

ATE = (1/N) * sum_i [ Q(1, X_i) - Q(0, X_i) ]

Where Q(a, X) = E[Y | A=a, X] is the outcome model (Q-model).

Algorithm: 1. Fit outcome model Q on all observations 2. Predict Y(1) for everyone (set A=1) 3. Predict Y(0) for everyone (set A=0) 4. ATE = mean(Y(1)) - mean(Y(0))

g_computation <- function(df, treatment, outcome, covariates, model_type = "logistic") {
  x_cols <- c(covariates, treatment)
  form <- as.formula(paste(outcome, "~", paste(x_cols, collapse = " + ")))
  X_a1 <- df
  X_a1[[treatment]] <- 1
  X_a0 <- df
  X_a0[[treatment]] <- 0

  if (model_type == "logistic") {
    model <- glm(form, data = df, family = binomial())
    mu1 <- predict(model, newdata = X_a1, type = "response")
    mu0 <- predict(model, newdata = X_a0, type = "response")
  } else if (model_type == "gbm") {
    x_mat <- as.matrix(df[, x_cols, drop = FALSE])
    dtrain <- xgb.DMatrix(x_mat, label = df[[outcome]])
    model <- xgb.train(
      params = list(objective = "binary:logistic", max_depth = 3, eta = 0.1, nthread = 1),
      data = dtrain, nrounds = 200, verbose = 0
    )
    mu1 <- predict(model, xgb.DMatrix(as.matrix(X_a1[, x_cols, drop = FALSE])))
    mu0 <- predict(model, xgb.DMatrix(as.matrix(X_a0[, x_cols, drop = FALSE])))
  } else {
    df_rf <- df[, c(x_cols, outcome), drop = FALSE]
    df_rf[[outcome]] <- factor(df_rf[[outcome]], levels = c(0, 1))
    model <- ranger(
      formula = form, data = df_rf, probability = TRUE,
      num.trees = 200, seed = 42, num.threads = 1
    )
    mu1 <- predict(model, data = X_a1)$predictions[, "1"]
    mu0 <- predict(model, data = X_a0)$predictions[, "1"]
  }

  ate_rd  <- mean(mu1 - mu0)
  p1 <- mean(mu1); p0 <- mean(mu0)
  ate_lor <- log(p1 / (1 - p1)) - log(p0 / (1 - p0))
  list(
    ATE_RD    = round(ate_rd, 4),
    ATE_logOR = round(ate_lor, 4),
    mu1 = mu1, mu0 = mu0, model = model
  )
}

3. TMLE — Targeted Maximum Likelihood Estimation

tmle_ate <- function(df, treatment, outcome, covariates, n_folds = 5) {
  # TMLE with cross-fitting (cross-validated TMLE).
  # Combines:
  # - Q-model (outcome model)
  # - g-model (propensity score)
  # - Targeting step: fluctuates Q toward efficient influence function
  n <- nrow(df)
  A <- df[[treatment]]
  Y <- df[[outcome]]

  set.seed(42)
  fold <- sample(rep(seq_len(n_folds), length.out = n))
  mu1 <- numeric(n)
  mu0 <- numeric(n)
  ps  <- numeric(n)

  q_form <- as.formula(paste(outcome, "~", paste(c(covariates, treatment), collapse = " + ")))
  g_form <- as.formula(paste(treatment, "~", paste(covariates, collapse = " + ")))

  for (k in seq_len(n_folds)) {
    train <- fold != k
    val   <- fold == k

    q_model <- glm(q_form, data = df[train, ], family = binomial())
    new1 <- df[val, ]
    new1[[treatment]] <- 1
    new0 <- df[val, ]
    new0[[treatment]] <- 0
    mu1[val] <- predict(q_model, newdata = new1, type = "response")
    mu0[val] <- predict(q_model, newdata = new0, type = "response")

    g_model <- glm(g_form, data = df[train, ], family = binomial())
    ps[val] <- predict(g_model, newdata = df[val, ], type = "response")
  }

  ps <- pmin(pmax(ps, 1e-4), 1 - 1e-4)
  mu1 <- pmin(pmax(mu1, 1e-6), 1 - 1e-6)
  mu0 <- pmin(pmax(mu0, 1e-6), 1 - 1e-6)

  # TMLE targeting step — clever covariate H
  H1 <- A / ps                  # for treated
  H0 <- -(1 - A) / (1 - ps)     # for controls
  H  <- H1 + H0

  mu_init    <- pmin(pmax(A * mu1 + (1 - A) * mu0, 1e-6), 1 - 1e-6)
  logit_init <- logit(mu_init)

  # Fit targeting model (fluctuation) with offset = logit(Q_n)
  target_df <- tibble(Y = Y, H = H, logit_init = logit_init)
  target_model <- glm(Y ~ H - 1 + offset(logit_init),
                      data = target_df, family = binomial())
  eps <- unname(coef(target_model)[["H"]])

  # Update predictions
  mu1_star <- logistic(logit(mu1) + eps / ps)
  mu0_star <- logistic(logit(mu0) - eps / (1 - ps))
  mu1_star <- pmin(pmax(mu1_star, 1e-6), 1 - 1e-6)
  mu0_star <- pmin(pmax(mu0_star, 1e-6), 1 - 1e-6)

  # ATE estimate
  ate_tmle <- mean(mu1_star - mu0_star)

  # Efficient influence function for SE
  eif <- (A * (Y - mu1_star) / ps + mu1_star) -
    ((1 - A) * (Y - mu0_star) / (1 - ps) + mu0_star) - ate_tmle
  se_tmle <- sd(eif) / sqrt(n)

  p1 <- mean(mu1_star); p0 <- mean(mu0_star)
  ate_lor <- log(p1 / (1 - p1)) - log(p0 / (1 - p0))
  list(
    ATE_RD    = round(ate_tmle, 4),
    ATE_logOR = round(ate_lor, 4),
    SE        = round(se_tmle, 4),
    mu1_star  = mu1_star,
    mu0_star  = mu0_star
  )
}

4. Mediation Analysis — Natural Direct & Indirect Effects

# Research question: Statin -> LDL -> MI (mediated path) vs
#                   Statin -> MI directly (direct path)
# Generate mediator: LDL post-statin

# G-computation based mediation analysis.
# Decomposes total effect into natural direct (NDE) and indirect (NIE) effects.
mediation_analysis <- function(df, treatment, mediator, outcome, covariates) {
  X_base <- as.data.frame(df[, covariates, drop = FALSE])
  for (nm in names(X_base)) {
    if (anyNA(X_base[[nm]])) {
      X_base[[nm]][is.na(X_base[[nm]])] <- median(X_base[[nm]], na.rm = TRUE)
    }
  }

  # Step 1: Mediator model M ~ A, X
  med_df <- cbind(X_base, treat = df[[treatment]], M = df[[mediator]])
  names(med_df)[names(med_df) == "treat"] <- treatment
  names(med_df)[names(med_df) == "M"] <- mediator
  m_reg <- lm(as.formula(paste(mediator, "~ .")), data = med_df)

  # Step 2: Outcome model Y ~ A, M, X
  out_df <- cbind(X_base, treat = df[[treatment]],
                  M = df[[mediator]], Y = df[[outcome]])
  names(out_df)[names(out_df) == "treat"] <- treatment
  names(out_df)[names(out_df) == "M"] <- mediator
  names(out_df)[names(out_df) == "Y"] <- outcome
  q_model <- glm(as.formula(paste(outcome, "~ .")), data = out_df, family = binomial())

  predict_outcome <- function(A_val, M_vals, X) {
    X_pred <- X
    X_pred[[treatment]] <- A_val
    X_pred[[mediator]]  <- M_vals
    predict(q_model, newdata = X_pred, type = "response")
  }

  # Natural mediator values under A=1 and A=0
  X1 <- X_base; X1[[treatment]] <- 1
  X0 <- X_base; X0[[treatment]] <- 0
  M_a1 <- predict(m_reg, newdata = X1)
  M_a0 <- predict(m_reg, newdata = X0)

  # E[Y(1, M(1))], E[Y(0, M(0))], E[Y(1, M(0))], E[Y(0, M(1))]
  EY11 <- mean(predict_outcome(1, M_a1, X_base))  # Y when A=1, M=M(1)
  EY00 <- mean(predict_outcome(0, M_a0, X_base))  # Y when A=0, M=M(0)
  EY10 <- mean(predict_outcome(1, M_a0, X_base))  # Y when A=1, M=M(0)
  EY01 <- mean(predict_outcome(0, M_a1, X_base))  # Y when A=0, M=M(1)

  TE  <- EY11 - EY00   # Total effect
  NDE <- EY10 - EY00   # Natural direct effect
  NIE <- EY11 - EY10   # Natural indirect effect (through M)

  list(TE = TE, NDE = NDE, NIE = NIE, EY01 = EY01,
       prop_mediated = if (TE != 0) NIE / TE else 0)
}

5. Estimator Comparison

# Bootstrap-based comparison of all estimators
iptw_simple <- function(df_sub, covariates, treatment, outcome) {
  g_form <- as.formula(paste(treatment, "~", paste(covariates, collapse = " + ")))
  lr_ps <- glm(g_form, data = df_sub, family = binomial())
  ps <- pmin(pmax(predict(lr_ps, type = "response"), 0.05), 0.95)
  w  <- ifelse(df_sub[[treatment]] == 1, 1 / ps, 1 / (1 - ps))
  treated <- df_sub[[treatment]] == 1
  r1 <- sum(df_sub[[outcome]][treated] * w[treated]) / sum(w[treated])
  r0 <- sum(df_sub[[outcome]][!treated] * w[!treated]) / sum(w[!treated])
  log(r1 / (1 - r1)) - log(r0 / (1 - r0))
}

tmle_approx <- function(df_sub, treatment, outcome, covariates) {
  # Single-sample TMLE (no cross-fitting) for bootstrap speed
  q_form <- as.formula(paste(outcome, "~", paste(c(covariates, treatment), collapse = " + ")))
  g_form <- as.formula(paste(treatment, "~", paste(covariates, collapse = " + ")))
  q_model <- glm(q_form, data = df_sub, family = binomial())
  g_model <- glm(g_form, data = df_sub, family = binomial())
  new1 <- df_sub; new1[[treatment]] <- 1
  new0 <- df_sub; new0[[treatment]] <- 0
  mu1 <- pmin(pmax(predict(q_model, newdata = new1, type = "response"), 1e-6), 1 - 1e-6)
  mu0 <- pmin(pmax(predict(q_model, newdata = new0, type = "response"), 1e-6), 1 - 1e-6)
  ps  <- pmin(pmax(predict(g_model, type = "response"), 1e-4), 1 - 1e-4)
  A <- df_sub[[treatment]]; Y <- df_sub[[outcome]]
  H <- A / ps - (1 - A) / (1 - ps)
  mu_init <- pmin(pmax(A * mu1 + (1 - A) * mu0, 1e-6), 1 - 1e-6)
  target <- glm(Y ~ H - 1 + offset(logit(mu_init)), family = binomial())
  eps <- unname(coef(target)[1])
  mu1_star <- logistic(logit(mu1) + eps / ps)
  mu0_star <- logistic(logit(mu0) - eps / (1 - ps))
  p1 <- mean(mu1_star); p0 <- mean(mu0_star)
  log(p1 / (1 - p1)) - log(p0 / (1 - p0))
}

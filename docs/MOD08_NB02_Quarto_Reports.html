set.seed(42)
N <- 2000
df <- tibble(
  patient_id = sprintf("PT-%05d", seq_len(N) - 1),
  age        = as.integer(pmin(pmax(rnorm(N, 60, 15), 18), 90)),
  los_days   = as.integer(pmin(pmax(rgamma(N, 2.5, scale = 1.8), 1), 30)),
  diabetes   = rbinom(N, 1, 0.35),
  hf         = rbinom(N, 1, 0.22),
  smoking    = rbinom(N, 1, 0.25),
  payer      = sample(
    c("Medicare", "Medicaid", "Private", "Self-pay"),
    N, replace = TRUE, prob = c(0.40, 0.22, 0.28, 0.10)
  )
)
sigmoid <- function(x) 1 / (1 + exp(-x))
logit <- -3.2 + 0.6 * df$hf + 0.5 * df$diabetes + 0.02 * (df$age - 60) +
  0.2 * as.integer(df$los_days > 7)
df$readmitted <- rbinom(N, 1, sigmoid(logit))
cat(sprintf("Dataset: %d x %d | Readmission rate: %.1f%%\n",
            nrow(df), ncol(df), mean(df$readmitted) * 100))

2. Quarto Document Structure

A .qmd file is a Markdown file with YAML front matter and executable code chunks. This notebook is a Quarto document for R: you are looking at the same format you will use for journal-ready reports.

---                          # YAML front matter (metadata)
title: "Report Title"
format: html
engine: knitr

## Section                   # Markdown content

```{r}
#| label: fig-roc            # Cross-referenceable label
#| fig-cap: "ROC Curve"      # Caption
plot_roc_curve()
```

See @fig-roc for the discrimination results.   # Auto-numbered cross-ref

Key Quarto advantages over R Markdown / notebooks: - Clean separation of code and output - Native multi-format (HTML/PDF/Word) from same source - Built-in bibliography (.bib) support - Cross-references, callouts, panel layouts - Version-control friendly (.qmd is plain text) - Knitr engine is first-class for R ({r} chunks, echo, cache, fig-cap)

Chunk options you will use constantly:

Option Purpose
#\| label: fig-roc Cross-reference target (@fig-roc)
#\| fig-cap: Figure caption
#\| tbl-cap: Table caption
#\| echo: false Hide code in the rendered report
#\| cache: true Skip re-running unchanged chunks
#\| fig-width: / #\| fig-height: Figure size in inches
#\| message: false Suppress package startup messages

set.seed(42)
split <- initial_split(df, prop = 0.8, strata = readmitted)
train <- training(split)
test  <- testing(split)

fit <- glm(
  readmitted ~ age + los_days + diabetes + hf + smoking,
  data = train, family = binomial()
)
prob <- predict(fit, newdata = test, type = "response")
auc  <- as.numeric(pROC::auc(test$readmitted, prob, quiet = TRUE))
roc_obj <- pROC::roc(test$readmitted, prob, quiet = TRUE)
roc_df <- tibble(fpr = 1 - roc_obj$specificities, tpr = roc_obj$sensitivities)

  geom_ribbon(aes(ymin = 0, ymax = tpr), fill = col_red, alpha = 0.10) +
  geom_line(colour = col_red, linewidth = 1.1) +
  geom_abline(linetype = "dashed") +
  labs(x = "1 - Specificity", y = "Sensitivity",
       title = "A  ROC Curve", subtitle = sprintf("AUC = %.3f", auc))

5. Programmatic Table 1

# Generate publication-ready Table 1 (patient characteristics)
table1_row_continuous <- function(df_all, col, group_col = "readmitted", label = NULL) {
  label <- label %||% col
  overall <- sprintf("%.1f (%.1f)", mean(df_all[[col]]), sd(df_all[[col]]))
  g0 <- df_all[[col]][df_all[[group_col]] == 0]
  g1 <- df_all[[col]][df_all[[group_col]] == 1]
  p  <- wilcox.test(g0, g1, alternative = "two.sided")$p.value
  tibble(
    Variable = label, Overall = overall,
    `Not Readmitted` = sprintf("%.1f (%.1f)", mean(g0), sd(g0)),
    Readmitted = sprintf("%.1f (%.1f)", mean(g1), sd(g1)),
    `p-value` = sprintf("%.3f", p)
  )
}

table1_row_binary <- function(df_all, col, group_col = "readmitted", label = NULL) {
  label <- label %||% col
  overall <- sprintf("%.1f%%", mean(df_all[[col]]) * 100)
  g0 <- df_all[[col]][df_all[[group_col]] == 0]
  g1 <- df_all[[col]][df_all[[group_col]] == 1]
  ct <- table(df_all[[col]], df_all[[group_col]])
  p  <- chisq.test(ct)$p.value
  tibble(
    Variable = label, Overall = overall,
    `Not Readmitted` = sprintf("%.1f%%", mean(g0) * 100),
    Readmitted = sprintf("%.1f%%", mean(g1) * 100),
    `p-value` = sprintf("%.3f", p)
  )
}

table1 <- bind_rows(
  tibble(
    Variable = "N", Overall = as.character(nrow(df)),
    `Not Readmitted` = as.character(sum(df$readmitted == 0)),
    Readmitted = as.character(sum(df$readmitted == 1)),
    `p-value` = ""
  ),
  table1_row_continuous(df, "age", "readmitted", "Age (years), mean (SD)"),
  table1_row_continuous(df, "los_days", "readmitted", "Length of stay (days), mean (SD)"),
  table1_row_binary(df, "diabetes", "readmitted", "Diabetes, n (%)"),
  table1_row_binary(df, "hf", "readmitted", "Heart failure, n (%)"),
  table1_row_binary(df, "smoking", "readmitted", "Current smoker, n (%)")
)

Health Analytics with R · Module 06: Causal Inference in Health Research

This capstone integrates NB-01 through NB-07 into a single causal analysis pipeline.

Research Question: Does early ICU transfer (within 6 hours of ED presentation) reduce 30-day in-hospital mortality for sepsis patients?

Pipeline stages: 1. DAG specification and identification strategy 2. Propensity score estimation and balance assessment 3. Multiple causal estimators: G-comp, IPTW, AIPW, PS-matching 4. Sensitivity analysis: E-value, overlap, permutation test 5. ITS analysis of a policy change affecting ICU transfer rates 6. Mediation analysis: Is the effect mediated through organ support? 7. Publication-quality 8-panel figure at 300 dpi 8. Clinical methods narrative template

Estimated time: 4 hours | Level: Advanced

# Patient characteristics (pre-treatment)

comorbid_ct <- pmin(pmax(rpois(N, 2), 0), 6)
night_shift <- rbinom(N, 1, 0.35)   # instrument candidate

# Treatment: early ICU transfer (<6h)
# Confounded by severity (sicker = earlier transfer) and bed availability
icu_logit <- (-0.5 + 0.2 * sofa_score + 0.4 * vasopressor + 0.1 * lactate
              - 0.05 * (age - 62) / 15 + 0.6 * night_shift
              - 0.3 * comorbid_ct + rnorm(N, 0, 0.8))
early_icu <- rbinom(N, 1, logistic(icu_logit))

# Mediator: organ support initiated (intubation, vasopressors initiated, dialysis)
organ_support <- rbinom(N, 1, logistic(-1 + 0.5 * early_icu + 0.3 * sofa_score))

# Outcome: 30-day in-hospital mortality

TRUE_EFFECT_MED  <- -0.3   # portion mediated through organ support
mort_logit <- (-2.5 + TRUE_EFFECT_MAIN * early_icu + 0.15 * sofa_score
               + 0.05 * (age - 62) / 15 + 0.4 * vasopressor + 0.08 * lactate

# SMD before/after
smd <- function(df_sub, treatment, covariates, weights = NULL) {
  smds <- numeric(length(covariates))
  names(smds) <- covariates
  for (cov in covariates) {
    t <- df_sub[[cov]][df_sub[[treatment]] == 1]
    c <- df_sub[[cov]][df_sub[[treatment]] == 0]
    if (!is.null(weights)) {
      wt <- weights[df_sub[[treatment]] == 1]
      wc <- weights[df_sub[[treatment]] == 0]
      m1 <- weighted.mean(t, wt); m0 <- weighted.mean(c, wc)
      v1 <- weighted.mean((t - m1)^2, wt); v0 <- weighted.mean((c - m0)^2, wc)
    } else {
      m1 <- mean(t); m0 <- mean(c)
      v1 <- var(t); v0 <- var(c)
    }
    pool_sd <- sqrt((v1 + v0) / 2)
    smds[[cov]] <- if (pool_sd > 0) (m1 - m0) / pool_sd else 0
  }
  smds
}

3. Multiple Causal Estimators

g_comp <- function(df_sub, treat, out, cov) {
  form <- as.formula(paste(out, "~", paste(c(cov, treat), collapse = " + ")))
  lr <- glm(form, data = df_sub, family = binomial())
  X1 <- df_sub; X1[[treat]] <- 1
  X0 <- df_sub; X0[[treat]] <- 0
  mu1 <- predict(lr, newdata = X1, type = "response")
  mu0 <- predict(lr, newdata = X0, type = "response")
  rd  <- mean(mu1 - mu0)
  p1 <- mean(mu1); p0 <- mean(mu0)
  lor <- log(p1 / (1 - p1)) - log(p0 / (1 - p0))
  c(rd = rd, lor = lor)
}

iptw_est <- function(df_sub, treat, out, ps_col = "ps") {
  ps <- pmin(pmax(df_sub[[ps_col]], 0.05), 0.95)
  w  <- ifelse(df_sub[[treat]] == 1, 1 / ps, 1 / (1 - ps))
  treated <- df_sub[[treat]] == 1
  r1 <- sum(df_sub[[out]][treated] * w[treated]) / sum(w[treated])
  r0 <- sum(df_sub[[out]][!treated] * w[!treated]) / sum(w[!treated])
  rd  <- r1 - r0
  lor <- log(r1 / (1 - r1)) - log(r0 / (1 - r0))
  c(rd = rd, lor = lor)
}

aipw_est <- function(df_sub, treat, out, cov, ps_col = "ps") {
  A  <- df_sub[[treat]]
  Y  <- df_sub[[out]]
  ps <- pmin(pmax(df_sub[[ps_col]], 0.05), 0.95)
  form <- as.formula(paste(out, "~", paste(c(cov, treat), collapse = " + ")))
  lr <- glm(form, data = df_sub, family = binomial())
  X1 <- df_sub; X1[[treat]] <- 1
  X0 <- df_sub; X0[[treat]] <- 0
  mu1 <- predict(lr, newdata = X1, type = "response")
  mu0 <- predict(lr, newdata = X0, type = "response")
  aipw1 <- mu1 + A * (Y - mu1) / ps
  aipw0 <- mu0 + (1 - A) * (Y - mu0) / (1 - ps)
  rd  <- mean(aipw1 - aipw0)
  m1m <- mean(aipw1); m0m <- mean(aipw0)
  lor <- log(m1m / (1 - m1m)) - log(m0m / (1 - m0m))
  c(rd = rd, lor = lor)
}

# PS matching (nearest neighbour on logit PS, analogue of 1:1 matching)
ps_match_est <- function(df_sub, treat, out, ps_col = "ps") {
  treated <- which(df_sub[[treat]] == 1)
  control <- which(df_sub[[treat]] == 0)
  lps_t <- logit(pmin(pmax(df_sub[[ps_col]][treated], 1e-4), 1 - 1e-4))
  lps_c <- logit(pmin(pmax(df_sub[[ps_col]][control], 1e-4), 1 - 1e-4))
  # Match each treated unit to nearest control (without replacement)
  used <- logical(length(control))
  y_t <- df_sub[[out]][treated]
  y_c <- numeric(length(treated))
  n_matched <- 0L
  for (i in seq_along(treated)) {
    d <- abs(lps_t[i] - lps_c)
    d[used] <- Inf
    j <- which.min(d)
    if (is.finite(d[j])) {
      used[j] <- TRUE
      y_c[i] <- df_sub[[out]][control[j]]
      n_matched <- n_matched + 1L
    } else {
      y_c[i] <- NA_real_
    }
  }
  ok <- !is.na(y_c)
  r1 <- mean(y_t[ok]); r0 <- mean(y_c[ok])
  rd  <- r1 - r0
  lor <- log(r1 / (1 - r1)) - log(r0 / (1 - r0))
  c(rd = rd, lor = lor)
}

evalue_or <- function(OR) {
  if (OR >= 1) {
    OR + sqrt(OR * (OR - 1))
  } else {
    inv <- 1 / OR
    inv + sqrt(inv * (inv - 1))
  }
}

8. Methods Narrative Template

Causal Inference Methods

We estimated the causal effect of early ICU transfer (< 6 hours) on 30-day in-hospital mortality in a cohort of N=6,000 sepsis patients. Our identification strategy relied on the backdoor criterion, adjusting for Age, Sex, SOFA score, lactate, vasopressor initiation, and comorbidity count — all pre-treatment variables identified through a Directed Acyclic Graph.

Propensity scores were estimated using a probability random forest (ranger), producing adequate covariate balance (max IPTW-weighted SMD reported in the love plot). We applied four complementary estimators: G-computation (standardisation), IPTW (Horvitz-Thompson), AIPW (doubly robust), and nearest-neighbour PS matching — all yielding consistent estimates.

The E-value for our point estimate was computed with the EValue package (and cross-checked with sensemakr), indicating robustness to unmeasured confounding. Placebo outcome and permutation refutation tests showed no evidence of residual confounding or methodological artefact.

Mediation analysis decomposed the total effect using the natural direct/indirect effect framework (mediation package), revealing the share of the mortality benefit of early ICU transfer that operated through earlier organ support initiation.

ITS analysis of a hospital protocol mandating early ICU notification showed a significant immediate drop in sepsis mortality with an additional monthly trend improvement.

All analyses used R (ivreg, ranger, mediation, EValue, sensemakr, ggplot2).
